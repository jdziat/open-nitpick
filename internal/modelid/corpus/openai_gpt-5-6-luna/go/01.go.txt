package csvparser

import (
	"fmt"
	"io"
)

const (
	stateField = iota
	stateQuoted
	stateAfterQuote
)

// Parse parses input as delimited records using delimiter. Quoted fields may
// contain delimiters and newlines, and escaped quotes are written as "".
func Parse(input string, delimiter rune) ([][]string, error) {
	if delimiter == 0 || delimiter == '"' || delimiter == '\r' || delimiter == '\n' {
		return nil, fmt.Errorf("invalid delimiter %q", delimiter)
	}

	runes := []rune(input)
	rows := make([][]string, 0)
	row := make([]string, 0)
	field := make([]rune, 0)
	state := stateField
	fieldStarted := false

	emitRecord := func() {
		row = append(row, string(field))
		rows = append(rows, row)
		row = nil
		field = nil
		fieldStarted = false
		state = stateField
	}

	for i := 0; i < len(runes); i++ {
		ch := runes[i]

		switch state {
		case stateQuoted:
			if ch == '"' {
				if i+1 < len(runes) && runes[i+1] == '"' {
					field = append(field, '"')
					i++
				} else {
					state = stateAfterQuote
				}
			} else {
				field = append(field, ch)
			}

		case stateAfterQuote:
			switch ch {
			case delimiter:
				row = append(row, string(field))
				field = nil
				fieldStarted = false
				state = stateField
			case '\n':
				emitRecord()
			case '\r':
				if i+1 < len(runes) && runes[i+1] == '\n' {
					i++
				}
				emitRecord()
			default:
				return nil, fmt.Errorf("unexpected character %q after closing quote at position %d", ch, i)
			}

		default:
			switch ch {
			case delimiter:
				row = append(row, string(field))
				field = nil
				fieldStarted = false
			case '"':
				if fieldStarted {
					return nil, fmt.Errorf("unexpected quote at position %d", i)
				}
				fieldStarted = true
				state = stateQuoted
			case '\n':
				emitRecord()
			case '\r':
				if i+1 < len(runes) && runes[i+1] == '\n' {
					i++
				}
				emitRecord()
			default:
				field = append(field, ch)
				fieldStarted = true
			}
		}
	}

	if state == stateQuoted {
		return nil, fmt.Errorf("unterminated quoted field")
	}

	if state == stateAfterQuote || fieldStarted || len(row) > 0 {
		row = append(row, string(field))
		rows = append(rows, row)
	}

	return rows, nil
}

// ParseReader reads all data from r and parses it using delimiter.
func ParseReader(r io.Reader, delimiter rune) ([][]string, error) {
	data, err := io.ReadAll(r)
	if err != nil {
		return nil, err
	}
	return Parse(string(data), delimiter)
}
