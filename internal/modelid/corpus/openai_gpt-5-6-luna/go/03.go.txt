package retry

import (
	"errors"
	"math/rand"
	"time"
)

var (
	ErrInvalidAttempts = errors.New("retry: attempts must be greater than zero")
	ErrNilOperation    = errors.New("retry: operation must not be nil")
	ErrNegativeBackoff = errors.New("retry: backoff must not be negative")
)

const maxDuration = time.Duration(1<<63 - 1)

// Retry calls operation up to attempts times. If an attempt fails, it waits
// with exponentially increasing full-jitter backoff before trying again.
// It returns nil as soon as operation succeeds, or the last error otherwise.
func Retry(attempts int, initialBackoff time.Duration, operation func() error) error {
	if attempts <= 0 {
		return ErrInvalidAttempts
	}
	if operation == nil {
		return ErrNilOperation
	}
	if initialBackoff < 0 {
		return ErrNegativeBackoff
	}

	var lastErr error
	for attempt := 0; attempt < attempts; attempt++ {
		if err := operation(); err == nil {
			return nil
		} else {
			lastErr = err
		}

		if attempt == attempts-1 || initialBackoff == 0 {
			continue
		}

		delay := exponentialBackoff(initialBackoff, attempt)
		if delay > 0 {
			delay = time.Duration(rand.Int63n(int64(delay)))
			time.Sleep(delay)
		}
	}

	return lastErr
}

func exponentialBackoff(initial time.Duration, retryIndex int) time.Duration {
	delay := initial
	for i := 0; i < retryIndex; i++ {
		if delay > maxDuration/2 {
			return maxDuration
		}
		delay *= 2
	}
	return delay
}
