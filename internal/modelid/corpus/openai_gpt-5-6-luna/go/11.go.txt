package matrix

import "errors"

var (
	ErrShapeMismatch   = errors.New("matrix shape mismatch")
	ErrNotSquare       = errors.New("matrix is not square")
	ErrInvalidDimensions = errors.New("invalid matrix dimensions")
	ErrDataLength      = errors.New("matrix data length does not match dimensions")
	ErrIndexOutOfRange = errors.New("matrix index out of range")
)

type Matrix struct {
	Rows int
	Cols int
	Data []float64
}

func New(rows, cols int, values []float64) (Matrix, error) {
	if rows < 0 || cols < 0 {
		return Matrix{}, ErrInvalidDimensions
	}
	if len(values) != rows*cols {
		return Matrix{}, ErrDataLength
	}

	data := make([]float64, len(values))
	copy(data, values)

	return Matrix{
		Rows: rows,
		Cols: cols,
		Data: data,
	}, nil
}

func NewMatrix(rows, cols int, values []float64) (Matrix, error) {
	return New(rows, cols, values)
}

func FromRows(values [][]float64) (Matrix, error) {
	if len(values) == 0 {
		return New(0, 0, nil)
	}

	cols := len(values[0])
	data := make([]float64, 0, len(values)*cols)

	for _, row := range values {
		if len(row) != cols {
			return Matrix{}, ErrShapeMismatch
		}
		data = append(data, row...)
	}

	return New(len(values), cols, data)
}

func (m Matrix) Shape() (int, int) {
	return m.Rows, m.Cols
}

func (m Matrix) At(row, col int) (float64, error) {
	if row < 0 || row >= m.Rows || col < 0 || col >= m.Cols {
		return 0, ErrIndexOutOfRange
	}
	return m.Data[row*m.Cols+col], nil
}

func (m *Matrix) Set(row, col int, value float64) error {
	if row < 0 || row >= m.Rows || col < 0 || col >= m.Cols {
		return ErrIndexOutOfRange
	}
	m.Data[row*m.Cols+col] = value
	return nil
}

func (m Matrix) Transpose() Matrix {
	data := make([]float64, len(m.Data))
	for row := 0; row < m.Rows; row++ {
		for col := 0; col < m.Cols; col++ {
			data[col*m.Rows+row] = m.Data[row*m.Cols+col]
		}
	}

	return Matrix{
		Rows: m.Cols,
		Cols: m.Rows,
		Data: data,
	}
}

func (m Matrix) Mul(other Matrix) (Matrix, error) {
	if m.Rows < 0 || m.Cols < 0 || other.Rows < 0 || other.Cols < 0 ||
		len(m.Data) != m.Rows*m.Cols ||
		len(other.Data) != other.Rows*other.Cols {
		return Matrix{}, ErrInvalidDimensions
	}
	if m.Cols != other.Rows {
		return Matrix{}, ErrShapeMismatch
	}

	result := Matrix{
		Rows: m.Rows,
		Cols: other.Cols,
		Data: make([]float64, m.Rows*other.Cols),
	}

	for i := 0; i < m.Rows; i++ {
		for k := 0; k < m.Cols; k++ {
			left := m.Data[i*m.Cols+k]
			for j := 0; j < other.Cols; j++ {
				result.Data[i*result.Cols+j] += left * other.Data[k*other.Cols+j]
			}
		}
	}

	return result, nil
}

func (m Matrix) Multiply(other Matrix) (Matrix, error) {
	return m.Mul(other)
}

func (m Matrix) Determinant() (float64, error) {
	if m.Rows < 0 || m.Cols < 0 || len(m.Data) != m.Rows*m.Cols {
		return 0, ErrInvalidDimensions
	}
	if m.Rows != m.Cols {
		return 0, ErrNotSquare
	}

	n := m.Rows
	if n == 0 {
		return 1, nil
	}

	work := make([]float64, len(m.Data))
	copy(work, m.Data)

	determinant := 1.0
	sign := 1.0

	for col := 0; col < n; col++ {
		pivot := col
		for row := col + 1; row < n; row++ {
			if abs(work[row*n+col]) > abs(work[pivot*n+col]) {
				pivot = row
			}
		}

		pivotValue := work[pivot*n+col]
		if pivotValue == 0 {
			return 0, nil
		}

		if pivot != col {
			for j := 0; j < n; j++ {
				work[col*n+j], work[pivot*n+j] =
					work[pivot*n+j], work[col*n+j]
			}
			sign = -sign
		}

		determinant *= work[col*n+col]

		for row := col + 1; row < n; row++ {
			factor := work[row*n+col] / work[col*n+col]
			for j := col + 1; j < n; j++ {
				work[row*n+j] -= factor * work[col*n+j]
			}
		}
	}

	return determinant * sign, nil
}

func abs(value float64) float64 {
	if value < 0 {
		return -value
	}
	return value
}
