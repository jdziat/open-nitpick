package slugify

import (
	"strings"
	"unicode"
)

func Slugify(text string) string {
	text = strings.ToLower(text)

	var result strings.Builder
	pendingHyphen := false

	for _, r := range text {
		if unicode.IsLetter(r) || unicode.IsNumber(r) {
			if pendingHyphen && result.Len() > 0 {
				result.WriteByte('-')
			}
			result.WriteRune(r)
			pendingHyphen = false
		} else if result.Len() > 0 {
			pendingHyphen = true
		}
	}

	return result.String()
}

func slugify(text string) string {
	return Slugify(text)
}
