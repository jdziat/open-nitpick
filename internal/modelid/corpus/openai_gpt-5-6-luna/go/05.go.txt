package main

import (
	"flag"
	"fmt"
	"io"
	"os"
	"sort"
	"unicode"
)

type WordCount struct {
	Word  string
	Count int
}

func TopKWords(text string, k int) []WordCount {
	if k <= 0 {
		return []WordCount{}
	}

	counts := make(map[string]int)
	word := make([]rune, 0, 32)

	flush := func() {
		if len(word) == 0 {
			return
		}
		counts[string(word)]++
		word = word[:0]
	}

	for _, r := range text {
		if unicode.IsLetter(r) || unicode.IsDigit(r) || unicode.IsMark(r) {
			word = append(word, unicode.ToLower(r))
		} else {
			flush()
		}
	}
	flush()

	words := make([]WordCount, 0, len(counts))
	for w, c := range counts {
		words = append(words, WordCount{Word: w, Count: c})
	}

	sort.Slice(words, func(i, j int) bool {
		if words[i].Count != words[j].Count {
			return words[i].Count > words[j].Count
		}
		return words[i].Word < words[j].Word
	})

	if k > len(words) {
		k = len(words)
	}
	return words[:k]
}

func main() {
	k := flag.Int("k", 10, "number of words to return")
	flag.Parse()

	text, err := io.ReadAll(os.Stdin)
	if err != nil {
		fmt.Fprintln(os.Stderr, err)
		os.Exit(1)
	}

	for _, result := range TopKWords(string(text), *k) {
		fmt.Printf("%s\t%d\n", result.Word, result.Count)
	}
}
