package config

import (
	"encoding/json"
	"errors"
	"fmt"
	"io"
	"os"
	"strconv"
	"strings"
)

type Config struct {
	Host        string `json:"host"`
	Port        int    `json:"port"`
	Debug       bool   `json:"debug"`
	DatabaseURL string `json:"database_url"`
	APIKey      string `json:"api_key"`
}

func DefaultConfig() Config {
	return Config{
		Host:  "127.0.0.1",
		Port:  8080,
		Debug: false,
	}
}

func Load(path string) (*Config, error) {
	data, err := os.ReadFile(path)
	if err != nil {
		return nil, fmt.Errorf("read configuration file: %w", err)
	}

	return LoadBytes(data)
}

func LoadBytes(data []byte) (*Config, error) {
	cfg := DefaultConfig()

	decoder := json.NewDecoder(strings.NewReader(string(data)))
	decoder.DisallowUnknownFields()

	if err := decoder.Decode(&cfg); err != nil {
		return nil, fmt.Errorf("decode configuration: %w", err)
	}

	var extra any
	if err := decoder.Decode(&extra); err != io.EOF {
		if err == nil {
			return nil, errors.New("configuration contains multiple JSON values")
		}
		return nil, fmt.Errorf("decode configuration: %w", err)
	}

	if err := applyEnvironment(&cfg); err != nil {
		return nil, err
	}

	if err := cfg.Validate(); err != nil {
		return nil, err
	}

	return &cfg, nil
}

func LoadReader(r io.Reader) (*Config, error) {
	data, err := io.ReadAll(r)
	if err != nil {
		return nil, fmt.Errorf("read configuration: %w", err)
	}
	return LoadBytes(data)
}

func (c Config) Validate() error {
	var missing []string

	if strings.TrimSpace(c.DatabaseURL) == "" {
		missing = append(missing, "database_url")
	}
	if strings.TrimSpace(c.APIKey) == "" {
		missing = append(missing, "api_key")
	}

	if len(missing) > 0 {
		return fmt.Errorf("missing required configuration keys: %s", strings.Join(missing, ", "))
	}
	if c.Port < 1 || c.Port > 65535 {
		return fmt.Errorf("port must be between 1 and 65535, got %d", c.Port)
	}
	if strings.TrimSpace(c.Host) == "" {
		return errors.New("host must not be empty")
	}

	return nil
}

func applyEnvironment(c *Config) error {
	if value, ok := os.LookupEnv("APP_HOST"); ok {
		c.Host = value
	}
	if value, ok := os.LookupEnv("APP_DATABASE_URL"); ok {
		c.DatabaseURL = value
	}
	if value, ok := os.LookupEnv("APP_API_KEY"); ok {
		c.APIKey = value
	}
	if value, ok := os.LookupEnv("APP_PORT"); ok {
		port, err := strconv.Atoi(value)
		if err != nil {
			return fmt.Errorf("invalid APP_PORT value %q: %w", value, err)
		}
		c.Port = port
	}
	if value, ok := os.LookupEnv("APP_DEBUG"); ok {
		debug, err := strconv.ParseBool(value)
		if err != nil {
			return fmt.Errorf("invalid APP_DEBUG value %q: %w", value, err)
		}
		c.Debug = debug
	}

	return nil
}
