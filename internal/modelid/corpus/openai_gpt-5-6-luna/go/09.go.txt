package daterange

import "time"

// Iterator yields each calendar day from start through end, inclusively.
type Iterator struct {
	current      time.Time
	end          time.Time
	step         int
	skipWeekends bool
	done         bool
}

// New creates an iterator over the inclusive date range from start to end.
// If skipWeekends is true, Saturdays and Sundays are omitted.
func New(start, end time.Time, skipWeekends bool) *Iterator {
	loc := start.Location()
	startDate := time.Date(start.Year(), start.Month(), start.Day(), 0, 0, 0, 0, loc)
	endDate := time.Date(end.Year(), end.Month(), end.Day(), 0, 0, 0, 0, loc)

	step := 1
	if startDate.After(endDate) {
		step = -1
	}

	return &Iterator{
		current:      startDate,
		end:          endDate,
		step:         step,
		skipWeekends: skipWeekends,
	}
}

// Next returns the next date in the range and true, or the zero time and
// false when the iterator is exhausted.
func (it *Iterator) Next() (time.Time, bool) {
	if it.done {
		return time.Time{}, false
	}

	for {
		if (it.step > 0 && it.current.After(it.end)) ||
			(it.step < 0 && it.current.Before(it.end)) {
			it.done = true
			return time.Time{}, false
		}

		date := it.current
		it.current = it.current.AddDate(0, 0, it.step)

		if it.skipWeekends &&
			(date.Weekday() == time.Saturday || date.Weekday() == time.Sunday) {
			continue
		}

		return date, true
	}
}

// Dates returns a channel that receives each date in the inclusive range.
func Dates(start, end time.Time, skipWeekends bool) <-chan time.Time {
	ch := make(chan time.Time)
	go func() {
		defer close(ch)

		it := New(start, end, skipWeekends)
		for {
			date, ok := it.Next()
			if !ok {
				return
			}
			ch <- date
		}
	}()
	return ch
}
