import argparse
import copy
import json
import os
import re
import sys
from collections.abc import Mapping
from typing import Any, Dict, Iterable, Optional


class ConfigError(Exception):
    """Raised when configuration loading or validation fails."""


_MISSING = object()


def _deep_merge(base: Mapping[str, Any], override: Mapping[str, Any]) -> Dict[str, Any]:
    result = copy.deepcopy(dict(base))

    for key, value in override.items():
        if (
            key in result
            and isinstance(result[key], Mapping)
            and isinstance(value, Mapping)
        ):
            result[key] = _deep_merge(result[key], value)
        else:
            result[key] = copy.deepcopy(value)

    return result


def _get_path(config: Mapping[str, Any], path: str) -> Any:
    current: Any = config

    for part in path.split("."):
        if not isinstance(current, Mapping) or part not in current:
            return _MISSING
        current = current[part]

    return current


def _set_path(config: Dict[str, Any], path: str, value: Any) -> None:
    parts = path.split(".")
    current = config

    for part in parts[:-1]:
        existing = current.get(part)
        if not isinstance(existing, dict):
            existing = {}
            current[part] = existing
        current = existing

    current[parts[-1]] = value


def _leaf_paths(value: Any, prefix: str = "") -> Iterable[str]:
    if isinstance(value, Mapping):
        for key, child in value.items():
            path = f"{prefix}.{key}" if prefix else str(key)
            yield from _leaf_paths(child, path)
    elif prefix:
        yield prefix


def _environment_name(prefix: str, path: str) -> str:
    parts = []
    for part in path.split("."):
        normalized = re.sub(r"[^A-Za-z0-9]+", "_", part).strip("_")
        parts.append(normalized.upper())
    return prefix + "_".join(parts)


def _parse_environment_value(value: str, existing: Any = _MISSING) -> Any:
    if existing is bool:
        normalized = value.strip().lower()
        if normalized in {"true", "1", "yes", "on"}:
            return True
        if normalized in {"false", "0", "no", "off"}:
            return False
        raise ConfigError(f"invalid boolean environment value: {value!r}")

    if existing is int and not isinstance(existing, bool):
        try:
            return int(value)
        except ValueError as exc:
            raise ConfigError(f"invalid integer environment value: {value!r}") from exc

    if existing is float:
        try:
            return float(value)
        except ValueError as exc:
            raise ConfigError(f"invalid numeric environment value: {value!r}") from exc

    if isinstance(existing, (dict, list)):
        try:
            return json.loads(value)
        except json.JSONDecodeError as exc:
            raise ConfigError(
                f"environment value for a structured setting must be valid JSON: {value!r}"
            ) from exc

    if existing is not _MISSING and existing is not None:
        return value

    try:
        return json.loads(value)
    except json.JSONDecodeError:
        return value


def load_config(
    path: str,
    defaults: Optional[Mapping[str, Any]] = None,
    required_keys: Optional[Iterable[str]] = None,
    env_prefix: str = "APP_",
    environ: Optional[Mapping[str, str]] = None,
) -> Dict[str, Any]:
    """
    Load a JSON object, merge it over defaults, apply environment overrides,
    and validate required dotted keys.

    Environment variables are named using uppercase path components. For
    example, "database.host" with the default prefix becomes
    "APP_DATABASE_HOST".
    """
    try:
        with open(path, "r", encoding="utf-8") as file:
            loaded = json.load(file)
    except FileNotFoundError as exc:
        raise ConfigError(f"configuration file not found: {path}") from exc
    except json.JSONDecodeError as exc:
        raise ConfigError(f"invalid JSON in configuration file {path}: {exc}") from exc
    except OSError as exc:
        raise ConfigError(f"unable to read configuration file {path}: {exc}") from exc

    if not isinstance(loaded, Mapping):
        raise ConfigError("configuration file must contain a JSON object")

    if defaults is not None and not isinstance(defaults, Mapping):
        raise ConfigError("defaults must be a mapping")

    config = _deep_merge(defaults or {}, loaded)
    required = list(required_keys or ())
    paths = set(_leaf_paths(config))
    paths.update(required)

    environment = environ if environ is not None else os.environ

    for path_name in paths:
        variable = _environment_name(env_prefix, path_name)
        if variable in environment:
            existing = _get_path(config, path_name)
            value = _parse_environment_value(environment[variable], existing)
            _set_path(config, path_name, value)

    missing = [
        key
        for key in required
        if _get_path(config, key) is _MISSING or _get_path(config, key) is None
    ]
    if missing:
        raise ConfigError("missing required configuration keys: " + ", ".join(missing))

    return config


def main() -> int:
    parser = argparse.ArgumentParser(description="Load and validate a JSON configuration")
    parser.add_argument("config", help="path to the JSON configuration file")
    parser.add_argument(
        "--default",
        dest="defaults",
        help="path to an optional JSON file containing defaults",
    )
    parser.add_argument(
        "--required",
        action="append",
        default=[],
        help="required key, using dotted notation for nested keys; may be repeated",
    )
    parser.add_argument("--env-prefix", default="APP_")
    args = parser.parse_args()

    defaults = None
    if args.defaults:
        try:
            with open(args.defaults, "r", encoding="utf-8") as file:
                defaults = json.load(file)
        except (OSError, json.JSONDecodeError) as exc:
            print(f"error loading defaults: {exc}", file=sys.stderr)
            return 1

    try:
        config = load_config(
            args.config,
            defaults=defaults,
            required_keys=args.required,
            env_prefix=args.env_prefix,
        )
    except ConfigError as exc:
        print(f"error: {exc}", file=sys.stderr)
        return 1

    json.dump(config, sys.stdout, indent=2, sort_keys=True)
    sys.stdout.write("\n")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
