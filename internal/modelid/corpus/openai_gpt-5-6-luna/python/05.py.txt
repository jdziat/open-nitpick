from collections import Counter
import argparse
import sys


def top_k_words(text: str, k: int):
    """Return the K most frequent words as (word, count) pairs."""
    if not isinstance(text, str):
        raise TypeError("text must be a string")
    if not isinstance(k, int):
        raise TypeError("k must be an integer")
    if k <= 0:
        return []

    words = []
    for character in text.casefold():
        if character.isalnum():
            words.append(character)
        else:
            words.append(" ")

    counts = Counter("".join(words).split())
    return sorted(counts.items(), key=lambda item: (-item[1], item[0]))[:k]


def count_words(text: str, k: int):
    return top_k_words(text, k)


def main():
    parser = argparse.ArgumentParser(
        description="Count words and display the top K results."
    )
    parser.add_argument("k", type=int, help="number of words to return")
    parser.add_argument(
        "file",
        nargs="?",
        help="input file; reads from standard input when omitted",
    )
    args = parser.parse_args()

    if args.file:
        with open(args.file, "r", encoding="utf-8") as input_file:
            text = input_file.read()
    else:
        text = sys.stdin.read()

    for word, count in top_k_words(text, args.k):
        print(f"{word}\t{count}")


if __name__ == "__main__":
    main()
