def parse_csv(text: str, delimiter: str = ",") -> list[list[str]]:
    """Parse CSV text into rows of strings."""

    if not isinstance(text, str):
        raise TypeError("text must be a string")
    if not isinstance(delimiter, str) or len(delimiter) != 1:
        raise ValueError("delimiter must be a single character")
    if delimiter in {'"', "\r", "\n"}:
        raise ValueError("delimiter cannot be a quote or line break")

    rows: list[list[str]] = []
    row: list[str] = []
    field: list[str] = []
    state = "start"
    i = 0

    while i < len(text):
        char = text[i]

        if state == "quoted":
            if char == '"':
                state = "after_quote"
            else:
                field.append(char)
            i += 1
            continue

        if state == "after_quote":
            if char == '"':
                field.append('"')
                state = "quoted"
                i += 1
            elif char == delimiter:
                row.append("".join(field))
                field.clear()
                state = "start"
                i += 1
            elif char == "\r" or char == "\n":
                row.append("".join(field))
                field.clear()
                rows.append(row)
                row = []
                state = "start"
                if char == "\r" and i + 1 < len(text) and text[i + 1] == "\n":
                    i += 1
                i += 1
            else:
                raise ValueError(f"unexpected character after closing quote at position {i}")
            continue

        if char == delimiter:
            if state == "start":
                row.append("")
            else:
                row.append("".join(field))
                field.clear()
            state = "start"
            i += 1
        elif char == '"':
            if state != "start":
                raise ValueError(f"unexpected quote in unquoted field at position {i}")
            state = "quoted"
            i += 1
        elif char == "\r" or char == "\n":
            if state == "start":
                if row:
                    row.append("")
            else:
                row.append("".join(field))
                field.clear()
            rows.append(row)
            row = []
            state = "start"
            if char == "\r" and i + 1 < len(text) and text[i + 1] == "\n":
                i += 1
            i += 1
        else:
            field.append(char)
            state = "unquoted"
            i += 1

    if state == "quoted":
        raise ValueError("unterminated quoted field")

    if state == "after_quote":
        row.append("".join(field))
    elif state == "unquoted":
        row.append("".join(field))
    elif row:
        row.append("")

    if row:
        rows.append(row)

    return rows


__all__ = ["parse_csv"]
