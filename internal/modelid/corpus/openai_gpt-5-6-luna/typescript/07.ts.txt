import * as fs from "node:fs";

export type JsonPrimitive = string | number | boolean | null;
export type JsonValue = JsonPrimitive | JsonValue[] | { [key: string]: JsonValue };

export interface LoadConfigOptions<T extends Record<string, unknown> = Record<string, unknown>> {
  defaults?: Partial<T>;
  requiredKeys?: string[];
  env?: NodeJS.ProcessEnv;
  envPrefix?: string;
}

function isPlainObject(value: unknown): value is Record<string, unknown> {
  return typeof value === "object" && value !== null && !Array.isArray(value);
}

function clone<T>(value: T): T {
  if (Array.isArray(value)) {
    return value.map(clone) as T;
  }

  if (isPlainObject(value)) {
    const result: Record<string, unknown> = {};
    for (const [key, entry] of Object.entries(value)) {
      result[key] = clone(entry);
    }
    return result as T;
  }

  return value;
}

function merge(
  base: Record<string, unknown>,
  override: Record<string, unknown>
): Record<string, unknown> {
  const result: Record<string, unknown> = clone(base);

  for (const [key, value] of Object.entries(override)) {
    if (isPlainObject(result[key]) && isPlainObject(value)) {
      result[key] = merge(result[key] as Record<string, unknown>, value);
    } else {
      result[key] = clone(value);
    }
  }

  return result;
}

function getPath(
  object: Record<string, unknown>,
  path: string
): unknown {
  return path.split(".").reduce<unknown>((current, segment) => {
    if (!isPlainObject(current)) {
      return undefined;
    }
    return current[segment];
  }, object);
}

function setPath(
  object: Record<string, unknown>,
  path: string,
  value: unknown
): void {
  const segments = path.split(".");
  let current = object;

  for (let index = 0; index < segments.length - 1; index++) {
    const segment = segments[index];
    if (!isPlainObject(current[segment])) {
      current[segment] = {};
    }
    current = current[segment] as Record<string, unknown>;
  }

  current[segments[segments.length - 1]] = value;
}

function collectLeafPaths(
  value: unknown,
  prefix = ""
): string[] {
  if (!isPlainObject(value)) {
    return prefix ? [prefix] : [];
  }

  const paths: string[] = [];

  for (const [key, child] of Object.entries(value)) {
    const path = prefix ? `${prefix}.${key}` : key;

    if (isPlainObject(child)) {
      paths.push(...collectLeafPaths(child, path));
    } else {
      paths.push(path);
    }
  }

  return paths;
}

function toEnvironmentName(path: string, prefix: string): string {
  const normalized = path
    .replace(/([a-z0-9])([A-Z])/g, "$1_$2")
    .replace(/\./g, "__")
    .replace(/[^a-zA-Z0-9_]/g, "_")
    .toUpperCase();

  return `${prefix}${normalized}`;
}

function parseEnvironmentValue(
  rawValue: string,
  existingValue: unknown
): unknown {
  if (typeof existingValue === "boolean") {
    const normalized = rawValue.trim().toLowerCase();
    if (normalized === "true") return true;
    if (normalized === "false") return false;
  }

  if (typeof existingValue === "number") {
    const number = Number(rawValue);
    if (!Number.isNaN(number)) return number;
  }

  if (typeof existingValue === "bigint") {
    try {
      return BigInt(rawValue);
    } catch {
      return rawValue;
    }
  }

  if (existingValue !== undefined && typeof existingValue === "string") {
    return rawValue;
  }

  try {
    return JSON.parse(rawValue);
  } catch {
    return rawValue;
  }
}

function validateRequiredKeys(
  config: Record<string, unknown>,
  requiredKeys: string[]
): void {
  const missing = requiredKeys.filter((key) => {
    const value = getPath(config, key);

    return (
      value === undefined ||
      value === null ||
      (typeof value === "string" && value.trim().length === 0)
    );
  });

  if (missing.length > 0) {
    throw new Error(`Missing required configuration key(s): ${missing.join(", ")}`);
  }
}

export function loadConfig<T extends Record<string, unknown> = Record<string, unknown>>(
  filePath: string,
  options: LoadConfigOptions<T> = {}
): T {
  let fileConfig: unknown;

  try {
    fileConfig = JSON.parse(fs.readFileSync(filePath, "utf8"));
  } catch (error) {
    if (error instanceof SyntaxError) {
      throw new Error(`Invalid JSON configuration file: ${filePath}`);
    }
    throw new Error(
      `Unable to read configuration file "${filePath}": ${
        error instanceof Error ? error.message : String(error)
      }`
    );
  }

  if (!isPlainObject(fileConfig)) {
    throw new Error("Configuration file must contain a JSON object");
  }

  const config = merge(
    isPlainObject(options.defaults) ? options.defaults : {},
    fileConfig
  );

  const paths = new Set([
    ...collectLeafPaths(config),
    ...(options.requiredKeys ?? [])
  ]);

  const environment = options.env ?? process.env;
  const prefix = options.envPrefix ?? "";

  for (const path of paths) {
    const environmentName = toEnvironmentName(path, prefix);
    const rawValue = environment[environmentName];

    if (rawValue !== undefined) {
      setPath(
        config,
        path,
        parseEnvironmentValue(rawValue, getPath(config, path))
      );
    }
  }

  validateRequiredKeys(config, options.requiredKeys ?? []);

  return config as T;
}

export default loadConfig;
