type Waiter = {
  amount: number;
  resolve: () => void;
};

export class TokenBucketRateLimiter {
  private readonly capacity: number;
  private readonly refillRatePerSecond: number;
  private tokens: number;
  private lastRefillTime: number;
  private readonly waiters: Waiter[] = [];
  private timer: ReturnType<typeof setTimeout> | null = null;

  public constructor(capacity: number, refillRatePerSecond: number) {
    if (!Number.isFinite(capacity) || capacity <= 0) {
      throw new RangeError("capacity must be a finite number greater than zero");
    }

    if (!Number.isFinite(refillRatePerSecond) || refillRatePerSecond <= 0) {
      throw new RangeError(
        "refillRatePerSecond must be a finite number greater than zero",
      );
    }

    this.capacity = capacity;
    this.refillRatePerSecond = refillRatePerSecond;
    this.tokens = capacity;
    this.lastRefillTime = Date.now();
  }

  public get availableTokens(): number {
    this.refill();
    return this.tokens;
  }

  public tryAcquire(amount = 1): boolean {
    this.validateAmount(amount);
    this.refill();

    if (this.waiters.length > 0) {
      this.pump();
      return false;
    }

    if (this.tokens < amount) {
      return false;
    }

    this.tokens -= amount;
    return true;
  }

  public acquire(amount = 1): Promise<void> {
    this.validateAmount(amount);

    return new Promise<void>((resolve) => {
      this.waiters.push({ amount, resolve });
      this.pump();
    });
  }

  private validateAmount(amount: number): void {
    if (!Number.isFinite(amount) || amount <= 0 || amount > this.capacity) {
      throw new RangeError(
        "amount must be finite, greater than zero, and no greater than capacity",
      );
    }
  }

  private refill(): void {
    const now = Date.now();
    const elapsedSeconds = Math.max(0, (now - this.lastRefillTime) / 1000);

    if (elapsedSeconds > 0) {
      this.tokens = Math.min(
        this.capacity,
        this.tokens + elapsedSeconds * this.refillRatePerSecond,
      );
      this.lastRefillTime = now;
    }
  }

  private pump(): void {
    if (this.timer !== null) {
      clearTimeout(this.timer);
      this.timer = null;
    }

    this.refill();

    while (this.waiters.length > 0) {
      const waiter = this.waiters[0];

      if (this.tokens < waiter.amount) {
        break;
      }

      this.tokens -= waiter.amount;
      this.waiters.shift();
      waiter.resolve();
    }

    if (this.waiters.length === 0) {
      return;
    }

    const requiredTokens = this.waiters[0].amount - this.tokens;
    const delayMilliseconds = Math.max(
      1,
      Math.ceil((requiredTokens / this.refillRatePerSecond) * 1000),
    );

    this.timer = setTimeout(
      () => {
        this.timer = null;
        this.pump();
      },
      Math.min(delayMilliseconds, 2_147_483_647),
    );
  }
}

export default TokenBucketRateLimiter;
