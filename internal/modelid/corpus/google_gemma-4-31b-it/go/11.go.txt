package main

import (
	"errors"
	"fmt"
)

type Matrix [][]float64

func NewMatrix(rows, cols int) Matrix {
	m := make(Matrix, rows)
	for i := range m {
		m[i] = make([]float64, cols)
	}
	return m
}

func (m Matrix) Rows() int { return len(m) }
func (m Matrix) Cols() int { return len(m[0]) }

func (m Matrix) Multiply(other Matrix) (Matrix, error) {
	if m.Cols() != other.Rows() {
		return nil, errors.New("matrix dimension mismatch for multiplication")
	}
	res := NewMatrix(m.Rows(), other.Cols())
	for i := 0; i < m.Rows(); i++ {
		for j := 0; j < other.Cols(); j++ {
			for k := 0; k < m.Cols(); k++ {
				res[i][j] += m[i][k] * other[k][j]
			}
		}
	}
	return res, nil
}

func (m Matrix) Transpose() Matrix {
	res := NewMatrix(m.Cols(), m.Rows())
	for i := 0; i < m.Rows(); i++ {
		for j := 0; j < m.Cols(); j++ {
			res[j][i] = m[i][j]
		}
	}
	return res
}

func (m Matrix) Determinant() (float64, error) {
	rows := m.Rows()
	if rows == 0 || rows != m.Cols() {
		return 0, errors.New("determinant requires a non-empty square matrix")
	}

	if rows == 1 {
		return m[0][0], nil
	}

	if rows == 2 {
		return m[0][0]*m[1][1] - m[0][1]*m[1][0], nil
	}

	det := 0.0
	for i := 0; i < rows; i++ {
		sub := NewMatrix(rows-1, rows-1)
		for r := 1; r < rows; r++ {
			colIdx := 0
			for c := 0; c < rows; c++ {
				if c == i {
					continue
				}
				sub[r-1][colIdx] = m[r][c]
				colIdx++
			}
		}
		subDet, _ := sub.Determinant()
		if i%2 == 1 {
			det -= m[0][i] * subDet
		} else {
			det += m[0][i] * subDet
		}
	}
	return det, nil
}

func main() {
	a := Matrix{{1, 2}, {3, 4}}
	b := Matrix{{5, 6}, {7, 8}}

	prod, _ := a.Multiply(b)
	fmt.Println("Product:", prod)

	trans := a.Transpose()
	fmt.Println("Transpose:", trans)

	det, _ := a.Determinant()
	fmt.Println("Determinant:", det)
}
