package main

import (
	"bufio"
	"fmt"
	"io"
	"strings"
)

type CSVParser struct {
	Delimiter rune
}

func NewCSVParser(delimiter rune) *CSVParser {
	return &CSVParser{Delimiter: delimiter}
}

func (p *CSVParser) Parse(r io.Reader) ([][]string, error) {
	var rows [][]string
	reader := bufio.NewReader(r)
	var currentRow []string
	var currentField strings.Builder
	inQuotes := false

	for {
		rn, _, err := reader.ReadRune()
		if err != nil {
			if err == io.EOF {
				break
			}
			return nil, err
		}

		if inQuotes {
			if rn == '"' {
				nextRn, _, err := reader.ReadRune()
				if err == nil && nextRn == '"' {
					currentField.WriteRune('"')
				} else {
					inQuotes = false
					if err == nil {
						reader.UnreadRune()
					}
				}
			} else {
				currentField.WriteRune(rn)
			}
		} else {
			switch rn {
			case '"':
				inQuotes = true
			case p.Delimiter:
				currentRow = append(currentRow, currentField.String())
				currentField.Reset()
			case '\n':
				currentRow = append(currentRow, currentField.String())
				rows = append(rows, currentRow)
				currentRow = nil
				currentField.Reset()
			case '\r':
				nextRn, _, err := reader.ReadRune()
				if err == nil && nextRn == '\n' {
					// skip \n
				} else if err == nil {
					reader.UnreadRune()
				}
				currentRow = append(currentRow, currentField.String())
				rows = append(rows, currentRow)
				currentRow = nil
				currentField.Reset()
			default:
				currentField.WriteRune(rn)
			}
		}
	}

	if currentField.Len() > 0 || len(currentRow) > 0 {
		currentRow = append(currentRow, currentField.String())
		rows = append(rows, currentRow)
	}

	return rows, nil
}

func main() {
	input := "Name,Note\n\"John \"\"Big\"\" Doe\",Hello\n\"Jane\",World"
	parser := NewCSVParser(',')
	rows, err := parser.Parse(strings.NewReader(input))
	if err != nil {
		fmt.Println("Error:", err)
		return
	}
	for _, row := range rows {
		fmt.Printf("%#v\n", row)
	}
}
