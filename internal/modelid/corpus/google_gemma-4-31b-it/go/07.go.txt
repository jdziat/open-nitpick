package main

import (
	"encoding/json"
	"fmt"
	"os"
	"reflect"
	"strings"
)

type Config struct {
	Port    int    `json:"port" env:"APP_PORT" default:"8080"`
	APIKey  string `json:"api_key" env:"APP_API_KEY" required:"true"`
	Timeout int    `json:"timeout" env:"APP_TIMEOUT" default:"30"`
}

func LoadConfig(path string) (*Config, error) {
	cfg := &Config{}
	
	// 1. Load Defaults
	if err := setDefaults(cfg); err != nil {
		return nil, err
	}

	// 2. Load JSON file
	file, err := os.ReadFile(path)
	if err == nil {
		if err := json.Unmarshal(file, cfg); err != nil {
			return nil, fmt.Errorf("json error: %w", err)
		}
	}

	// 3. Override with Env Vars
	if err := overrideFromEnv(cfg); err != nil {
		return nil, err
	}

	// 4. Validate Required
	if err := validateRequired(cfg); err != nil {
		return nil, err
	}

	return cfg, nil
}

func setDefaults(cfg interface{}) error {
	v := reflect.ValueOf(cfg).Elem()
	t := v.Type()

	for i := 0; i < v.NumField(); i++ {
		field := v.Field(i)
		tag := t.Field(i).Tag.Get("default")
		if tag == "" {
			continue
		}

		switch field.Kind() {
		case reflect.String:
			field.SetString(tag)
		case reflect.Int:
			var val int
			fmt.Sscanf(tag, "%d", &val)
			field.SetInt(int64(val))
		}
	}
	return nil
}

func overrideFromEnv(cfg interface{}) error {
	v := reflect.ValueOf(cfg).Elem()
	t := v.Type()

	for i := 0; i < v.NumField(); i++ {
		envKey := t.Field(i).Tag.Get("env")
		if envKey == "" {
			continue
		}

		if val, ok := os.LookupEnv(envKey); ok {
			field := v.Field(i)
			switch field.Kind() {
			case reflect.String:
				field.SetString(val)
			case reflect.Int:
				var intVal int
				fmt.Sscanf(val, "%d", &intVal)
				field.SetInt(int64(intVal))
			}
		}
	}
	return nil
}

func validateRequired(cfg interface{}) error {
	v := reflect.ValueOf(cfg).Elem()
	t := v.Type()

	for i := 0; i < v.NumField(); i++ {
		req := t.Field(i).Tag.Get("required")
		if req == "true" {
			field := v.Field(i)
			if field.IsZero() {
				return fmt.Errorf("missing required field: %s", t.Field(i).Name)
			}
		}
	}
	return nil
}

func main() {
	// Create a dummy config file for demonstration
	dummyJSON := `{"port": 9000}`
	os.WriteFile("config.json", []byte(dummyJSON), 0644)
	defer os.Remove("config.json")

	// Set required env var
	os.Setenv("APP_API_KEY", "secret-123")

	cfg, err := LoadConfig("config.json")
	if err != nil {
		fmt.Printf("Error loading config: %v\n", err)
		os.Exit(1)
	}

	fmt.Printf("Config Loaded: %+v\n", cfg)
}
