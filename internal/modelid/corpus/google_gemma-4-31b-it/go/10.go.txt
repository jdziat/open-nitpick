package main

import (
	"crypto/sha256"
	"encoding/base64"
	"fmt"
	"sync"
	"time"
)

type Entry struct {
	URL    string
	Expiry time.Time
}

type Shortener struct {
	mu    sync.RWMutex
	store map[string]Entry
}

func NewShortener() *Shortener {
	return &Shortener{
		store: make(map[string]Entry),
	}
}

func (s *Shortener) Shorten(url string, duration time.Duration) string {
	hash := sha256.Sum256([]byte(url))
	code := base64.URLEncoding.EncodeToString(hash[:8])

	s.mu.Lock()
	defer s.mu.Unlock()

	s.store[code] = Entry{
		URL:    url,
		Expiry: time.Now().Add(duration),
	}
	return code
}

func (s *Shortener) Resolve(code string) (string, bool) {
	s.mu.RLock()
	entry, exists := s.store[code]
	s.mu.RUnlock()

	if !exists || time.Now().After(entry.Expiry) {
		return "", false
	}
	return entry.URL, true
}

func main() {
	s := NewShortener()
	
	url := "https://www.google.com"
	code := s.Shorten(url, 2*time.Second)
	fmt.Printf("Short code: %s\n", code)

	if res, ok := s.Resolve(code); ok {
		fmt.Printf("Resolved: %s\n", res)
	}

	fmt.Println("Waiting for expiry...")
	time.Sleep(3 * time.Second)

	if _, ok := s.Resolve(code); !ok {
		fmt.Println("URL expired successfully")
	}
}
