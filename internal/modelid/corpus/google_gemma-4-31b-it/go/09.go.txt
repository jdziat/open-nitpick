package main

import (
	"fmt"
	"time"
)

type DateRangeIterator struct {
	current   time.Time
	end       time.Time
	skipWeekends bool
}

func NewDateRangeIterator(start, end time.Time, skipWeekends bool) *DateRangeIterator {
	return &DateRangeIterator{
		current:   start,
		end:       end,
		skipWeekends: skipWeekends,
	}
}

func (it *DateRangeIterator) Next() (time.Time, bool) {
	for !it.current.After(it.end) {
		day := it.current
		it.current = it.current.AddDate(0, 0, 1)

		if it.skipWeekends {
			weekday := day.Weekday()
			if weekday == time.Saturday || weekday == time.Sunday {
				continue
			}
		}
		return day, true
	}
	return time.Time{}, false
}

func main() {
	start := time.Date(2023, time.October, 6, 0, 0, 0, 0, time.UTC) // Friday
	end := time.Date(2023, time.October, 10, 0, 0, 0, 0, time.UTC) // Tuesday

	fmt.Println("All days:")
	it1 := NewDateRangeIterator(start, end, false)
	for d, ok := it1.Next(); ok; d, ok = it1.Next() {
		fmt.Println(d.Format("2006-01-02"))
	}

	fmt.Println("\nSkipping weekends:")
	it2 := NewDateRangeIterator(start, end, true)
	for d, ok := it2.Next(); ok; d, ok = it2.Next() {
		fmt.Println(d.Format("2006-01-02"))
	}
}
