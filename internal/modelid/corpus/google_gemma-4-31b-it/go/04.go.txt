package main

import (
	"fmt"
	"regexp"
	"strings"
	"unicode"
)

var nonAlphaNumRegex = regexp.MustCompile(`[^a-z0-9]+`)

func Slugify(s string) string {
	var builder strings.Builder
	for _, r := range s {
		if unicode.IsLetter(r) || unicode.IsDigit(r) {
			builder.WriteRune(unicode.ToLower(r))
		} else {
			builder.WriteRune(' ')
		}
	}

	text := builder.String()
	text = nonAlphaNumRegex.ReplaceAllString(text, "-")
	return strings.Trim(text, "-")
}

func main() {
	tests := []string{
		"Hello World!",
		"Go language is AWESOME!!!",
		"Unicode: 你好, World!",
		"---Trim Me---",
		"Multiple   Spaces  and --- Hyphens",
		"123-abc_DEF",
	}

	for _, t := range tests {
		fmt.Printf("%q -> %q\n", t, Slugify(t))
	}
}
