import * as fs from 'fs';
import * as path from 'path';

export interface ConfigLoaderOptions<T> {
  filePath: string;
  defaults: T;
  requiredKeys: (keyof T)[];
  envPrefix?: string;
}

export class ConfigLoader {
  static load<T extends Record<string, any>>(options: ConfigLoaderOptions<T>): T {
    const { filePath, defaults, requiredKeys, envPrefix = 'APP_' } = options;

    // 1. Load JSON file
    let fileConfig: Partial<T> = {};
    if (fs.existsSync(filePath)) {
      try {
        const content = fs.readFileSync(filePath, 'utf-8');
        fileConfig = JSON.parse(content);
      } catch (err) {
        throw new Error(`Failed to parse config file at ${filePath}: ${err}`);
      }
    }

    // 2. Merge defaults and file
    const config = { ...defaults, ...fileConfig };

    // 3. Environment Variable Overrides
    // Iterates through keys in defaults to find matching env vars (e.g., APP_DATABASE_URL)
    for (const key in defaults) {
      const envKey = `${envPrefix}${key.toUpperCase()}`;
      const envValue = process.env[envKey];
      if (envValue !== undefined) {
        // Attempt to cast env var to the type of the default value
        const defaultValue = defaults[key];
        if (typeof defaultValue === 'number') {
          config[key] = Number(envValue);
        } else if (typeof defaultValue === 'boolean') {
          config[key] = envValue.toLowerCase() === 'true';
        } else {
          config[key] = envValue as any;
        }
      }
    }

    // 4. Validation of required keys
    for (const key of requiredKeys) {
      if (config[key] === undefined || config[key] === null || config[key] === '') {
        throw new Error(`Missing required configuration key: ${key}`);
      }
    }

    return config;
  }
}
