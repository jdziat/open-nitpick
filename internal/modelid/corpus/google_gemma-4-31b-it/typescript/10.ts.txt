interface ShortUrl {
    originalUrl: string;
    expiry: number;
}

export class UrlShortener {
    private store: Map<string, ShortUrl> = new Map();
    private alphabet = "abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789";

    public shorten(url: string, ttlMs: number): string {
        const code = this.generateCode();
        const expiry = Date.now() + ttlMs;
        this.store.set(code, { originalUrl: url, expiry });
        return code;
    }

    public resolve(code: string): string | null {
        const entry = this.store.get(code);
        if (!entry) return null;

        if (Date.now() > entry.expiry) {
            this.store.delete(code);
            return null;
        }

        return entry.originalUrl;
    }

    private generateCode(): string {
        let code = "";
        for (let i = 0; i < 6; i++) {
            code += this.alphabet.charAt(Math.floor(Math.random() * this.alphabet.length));
        }
        return code;
    }
}
