import time
import secrets
import string

class URLShortener:
    def __init__(self):
        self.db = {}

    def shorten(self, url, expiry_seconds=3600):
        code = ''.join(secrets.choice(string.ascii_letters + string.digits) for _ in range(6))
        expires_at = time.time() + expiry_seconds
        self.db[code] = (url, expires_at)
        return code

    def resolve(self, code):
        if code not in self.db:
            return None
        
        url, expires_at = self.db[code]
        if time.time() > expires_at:
            del self.db[code]
            return None
            
        return url

if __name__ == "__main__":
    shortener = URLShortener()
    code = shortener.shorten("https://www.google.com", expiry_seconds=2)
    print(f"Short code: {code}")
    print(f"Resolved: {shortener.resolve(code)}")
    time.sleep(3)
    print(f"Resolved after expiry: {shortener.resolve(code)}")
