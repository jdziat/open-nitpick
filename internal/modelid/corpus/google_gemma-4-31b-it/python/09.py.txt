import datetime

def date_range(start_date, end_date, skip_weekends=False):
    """
    Yields each day between start_date and end_date inclusive.
    If skip_weekends is True, Saturday (5) and Sunday (6) are omitted.
    """
    current_date = start_date
    while current_date <= end_date:
        if skip_weekends:
            # weekday() returns 0 for Monday and 6 for Sunday
            if current_date.weekday() < 5:
                yield current_date
        else:
            yield current_date
        current_date += datetime.timedelta(days=1)

if __name__ == "__main__":
    start = datetime.date(2023, 10, 6) # Friday
    end = datetime.date(2023, 10, 13)   # Friday
    
    print("All days:")
    for d in date_range(start, end):
        print(d)
        
    print("\nWeekdays only:")
    for d in date_range(start, end, skip_weekends=True):
        print(d)
