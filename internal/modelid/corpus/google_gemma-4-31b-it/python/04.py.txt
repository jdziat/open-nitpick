import re
import unicodedata

def slugify(text):
    # Normalize unicode characters to decompose combined characters
    text = unicodedata.normalize('NFKD', text)
    # Encode to ASCII bytes, ignoring non-ASCII, then decode back to string
    text = text.encode('ascii', 'ignore').decode('ascii')
    # Lowercase the text
    text = text.lower()
    # Replace runs of non-alphanumerics with a single hyphen
    text = re.sub(r'[^a-z0-9]+', '-', text)
    # Trim hyphens from start and end
    return text.strip('-')

if __name__ == "__main__":
    test_cases = [
        "Hello World!", 
        "Python & Unicode: Café ☕", 
        "---Too many---hyphens---", 
        "123 ABC 456", 
        "   Spaces   Everywhere   "
    ]
    for case in test_cases:
        print(f"'{case}' -> '{slugify(case)}'")
