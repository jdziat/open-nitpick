class Matrix:
    def __init__(self, data):
        if not data or not isinstance(data, list) or not all(isinstance(row, list) for row in data):
            raise ValueError("Data must be a list of lists.")
        self.rows = len(data)
        self.cols = len(data[0]) if self.rows > 0 else 0
        if any(len(row) != self.cols for row in data):
            raise ValueError("All rows must have the same length.")
        self.data = data

    def __repr__(self):
        return f"Matrix({self.data})"

    def transpose(self):
        new_data = [[self.data[r][c] for r in range(self.rows)] for c in range(self.cols)]
        return Matrix(new_data)

    def __mul__(self, other):
        if not isinstance(other, Matrix):
            raise TypeError("Multiplication only supported between Matrix instances.")
        if self.cols != other.rows:
            raise ValueError(f"Shape mismatch: {self.rows}x{self.cols} and {other.rows}x{other.cols}")
        
        result = [[0 for _ in range(other.cols)] for _ in range(self.rows)]
        for i in range(self.rows):
            for j in range(other.cols):
                for k in range(self.cols):
                    result[i][j] += self.data[i][k] * other.data[k][j]
        return Matrix(result)

    def determinant(self):
        if self.rows != self.cols:
            raise ValueError("Determinant requires a square matrix.")
        
        def _det(m):
            if len(m) == 1:
                return m[0][0]
            if len(m) == 2:
                return m[0][0] * m[1][1] - m[0][1] * m[1][0]
            
            res = 0
            for c in range(len(m)):
                minor = [row[:c] + row[c+1:] for row in m[1:]]
                res += ((-1)**c) * m[0][c] * _det(minor)
            return res

        return _det(self.data)
