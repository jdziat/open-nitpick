import threading
import time

class TokenBucket:
    def __init__(self, capacity: float, refill_rate: float):
        """
        :param capacity: Maximum number of tokens the bucket can hold.
        :param refill_rate: Tokens added per second.
        """
        self.capacity = capacity
        self.refill_rate = refill_rate
        self.tokens = capacity
        self.last_refill = time.monotonic()
        self.lock = threading.Lock()

    def consume(self, amount: float = 1.0) -> bool:
        """
        Attempts to consume a specified amount of tokens.
        Returns True if successful, False otherwise.
        """
        with self.lock:
            now = time.monotonic()
            elapsed = now - self.last_refill
            
            # Refill tokens based on time elapsed
            refill_amount = elapsed * self.refill_rate
            self.tokens = min(self.capacity, self.tokens + refill_amount)
            self.last_refill = now

            if self.tokens >= amount:
                self.tokens -= amount
                return True
            return False

    def block_consume(self, amount: float = 1.0):
        """
        Blocks until the requested amount of tokens are available.
        """
        while True:
            with self.lock:
                now = time.monotonic()
                elapsed = now - self.last_refill
                
                refill_amount = elapsed * self.refill_rate
                self.tokens = min(self.capacity, self.tokens + refill_amount)
                self.last_refill = now

                if self.tokens >= amount:
                    self.tokens -= amount
                    return

                # Calculate sleep time to avoid busy-waiting
                needed = amount - self.tokens
                sleep_time = needed / self.refill_rate
            
            time.sleep(sleep_time)
