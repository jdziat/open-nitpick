import heapq

class PriorityQueue:
    def __init__(self, comparator):
        """
        comparator: A function that takes two elements (a, b) and 
        returns True if a has higher priority than b.
        """
        self._comparator = comparator
        self._heap = []
        self._count = 0

    class Wrapper:
        def __init__(self, priority_queue, item):
            self.pq = priority_queue
            self.item = item

        def __lt__(self, other):
            # heapq is a min-heap. To support a custom comparator,
            # we define < such that the 'smallest' element is the one 
            # the comparator deems highest priority.
            return self.pq._comparator(self.item, other.item)

    def push(self, item):
        heapq.heappush(self._heap, self.Wrapper(self, item))

    def pop(self):
        if not self._heap:
            raise IndexError("pop from empty priority queue")
        return heapq.heappop(self._heap).item

    def peek(self):
        if not self._heap:
            raise IndexError("peek from empty priority queue")
        return self._heap[0].item

    def __len__(self):
        return len(self._heap)
