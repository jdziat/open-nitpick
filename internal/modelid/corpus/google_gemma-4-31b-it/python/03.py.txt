import time
import random
from typing import Callable, Any, Type

def retry(
    func: Callable[..., Any],
    max_attempts: int = 5,
    base_delay: float = 1.0,
    max_delay: float = 30.0,
    exceptions: Type[Exception] = Exception
):
    """
    Retries a function with exponential backoff and jitter.
    Returns the result of the function if successful, or the last exception if all attempts fail.
    """
    last_exception = None
    
    for attempt in range(max_attempts):
        try:
            return func()
        except exceptions as e:
            last_exception = e
            if attempt == max_attempts - 1:
                break
            
            # Exponential backoff: base_delay * 2^attempt
            delay = min(base_delay * (2 ** attempt), max_delay)
            # Full Jitter: random value between 0 and delay
            jittered_delay = random.uniform(0, delay)
            
            time.sleep(jittered_delay)
            
    return last_exception

if __name__ == "__main__":
    # Example usage
    counter = 0
    def unstable_function():
        global counter
        counter += 1
        if counter < 3:
            print(f"Attempt {counter} failed...")
            raise ValueError("Transient error")
        print(f"Attempt {counter} succeeded!")
        return "Success!"

    result = retry(unstable_function)
    print(f"Final Result: {result}")
