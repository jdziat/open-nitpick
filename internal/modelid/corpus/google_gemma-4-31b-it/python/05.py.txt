import re
from collections import Counter

def get_top_words(text, k=10):
    """
    Normalises text by removing punctuation and converting to lowercase,
    then returns the top K most frequent words.
    """
    # Remove punctuation using regex and convert to lowercase
    normalized_text = re.sub(r'[^\w\s]', '', text.lower())
    
    # Split into words
    words = normalized_text.split()
    
    # Count frequencies
    counts = Counter(words)
    
    # Return the k most common
    return counts.most_common(k)

if __name__ == "__main__":
    sample_text = "Hello world! This is a test. Hello again, world. This test is only a test."
    K = 3
    result = get_top_words(sample_text, K)
    print(f"Top {K} words:")
    for word, count in result:
        print(f"{word}: {count}")
