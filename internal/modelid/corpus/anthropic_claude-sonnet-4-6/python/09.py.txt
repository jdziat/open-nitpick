from datetime import date, timedelta
from typing import Iterator


def date_range(
    start: date,
    end: date,
    skip_weekends: bool = False,
) -> Iterator[date]:
    """Yield each day between start and end inclusive.

    Args:
        start: The first date of the range.
        end: The last date of the range (inclusive).
        skip_weekends: If True, Saturday (weekday 5) and Sunday (weekday 6)
                       are skipped.

    Yields:
        date objects for each day in the range.
    """
    if start > end:
        return

    current = start
    while current <= end:
        if skip_weekends and current.weekday() >= 5:
            current += timedelta(days=1)
            continue
        yield current
        current += timedelta(days=1)


def main() -> None:
    start = date(2024, 1, 1)
    end = date(2024, 1, 14)

    print("All days from 2024-01-01 to 2024-01-14:")
    for d in date_range(start, end):
        print(f"  {d}  ({d.strftime('%A')})")

    print()
    print("Weekdays only from 2024-01-01 to 2024-01-14:")
    for d in date_range(start, end, skip_weekends=True):
        print(f"  {d}  ({d.strftime('%A')})")

    print()
    print("Edge case – same day (weekday):")
    for d in date_range(date(2024, 1, 3), date(2024, 1, 3)):
        print(f"  {d}  ({d.strftime('%A')})")

    print()
    print("Edge case – same day (weekend, skip_weekends=True):")
    results = list(date_range(date(2024, 1, 6), date(2024, 1, 6), skip_weekends=True))
    print(f"  Yielded {len(results)} date(s)")

    print()
    print("Edge case – start after end:")
    results = list(date_range(date(2024, 1, 10), date(2024, 1, 1)))
    print(f"  Yielded {len(results)} date(s)")


if __name__ == "__main__":
    main()
