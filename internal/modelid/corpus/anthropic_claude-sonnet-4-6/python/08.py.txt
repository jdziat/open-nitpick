class BinaryHeap:
    def __init__(self, compare=None):
        """
        Initialize the binary heap with an optional comparison function.
        
        The comparison function should take two arguments and return True if
        the first argument should have higher priority (closer to the top) than
        the second argument.
        
        If no comparison function is provided, a min-heap is used by default.
        """
        self._heap = []
        if compare is None:
            self._compare = lambda a, b: a < b
        else:
            self._compare = compare

    def _parent(self, index):
        return (index - 1) // 2

    def _left_child(self, index):
        return 2 * index + 1

    def _right_child(self, index):
        return 2 * index + 2

    def _has_higher_priority(self, index_a, index_b):
        return self._compare(self._heap[index_a], self._heap[index_b])

    def _swap(self, i, j):
        self._heap[i], self._heap[j] = self._heap[j], self._heap[i]

    def _sift_up(self, index):
        """Move element at index up to its correct position."""
        while index > 0:
            parent = self._parent(index)
            if self._has_higher_priority(index, parent):
                self._swap(index, parent)
                index = parent
            else:
                break

    def _sift_down(self, index):
        """Move element at index down to its correct position."""
        size = len(self._heap)
        while True:
            highest = index
            left = self._left_child(index)
            right = self._right_child(index)

            if left < size and self._has_higher_priority(left, highest):
                highest = left

            if right < size and self._has_higher_priority(right, highest):
                highest = right

            if highest != index:
                self._swap(index, highest)
                index = highest
            else:
                break

    def push(self, item):
        """Add an item to the priority queue."""
        self._heap.append(item)
        self._sift_up(len(self._heap) - 1)

    def pop(self):
        """
        Remove and return the highest priority item.
        
        Raises IndexError if the heap is empty.
        """
        if not self._heap:
            raise IndexError("pop from an empty priority queue")

        if len(self._heap) == 1:
            return self._heap.pop()

        # Move the last element to the root and sift down
        top = self._heap[0]
        self._heap[0] = self._heap.pop()
        self._sift_down(0)
        return top

    def peek(self):
        """
        Return the highest priority item without removing it.
        
        Raises IndexError if the heap is empty.
        """
        if not self._heap:
            raise IndexError("peek from an empty priority queue")
        return self._heap[0]

    def __len__(self):
        return len(self._heap)

    def __bool__(self):
        return bool(self._heap)

    def __repr__(self):
        return f"BinaryHeap({self._heap!r})"


class PriorityQueue:
    def __init__(self, compare=None):
        """
        Initialize the priority queue with an optional comparison function.
        
        The comparison function should take two arguments and return True if
        the first argument should have higher priority than the second.
        
        Defaults to a min-heap if no comparison is provided.
        """
        self._heap = BinaryHeap(compare)

    def push(self, item):
        """Add an item to the priority queue."""
        self._heap.push(item)

    def pop(self):
        """
        Remove and return the highest priority item.
        
        Raises IndexError if the queue is empty.
        """
        return self._heap.pop()

    def peek(self):
        """
        Return the highest priority item without removing it.
        
        Raises IndexError if the queue is empty.
        """
        return self._heap.peek()

    def __len__(self):
        return len(self._heap)

    def __bool__(self):
        return bool(self._heap)

    def __repr__(self):
        return f"PriorityQueue(size={len(self._heap)})"


def main():
    print("=== Min-Heap (default) ===")
    pq = PriorityQueue()
    for value in [5, 3, 8, 1, 9, 2, 7]:
        pq.push(value)
        print(f"  Pushed {value}, peek = {pq.peek()}")

    print(f"\nPopping all elements from min-heap:")
    while pq:
        print(f"  {pq.pop()}", end=" ")
    print()

    print("\n=== Max-Heap (custom comparator) ===")
    pq_max = PriorityQueue(compare=lambda a, b: a > b)
    for value in [5, 3, 8, 1, 9, 2, 7]:
        pq_max.push(value)
        print(f"  Pushed {value}, peek = {pq_max.peek()}")

    print(f"\nPopping all elements from max-heap:")
    while pq_max:
        print(f"  {pq_max.pop()}", end=" ")
    print()

    print("\n=== Priority Queue with tuples (priority, task) ===")
    pq_tasks = PriorityQueue(compare=lambda a, b: a[0] < b[0])
    tasks = [(3, "low priority task"), (1, "high priority task"), (2, "medium priority task")]
    for task in tasks:
        pq_tasks.push(task)
        print(f"  Added: {task}")

    print(f"\nProcessing tasks by priority:")
    while pq_tasks:
        priority, task = pq_tasks.pop()
        print(f"  [{priority}] {task}")

    print("\n=== Error handling ===")
    empty_pq = PriorityQueue()
    try:
        empty_pq.pop()
    except IndexError as e:
        print(f"  pop error: {e}")

    try:
        empty_pq.peek()
    except IndexError as e:
        print(f"  peek error: {e}")


if __name__ == "__main__":
    main()
