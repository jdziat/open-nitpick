import os
import json
import sys
from typing import Any

DEFAULTS: dict[str, Any] = {
    "host": "localhost",
    "port": 8080,
    "debug": False,
    "log_level": "INFO",
    "database_url": None,
    "secret_key": None,
    "max_connections": 10,
    "timeout": 30,
}

REQUIRED_KEYS: list[str] = ["database_url", "secret_key"]

ENV_PREFIX = "APP_"

ENV_VAR_MAP: dict[str, str] = {
    "APP_HOST": "host",
    "APP_PORT": "port",
    "APP_DEBUG": "debug",
    "APP_LOG_LEVEL": "log_level",
    "APP_DATABASE_URL": "database_url",
    "APP_SECRET_KEY": "secret_key",
    "APP_MAX_CONNECTIONS": "max_connections",
    "APP_TIMEOUT": "timeout",
}

TYPE_COERCIONS: dict[str, type] = {
    "port": int,
    "debug": bool,
    "max_connections": int,
    "timeout": int,
}


def _coerce_value(key: str, value: str) -> Any:
    """Coerce a string value to the appropriate type based on the key."""
    if key not in TYPE_COERCIONS:
        return value
    target_type = TYPE_COERCIONS[key]
    if target_type == bool:
        if isinstance(value, bool):
            return value
        return value.strip().lower() in ("1", "true", "yes", "on")
    try:
        return target_type(value)
    except (ValueError, TypeError) as exc:
        raise ValueError(
            f"Cannot coerce value '{value}' for key '{key}' to {target_type.__name__}: {exc}"
        ) from exc


def load_json_file(filepath: str) -> dict[str, Any]:
    """Load configuration from a JSON file."""
    try:
        with open(filepath, "r", encoding="utf-8") as fh:
            data = json.load(fh)
        if not isinstance(data, dict):
            raise ValueError(f"JSON file '{filepath}' must contain a JSON object at the top level.")
        return data
    except FileNotFoundError:
        raise FileNotFoundError(f"Configuration file not found: '{filepath}'")
    except json.JSONDecodeError as exc:
        raise ValueError(f"Invalid JSON in configuration file '{filepath}': {exc}") from exc


def apply_env_overrides(config: dict[str, Any]) -> dict[str, Any]:
    """Override configuration values with environment variables."""
    for env_var, config_key in ENV_VAR_MAP.items():
        env_value = os.environ.get(env_var)
        if env_value is not None:
            config[config_key] = _coerce_value(config_key, env_value)
    return config


def validate_required_keys(config: dict[str, Any], required_keys: list[str]) -> None:
    """Validate that all required keys are present and not None."""
    missing = []
    for key in required_keys:
        if key not in config or config[key] is None:
            missing.append(key)
    if missing:
        raise ValueError(
            f"Missing required configuration keys: {', '.join(missing)}. "
            f"Set them in the config file or via environment variables: "
            f"{', '.join(ENV_PREFIX + k.upper() for k in missing)}"
        )


def load_config(
    filepath: str | None = None,
    defaults: dict[str, Any] | None = None,
    required_keys: list[str] | None = None,
    env_var_map: dict[str, str] | None = None,
) -> dict[str, Any]:
    """
    Load configuration from a JSON file with defaults and environment-variable overrides.

    Steps:
      1. Start with built-in defaults (or provided defaults).
      2. Overlay values from the JSON file (if provided).
      3. Apply environment-variable overrides.
      4. Validate that all required keys have non-None values.

    Args:
        filepath: Path to the JSON configuration file (optional).
        defaults: Default configuration values (uses module-level DEFAULTS if None).
        required_keys: Keys that must be present and non-None (uses module-level REQUIRED_KEYS if None).
        env_var_map: Mapping of ENV_VAR_NAME -> config_key (uses module-level ENV_VAR_MAP if None).

    Returns:
        Merged and validated configuration dictionary.
    """
    if defaults is None:
        defaults = DEFAULTS.copy()
    if required_keys is None:
        required_keys = REQUIRED_KEYS
    if env_var_map is not None:
        global ENV_VAR_MAP
        ENV_VAR_MAP = env_var_map

    config: dict[str, Any] = dict(defaults)

    if filepath is not None:
        file_config = load_json_file(filepath)
        for key, value in file_config.items():
            if key in TYPE_COERCIONS and not isinstance(value, str):
                config[key] = _coerce_value(key, str(value)) if not isinstance(value, TYPE_COERCIONS[key]) else value
            else:
                config[key] = value

    config = apply_env_overrides(config)
    validate_required_keys(config, required_keys)

    return config


def print_config(config: dict[str, Any]) -> None:
    """Pretty-print the configuration, masking sensitive keys."""
    sensitive_keys = {"secret_key", "database_url", "password", "token"}
    print("Configuration:")
    print("-" * 40)
    for key, value in sorted(config.items()):
        if key in sensitive_keys and value is not None:
            display_value = "***" + str(value)[-4:] if len(str(value)) > 4 else "****"
        else:
            display_value = value
        print(f"  {key}: {display_value}")
    print("-" * 40)


if __name__ == "__main__":
    import argparse
    import tempfile

    parser = argparse.ArgumentParser(description="Configuration loader demo")
    parser.add_argument("--config", "-c", default=None, help="Path to JSON config file")
    parser.add_argument("--demo", action="store_true", help="Run a demo with a temporary config file")
    args = parser.parse_args()

    if args.demo:
        demo_config_data = {
            "host": "example.com",
            "port": 9090,
            "debug": True,
            "database_url": "postgresql://user:pass@localhost/mydb",
            "secret_key": "super-secret-key-12345",
            "log_level": "DEBUG",
        }
        with tempfile.NamedTemporaryFile(
            mode="w", suffix=".json", delete=False, encoding="utf-8"
        ) as tmp:
            json.dump(demo_config_data, tmp, indent=2)
            tmp_path = tmp.name

        print(f"Demo: using temporary config file: {tmp_path}")
        os.environ.setdefault("APP_MAX_CONNECTIONS", "25")
        os.environ.setdefault("APP_TIMEOUT", "60")

        try:
            cfg = load_config(filepath=tmp_path)
            print_config(cfg)
        except (FileNotFoundError, ValueError) as e:
            print(f"Error loading config: {e}", file=sys.stderr)
            sys.exit(1)
        finally:
            os.unlink(tmp_path)

    elif args.config:
        try:
            cfg = load_config(filepath=args.config)
            print_config(cfg)
        except (FileNotFoundError, ValueError) as e:
            print(f"Error loading config: {e}", file=sys.stderr)
            sys.exit(1)
    else:
        print("No config file provided. Attempting to load from environment variables only...")
        try:
            cfg = load_config(filepath=None)
            print_config(cfg)
        except (FileNotFoundError, ValueError) as e:
            print(f"Error: {e}", file=sys.stderr)
            print(
                "\nHint: Run with --demo to see a working example, or provide a config file with --config.",
                file=sys.stderr,
            )
            sys.exit(1)
