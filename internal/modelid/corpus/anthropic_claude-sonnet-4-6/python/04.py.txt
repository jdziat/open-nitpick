import unicodedata
import re


def slugify(text: str) -> str:
    """
    Convert text to a slug by:
    1. Normalizing Unicode characters (NFKD decomposition)
    2. Converting to lowercase
    3. Replacing runs of non-alphanumeric characters with single hyphens
    4. Trimming leading/trailing hyphens
    5. Handling Unicode letters by keeping them
    """
    # Normalize unicode characters using NFKD decomposition
    # This separates base characters from combining characters (accents, etc.)
    text = unicodedata.normalize("NFKD", text)

    # Encode to ASCII bytes, ignoring characters that can't be converted,
    # then decode back to string - this strips accents and diacritics
    # but we want to keep unicode letters, so we'll handle this differently
    
    # Actually, let's keep unicode letters but normalize them
    # Re-normalize with NFC to keep composed unicode letters
    text = unicodedata.normalize("NFC", text)

    # Convert to lowercase
    text = text.lower()

    # Replace runs of non-alphanumeric characters (including unicode alphanumerics)
    # with a single hyphen. \w matches unicode word characters including letters,
    # digits, and underscore. We want to keep letters and digits but not underscore.
    # So we replace anything that's not a unicode letter or digit with a hyphen.
    text = re.sub(r"[^\w]|_", "-", text)

    # Replace multiple consecutive hyphens with a single hyphen
    text = re.sub(r"-+", "-", text)

    # Trim leading and trailing hyphens
    text = text.strip("-")

    return text


def main():
    test_cases = [
        ("Hello, World!", "hello-world"),
        ("  leading and trailing spaces  ", "leading-and-trailing-spaces"),
        ("multiple---hyphens", "multiple-hyphens"),
        ("café au lait", "café-au-lait"),
        ("Ångström", "ångström"),
        ("日本語テスト", "日本語テスト"),
        ("Python 3.9 is great!", "python-3-9-is-great"),
        ("--leading-hyphens--", "leading-hyphens"),
        ("special @#$% characters", "special-characters"),
        ("", ""),
        ("   ", ""),
        ("Hello_World", "hello-world"),
        ("naïve résumé", "naïve-résumé"),
        ("über cool", "über-cool"),
        ("C++ programming", "c-programming"),
        ("snake_case_variable", "snake-case-variable"),
        ("camelCaseText", "camelcasetext"),
        ("100% organic", "100-organic"),
        ("file.name.with.dots", "file-name-with-dots"),
        ("path/to/resource", "path-to-resource"),
    ]

    print("Slugify Function Tests")
    print("=" * 60)
    
    all_passed = True
    for text, expected in test_cases:
        result = slugify(text)
        status = "PASS" if result == expected else "FAIL"
        if status == "FAIL":
            all_passed = False
        print(f"[{status}] Input:    {repr(text)}")
        print(f"       Expected: {repr(expected)}")
        if status == "FAIL":
            print(f"       Got:      {repr(result)}")
        print()

    if all_passed:
        print("All tests passed!")
    else:
        print("Some tests failed!")

    # Interactive demo
    print("\n" + "=" * 60)
    print("Interactive Demo")
    print("=" * 60)
    demo_texts = [
        "The Quick Brown Fox Jumps Over The Lazy Dog",
        "Héllo Wörld - Ünïcödé Têxt",
        "   spaces   and---multiple-hyphens   ",
        "Email: user@example.com",
        "Price: $19.99 (USD)",
    ]
    
    for text in demo_texts:
        print(f"Input:  {repr(text)}")
        print(f"Slug:   {repr(slugify(text))}")
        print()


if __name__ == "__main__":
    main()
