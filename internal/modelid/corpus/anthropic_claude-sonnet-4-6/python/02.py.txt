import threading
import time


class TokenBucket:
    """
    A thread-safe token-bucket rate limiter.

    Parameters
    ----------
    capacity : float
        Maximum number of tokens the bucket can hold.
    refill_rate : float
        Number of tokens added to the bucket per second.
    """

    def __init__(self, capacity: float, refill_rate: float) -> None:
        if capacity <= 0:
            raise ValueError("capacity must be positive")
        if refill_rate <= 0:
            raise ValueError("refill_rate must be positive")

        self._capacity = float(capacity)
        self._refill_rate = float(refill_rate)
        self._tokens = float(capacity)          # start full
        self._last_refill = time.monotonic()
        self._lock = threading.Lock()

    # ------------------------------------------------------------------
    # Internal helpers (must be called with self._lock held)
    # ------------------------------------------------------------------

    def _refill(self) -> None:
        now = time.monotonic()
        elapsed = now - self._last_refill
        new_tokens = elapsed * self._refill_rate
        self._tokens = min(self._capacity, self._tokens + new_tokens)
        self._last_refill = now

    # ------------------------------------------------------------------
    # Public API
    # ------------------------------------------------------------------

    @property
    def capacity(self) -> float:
        return self._capacity

    @property
    def refill_rate(self) -> float:
        return self._refill_rate

    @property
    def tokens(self) -> float:
        """Current token count (snapshot; may be stale immediately after read)."""
        with self._lock:
            self._refill()
            return self._tokens

    def consume(self, tokens: float = 1.0) -> bool:
        """
        Try to consume *tokens* tokens without blocking.

        Returns
        -------
        bool
            True if the tokens were available and have been consumed,
            False otherwise (the bucket is unchanged on failure).
        """
        if tokens <= 0:
            raise ValueError("tokens to consume must be positive")
        with self._lock:
            self._refill()
            if self._tokens >= tokens:
                self._tokens -= tokens
                return True
            return False

    def consume_blocking(self, tokens: float = 1.0, timeout: float | None = None) -> bool:
        """
        Block until *tokens* tokens are available, then consume them.

        Parameters
        ----------
        tokens : float
            Number of tokens to consume.
        timeout : float | None
            Maximum number of seconds to wait.  None means wait forever.

        Returns
        -------
        bool
            True if the tokens were consumed, False if the timeout expired.
        """
        if tokens > self._capacity:
            raise ValueError(
                f"requested tokens ({tokens}) exceed bucket capacity ({self._capacity})"
            )
        if tokens <= 0:
            raise ValueError("tokens to consume must be positive")

        deadline = None if timeout is None else time.monotonic() + timeout

        while True:
            with self._lock:
                self._refill()
                if self._tokens >= tokens:
                    self._tokens -= tokens
                    return True

                # How long until we have enough tokens?
                deficit = tokens - self._tokens
                wait_needed = deficit / self._refill_rate

            if deadline is not None:
                remaining = deadline - time.monotonic()
                if remaining <= 0:
                    return False
                sleep_for = min(wait_needed, remaining)
            else:
                sleep_for = wait_needed

            # Sleep for slightly less than the theoretical wait to avoid
            # overshooting due to scheduling jitter, then re-check.
            time.sleep(max(0.0, sleep_for * 0.9))

    def __repr__(self) -> str:
        return (
            f"TokenBucket(capacity={self._capacity}, "
            f"refill_rate={self._refill_rate}, "
            f"tokens={self.tokens:.4f})"
        )


# ---------------------------------------------------------------------------
# Demo / smoke-test
# ---------------------------------------------------------------------------

def _demo() -> None:
    import random

    print("=== Token Bucket Demo ===\n")

    # --- Basic non-blocking usage ---
    bucket = TokenBucket(capacity=5, refill_rate=2)   # 2 tokens/sec, max 5
    print(f"Created: {bucket}")

    for i in range(7):
        ok = bucket.consume(1)
        print(f"  consume(1) -> {'OK' if ok else 'DENIED'}  | tokens remaining: {bucket.tokens:.2f}")

    print()

    # --- Refill demonstration ---
    print("Sleeping 1 second to let the bucket refill...")
    time.sleep(1.0)
    print(f"After 1 s: {bucket}\n")

    # --- Concurrent usage ---
    print("Concurrent test: 10 threads each trying to consume 1 token from a bucket(5, 5)...")
    shared_bucket = TokenBucket(capacity=5, refill_rate=5)
    results: list[bool] = []
    results_lock = threading.Lock()

    def worker(wid: int) -> None:
        # small random stagger so they roughly hit at the same time
        time.sleep(random.uniform(0, 0.01))
        ok = shared_bucket.consume(1)
        with results_lock:
            results.append(ok)
        print(f"  thread-{wid:02d}: {'OK' if ok else 'DENIED'}")

    threads = [threading.Thread(target=worker, args=(i,)) for i in range(10)]
    for t in threads:
        t.start()
    for t in threads:
        t.join()

    granted = sum(results)
    print(f"\n  Granted: {granted}/10  (expected ~5)\n")

    # --- Blocking usage ---
    print("Blocking test: drain bucket then consume_blocking with 2-second timeout...")
    small = TokenBucket(capacity=3, refill_rate=1)   # 1 token/sec
    small.consume(3)                                  # drain completely
    print(f"  Drained: {small}")
    start = time.monotonic()
    ok = small.consume_blocking(tokens=2, timeout=5.0)
    elapsed = time.monotonic() - start
    print(f"  consume_blocking(2, timeout=5) -> {'OK' if ok else 'TIMEOUT'} in {elapsed:.2f}s\n")


if __name__ == "__main__":
    _demo()
