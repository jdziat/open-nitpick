import csv
import io
from typing import Iterator


def parse_csv(
    source: str,
    delimiter: str = ",",
    quotechar: str = '"',
) -> list[list[str]]:
    """
    Parse a CSV string and return a list of rows, where each row is a list of
    strings.

    Supports:
      - Quoted fields (fields wrapped in ``quotechar``)
      - Escaped quotes inside quoted fields (doubled quotechar, e.g. ``""``)
      - A configurable field delimiter

    Parameters
    ----------
    source:
        The raw CSV text to parse.
    delimiter:
        The character used to separate fields.  Defaults to a comma.
    quotechar:
        The character used to quote fields that contain special characters.
        Defaults to a double-quote.

    Returns
    -------
    list[list[str]]
        A list of rows; each row is a list of field strings.
    """
    reader = csv.reader(
        io.StringIO(source),
        delimiter=delimiter,
        quotechar=quotechar,
        doublequote=True,          # "" inside a quoted field → literal "
        skipinitialspace=False,
        strict=True,
    )
    return [row for row in reader]


# ---------------------------------------------------------------------------
# Manual / pure-Python implementation (no stdlib csv) – used when you need
# full control or want to understand the algorithm.
# ---------------------------------------------------------------------------

def parse_csv_manual(
    source: str,
    delimiter: str = ",",
    quotechar: str = '"',
) -> list[list[str]]:
    """
    A hand-written CSV parser that handles the same subset as :func:`parse_csv`
    without relying on the ``csv`` module.

    Rules
    -----
    * A field may optionally be wrapped in *quotechar*.
    * Inside a quoted field a literal *quotechar* is represented by two
      consecutive *quotechar* characters (RFC 4180 double-quote escaping).
    * Outside a quoted field the field ends at the next *delimiter* or newline.
    * Newlines inside a quoted field are preserved as part of the field value.
    """
    rows: list[list[str]] = []
    current_row: list[str] = []
    field_chars: list[str] = []
    in_quotes = False
    i = 0
    n = len(source)

    while i < n:
        ch = source[i]

        if in_quotes:
            if ch == quotechar:
                # Peek ahead: doubled quotechar → escaped literal quotechar
                if i + 1 < n and source[i + 1] == quotechar:
                    field_chars.append(quotechar)
                    i += 2          # consume both quotechar characters
                    continue
                else:
                    # Closing quote
                    in_quotes = False
                    i += 1
                    continue
            else:
                # Any other character (including delimiter and newlines) is
                # part of the field while we are inside quotes.
                field_chars.append(ch)
                i += 1
                continue

        # ---- not inside quotes ----
        if ch == quotechar:
            in_quotes = True
            i += 1
            continue

        if ch == delimiter:
            current_row.append("".join(field_chars))
            field_chars = []
            i += 1
            continue

        if ch == "\r":
            # Handle \r\n as a single line ending
            current_row.append("".join(field_chars))
            field_chars = []
            rows.append(current_row)
            current_row = []
            i += 1
            if i < n and source[i] == "\n":
                i += 1          # skip the \n in \r\n
            continue

        if ch == "\n":
            current_row.append("".join(field_chars))
            field_chars = []
            rows.append(current_row)
            current_row = []
            i += 1
            continue

        field_chars.append(ch)
        i += 1

    # Flush the last field / row (file may not end with a newline)
    if field_chars or current_row:
        current_row.append("".join(field_chars))
        rows.append(current_row)

    return rows

# Quick self-test
# ---------------------------------------------------------------------------

if __name__ == "__main__":
    samples = [
        # (description, csv_text, kwargs)
        (
            "Basic comma-separated",
            "one,two,three\n1,2,3\n",
            {},
        ),
        (
            "Quoted field with delimiter inside",
            'a,"b,c",d\n',
            {},
        ),
        (
            "Escaped quote (doubled)",
            '"say ""hello""",world\n',
            {},
        ),
        (
            "Multiline quoted field",
            '"line1\nline2",end\n',
            {},
        ),
        (
            "Custom delimiter (semicolon)",
            "x;y;z\n1;2;3\n",
            {"delimiter": ";"},
        ),
        (
            "Empty fields",
            ",a,,b,\n",
            {},
        ),
        (
            "No trailing newline",
            "alpha,beta,gamma",
            {},
        ),
        (
            "Windows line endings",
            "a,b\r\nc,d\r\n",
            {},
        ),
    ]

    for description, text, kwargs in samples:
        stdlib_result = parse_csv(text, **kwargs)
        manual_result = parse_csv_manual(text, **kwargs)
        match = stdlib_result == manual_result
        status = "OK" if match else "MISMATCH"
        print(f"[{status}] {description}")
        print(f"       stdlib : {stdlib_result}")
        print(f"       manual : {manual_result}")
        if not match:
            print("       *** Results differ! ***")
        print()
