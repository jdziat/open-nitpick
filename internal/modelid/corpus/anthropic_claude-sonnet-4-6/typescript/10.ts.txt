import crypto from "crypto";

interface ShortenedURL {
  originalUrl: string;
  expiresAt: number;
}

const store = new Map<string, ShortenedURL>();

const DEFAULT_TTL_MS = 24 * 60 * 60 * 1000; // 24 hours

function generateCode(length = 6): string {
  return crypto.randomBytes(length).toString("base64url").slice(0, length);
}

function shorten(originalUrl: string, ttlMs: number = DEFAULT_TTL_MS): string {
  if (!originalUrl || typeof originalUrl !== "string") {
    throw new Error("Invalid URL provided.");
  }

  let code: string;
  do {
    code = generateCode();
  } while (store.has(code));

  store.set(code, {
    originalUrl,
    expiresAt: Date.now() + ttlMs,
  });

  return code;
}

function resolve(code: string): string {
  const entry = store.get(code);

  if (!entry) {
    throw new Error(`No URL found for code: ${code}`);
  }

  if (Date.now() > entry.expiresAt) {
    store.delete(code);
    throw new Error(`URL for code "${code}" has expired.`);
  }

  return entry.originalUrl;
}

function remove(code: string): boolean {
  return store.delete(code);
}

function purgeExpired(): number {
  const now = Date.now();
  let count = 0;
  for (const [code, entry] of store.entries()) {
    if (now > entry.expiresAt) {
      store.delete(code);
      count++;
    }
  }
  return count;
}

// --- Demo / Test ---

function main() {
  console.log("=== URL Shortener Demo ===\n");

  // Shorten a URL with default TTL
  const code1 = shorten("https://www.example.com/some/very/long/path?query=1");
  console.log(`Shortened URL code: ${code1}`);
  console.log(`Resolved: ${resolve(code1)}\n`);

  // Shorten another URL with a custom TTL
  const code2 = shorten("https://openai.com/research/gpt-4", 5000); // 5 seconds
  console.log(`Shortened with 5s TTL, code: ${code2}`);
  console.log(`Resolved immediately: ${resolve(code2)}\n`);

  // Simulate expiry by shortening with a very short TTL
  const code3 = shorten("https://expired.example.com", 1); // 1 ms TTL
  console.log(`Shortened with 1ms TTL, code: ${code3}`);

  setTimeout(() => {
    try {
      resolve(code3);
    } catch (err) {
      console.log(`Expected expiry error: ${(err as Error).message}`);
    }

    // Purge expired entries
    const purged = purgeExpired();
    console.log(`\nPurged ${purged} expired entry/entries.`);
    console.log(`Store size after purge: ${store.size}`);

    // Remove a code manually
    const removed = remove(code1);
    console.log(`\nManually removed code "${code1}": ${removed}`);
    console.log(`Store size after removal: ${store.size}`);

    // Try resolving removed code
    try {
      resolve(code1);
    } catch (err) {
      console.log(`Expected not-found error: ${(err as Error).message}`);
    }
  }, 50);
}

main();
