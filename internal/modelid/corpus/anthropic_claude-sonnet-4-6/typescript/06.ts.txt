class BSTNode {
  value: number;
  left: BSTNode | null;
  right: BSTNode | null;

  constructor(value: number) {
    this.value = value;
    this.left = null;
    this.right = null;
  }
}

class BinarySearchTree {
  root: BSTNode | null;

  constructor() {
    this.root = null;
  }

  insert(value: number): void {
    const newNode = new BSTNode(value);
    if (this.root === null) {
      this.root = newNode;
      return;
    }
    let current: BSTNode = this.root;
    while (true) {
      if (value === current.value) {
        return;
      }
      if (value < current.value) {
        if (current.left === null) {
          current.left = newNode;
          return;
        }
        current = current.left;
      } else {
        if (current.right === null) {
          current.right = newNode;
          return;
        }
        current = current.right;
      }
    }
  }

  search(value: number): BSTNode | null {
    let current: BSTNode | null = this.root;
    while (current !== null) {
      if (value === current.value) {
        return current;
      } else if (value < current.value) {
        current = current.left;
      } else {
        current = current.right;
      }
    }
    return null;
  }

  private findMin(node: BSTNode): BSTNode {
    let current = node;
    while (current.left !== null) {
      current = current.left;
    }
    return current;
  }

  private deleteNode(node: BSTNode | null, value: number): BSTNode | null {
    if (node === null) {
      return null;
    }
    if (value < node.value) {
      node.left = this.deleteNode(node.left, value);
    } else if (value > node.value) {
      node.right = this.deleteNode(node.right, value);
    } else {
      if (node.left === null && node.right === null) {
        return null;
      } else if (node.left === null) {
        return node.right;
      } else if (node.right === null) {
        return node.left;
      } else {
        const minNode = this.findMin(node.right);
        node.value = minNode.value;
        node.right = this.deleteNode(node.right, minNode.value);
      }
    }
    return node;
  }

  delete(value: number): void {
    this.root = this.deleteNode(this.root, value);
  }

  private inOrderTraversal(node: BSTNode | null, result: number[]): void {
    if (node === null) {
      return;
    }
    this.inOrderTraversal(node.left, result);
    result.push(node.value);
    this.inOrderTraversal(node.right, result);
  }

  inOrder(): number[] {
    const result: number[] = [];
    this.inOrderTraversal(this.root, result);
    return result;
  }
}

// Example usage
const bst = new BinarySearchTree();

bst.insert(10);
bst.insert(5);
bst.insert(15);
bst.insert(3);
bst.insert(7);
bst.insert(12);
bst.insert(20);

console.log("In-order traversal after insertions:");
console.log(bst.inOrder());

console.log("\nSearching for value 7:");
const found = bst.search(7);
console.log(found ? `Found node with value: ${found.value}` : "Not found");

console.log("\nSearching for value 99:");
const notFound = bst.search(99);
console.log(notFound ? `Found node with value: ${notFound.value}` : "Not found");

console.log("\nDeleting leaf node 3:");
bst.delete(3);
console.log(bst.inOrder());

console.log("\nDeleting node with one child (20 has no children, let's delete 15 which has children):");
bst.delete(15);
console.log(bst.inOrder());

console.log("\nDeleting root node 10:");
bst.delete(10);
console.log(bst.inOrder());
