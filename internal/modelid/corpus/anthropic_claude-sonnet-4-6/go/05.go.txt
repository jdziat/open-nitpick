package main

import (
	"bufio"
	"fmt"
	"os"
	"sort"
	"strings"
	"unicode"
)

func tokenize(text string) []string {
	var tokens []string
	var current strings.Builder

	for _, r := range text {
		if unicode.IsLetter(r) || unicode.IsDigit(r) || r == '\'' {
			current.WriteRune(unicode.ToLower(r))
		} else {
			if current.Len() > 0 {
				word := strings.Trim(current.String(), "'")
				if len(word) > 0 {
					tokens = append(tokens, word)
				}
				current.Reset()
			}
		}
	}

	if current.Len() > 0 {
		word := strings.Trim(current.String(), "'")
		if len(word) > 0 {
			tokens = append(tokens, word)
		}
	}

	return tokens
}

type wordCount struct {
	word  string
	count int
}

func topK(freq map[string]int, k int) []wordCount {
	results := make([]wordCount, 0, len(freq))
	for w, c := range freq {
		results = append(results, wordCount{w, c})
	}

	sort.Slice(results, func(i, j int) bool {
		if results[i].count != results[j].count {
			return results[i].count > results[j].count
		}
		return results[i].word < results[j].word
	})

	if k > len(results) {
		k = len(results)
	}
	return results[:k]
}

func countWords(text string) map[string]int {
	freq := make(map[string]int)
	for _, word := range tokenize(text) {
		freq[word]++
	}
	return freq
}

func readInput() string {
	var sb strings.Builder
	scanner := bufio.NewScanner(os.Stdin)
	scanner.Buffer(make([]byte, 1024*1024), 1024*1024)
	for scanner.Scan() {
		sb.WriteString(scanner.Text())
		sb.WriteRune('\n')
	}
	return sb.String()
}

func main() {
	k := 10
	if len(os.Args) > 1 {
		fmt.Sscanf(os.Args[1], "%d", &k)
	}
	if k <= 0 {
		fmt.Fprintln(os.Stderr, "k must be a positive integer")
		os.Exit(1)
	}

	text := readInput()
	freq := countWords(text)

	if len(freq) == 0 {
		fmt.Println("No words found.")
		return
	}

	results := topK(freq, k)

	maxWordLen := 0
	for _, wc := range results {
		if len(wc.word) > maxWordLen {
			maxWordLen = len(wc.word)
		}
	}

	fmt.Printf("%-*s  %s\n", maxWordLen, "WORD", "COUNT")
	fmt.Printf("%s  %s\n", strings.Repeat("-", maxWordLen), "-----")
	for _, wc := range results {
		fmt.Printf("%-*s  %d\n", maxWordLen, wc.word, wc.count)
	}
}
