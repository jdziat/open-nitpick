package retry

import (
	"errors"
	"math/rand"
	"time"
)

// Config holds the configuration for the retry helper.
type Config struct {
	// MaxAttempts is the maximum number of times the function will be called.
	MaxAttempts int
	// InitialDelay is the delay before the second attempt.
	InitialDelay time.Duration
	// MaxDelay caps the exponential backoff delay.
	MaxDelay time.Duration
	// Multiplier is the factor by which the delay grows each attempt.
	Multiplier float64
	// JitterFraction is the fraction of the current delay added as random jitter (0.0 to 1.0).
	JitterFraction float64
}

// DefaultConfig returns a Config with sensible defaults.
func DefaultConfig() Config {
	return Config{
		MaxAttempts:    5,
		InitialDelay:   100 * time.Millisecond,
		MaxDelay:       30 * time.Second,
		Multiplier:     2.0,
		JitterFraction: 0.2,
	}
}

// ErrNoAttempts is returned when MaxAttempts is less than 1.
var ErrNoAttempts = errors.New("retry: MaxAttempts must be at least 1")

// Do calls fn up to cfg.MaxAttempts times. If fn returns nil the function
// returns immediately with a nil error. If all attempts fail it returns the
// error from the last attempt.
//
// Between attempts the helper sleeps for a duration calculated as:
//
//	delay = min(InitialDelay * Multiplier^(attempt-1), MaxDelay)
//	delay += rand uniform in [0, JitterFraction * delay)
func Do(cfg Config, fn func() error) error {
	if cfg.MaxAttempts < 1 {
		return ErrNoAttempts
	}

	var lastErr error
	delay := cfg.InitialDelay

	for attempt := 1; attempt <= cfg.MaxAttempts; attempt++ {
		lastErr = fn()
		if lastErr == nil {
			return nil
		}

		// No sleep after the last attempt.
		if attempt == cfg.MaxAttempts {
			break
		}

		sleep := delay
		if cfg.JitterFraction > 0 {
			jitter := time.Duration(float64(sleep) * cfg.JitterFraction * rand.Float64())
			sleep += jitter
		}

		time.Sleep(sleep)

		// Grow delay for the next iteration.
		next := time.Duration(float64(delay) * cfg.Multiplier)
		if cfg.MaxDelay > 0 && next > cfg.MaxDelay {
			next = cfg.MaxDelay
		}
		delay = next
	}

	return lastErr
}

// DoSimple calls fn up to maxAttempts times using DefaultConfig (with the
// supplied maxAttempts substituted in) and returns the last error.
func DoSimple(maxAttempts int, fn func() error) error {
	cfg := DefaultConfig()
	cfg.MaxAttempts = maxAttempts
	return Do(cfg, fn)
}
