package main

import (
	"encoding/json"
	"errors"
	"fmt"
	"os"
	"reflect"
	"strconv"
	"strings"
)

// Config holds all application configuration values.
// Tags:
//
//	env:"ENV_VAR"      – environment variable name that overrides this field
//	required:"true"    – the field must be non-zero after loading
//	default:"value"    – value used when the field is zero and no env var is set
type Config struct {
	AppName  string `json:"app_name"  env:"APP_NAME"   default:"MyApp"      required:"true"`
	Host     string `json:"host"      env:"HOST"        default:"localhost"   required:"true"`
	Port     int    `json:"port"      env:"PORT"        default:"8080"        required:"true"`
	Debug    bool   `json:"debug"     env:"DEBUG"       default:"false"`
	LogLevel string `json:"log_level" env:"LOG_LEVEL"   default:"info"        required:"true"`
	DSN      string `json:"dsn"       env:"DSN"                               required:"true"`
}

// loader handles the multi-step loading process.
type loader struct {
	filePath string
}

// newLoader creates a loader that reads from filePath.
// Pass an empty string to skip file loading.
func newLoader(filePath string) *loader {
	return &loader{filePath: filePath}
}

// Load returns a fully populated and validated Config.
func (l *loader) Load() (*Config, error) {
	cfg := &Config{}

	// Step 1: apply defaults from struct tags.
	if err := applyDefaults(cfg); err != nil {
		return nil, fmt.Errorf("applying defaults: %w", err)
	}

	// Step 2: overlay values from JSON file (if provided).
	if l.filePath != "" {
		if err := loadFile(l.filePath, cfg); err != nil {
			return nil, fmt.Errorf("loading file %q: %w", l.filePath, err)
		}
	}

	// Step 3: overlay values from environment variables.
	if err := applyEnv(cfg); err != nil {
		return nil, fmt.Errorf("applying env vars: %w", err)
	}

	// Step 4: validate required fields.
	if err := validate(cfg); err != nil {
		return nil, fmt.Errorf("validation: %w", err)
	}

	return cfg, nil
}

// applyDefaults sets fields to their default tag value when the field is zero.
func applyDefaults(cfg *Config) error {
	rv := reflect.ValueOf(cfg).Elem()
	rt := rv.Type()

	for i := 0; i < rt.NumField(); i++ {
		field := rv.Field(i)
		sf := rt.Field(i)

		def, ok := sf.Tag.Lookup("default")
		if !ok {
			continue
		}

		// Only apply when the field is currently the zero value.
		if !field.IsZero() {
			continue
		}

		if err := setField(field, def); err != nil {
			return fmt.Errorf("field %s: %w", sf.Name, err)
		}
	}
	return nil
}

// loadFile reads JSON from path and unmarshals it into cfg,
// overwriting any fields present in the file.
func loadFile(path string, cfg *Config) error {
	f, err := os.Open(path)
	if err != nil {
		if errors.Is(err, os.ErrNotExist) {
			// Missing file is not fatal; just skip.
			return nil
		}
		return err
	}
	defer f.Close()

	dec := json.NewDecoder(f)
	dec.DisallowUnknownFields()
	if err := dec.Decode(cfg); err != nil {
		return err
	}
	return nil
}

// applyEnv overrides fields whose env tag variable is set in the environment.
func applyEnv(cfg *Config) error {
	rv := reflect.ValueOf(cfg).Elem()
	rt := rv.Type()

	for i := 0; i < rt.NumField(); i++ {
		field := rv.Field(i)
		sf := rt.Field(i)

		envKey, ok := sf.Tag.Lookup("env")
		if !ok || envKey == "" {
			continue
		}

		val, set := os.LookupEnv(envKey)
		if !set {
			continue
		}

		if err := setField(field, val); err != nil {
			return fmt.Errorf("env var %s (field %s): %w", envKey, sf.Name, err)
		}
	}
	return nil
}

// validate checks that all fields marked required:"true" are non-zero.
func validate(cfg *Config) error {
	rv := reflect.ValueOf(cfg).Elem()
	rt := rv.Type()

	var missing []string
	for i := 0; i < rt.NumField(); i++ {
		field := rv.Field(i)
		sf := rt.Field(i)

		req, ok := sf.Tag.Lookup("required")
		if !ok {
			continue
		}
		if strings.EqualFold(req, "true") && field.IsZero() {
			name := sf.Tag.Get("json")
			if name == "" {
				name = sf.Name
			}
			missing = append(missing, name)
		}
	}

	if len(missing) > 0 {
		return fmt.Errorf("missing required fields: %s", strings.Join(missing, ", "))
	}
	return nil
}

// setField converts the string value s and assigns it to field.
func setField(field reflect.Value, s string) error {
	switch field.Kind() {
	case reflect.String:
		field.SetString(s)

	case reflect.Bool:
		b, err := strconv.ParseBool(s)
		if err != nil {
			return fmt.Errorf("cannot parse %q as bool: %w", s, err)
		}
		field.SetBool(b)

	case reflect.Int, reflect.Int8, reflect.Int16, reflect.Int32, reflect.Int64:
		n, err := strconv.ParseInt(s, 10, 64)
		if err != nil {
			return fmt.Errorf("cannot parse %q as int: %w", s, err)
		}
		field.SetInt(n)

	case reflect.Uint, reflect.Uint8, reflect.Uint16, reflect.Uint32, reflect.Uint64:
		n, err := strconv.ParseUint(s, 10, 64)
		if err != nil {
			return fmt.Errorf("cannot parse %q as uint: %w", s, err)
		}
		field.SetUint(n)

	case reflect.Float32, reflect.Float64:
		f, err := strconv.ParseFloat(s, 64)
		if err != nil {
			return fmt.Errorf("cannot parse %q as float: %w", s, err)
		}
		field.SetFloat(f)

	default:
		return fmt.Errorf("unsupported field kind %s", field.Kind())
	}
	return nil
}

func main() {
	// Example: load from "config.json" if it exists.
	// Set DSN via env:  DSN=postgres://localhost/mydb ./program
	// Override port:    PORT=9090 ./program

	l := newLoader("config.json")
	cfg, err := l.Load()
	if err != nil {
		fmt.Fprintf(os.Stderr, "configuration error: %v\n", err)
		os.Exit(1)
	}

	fmt.Printf("AppName  : %s\n", cfg.AppName)
	fmt.Printf("Host     : %s\n", cfg.Host)
	fmt.Printf("Port     : %d\n", cfg.Port)
	fmt.Printf("Debug    : %v\n", cfg.Debug)
	fmt.Printf("LogLevel : %s\n", cfg.LogLevel)
	fmt.Printf("DSN      : %s\n", cfg.DSN)
}
