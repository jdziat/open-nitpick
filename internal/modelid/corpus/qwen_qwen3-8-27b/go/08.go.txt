package priorityqueue

type PriorityQueue[T any] struct {
	less  func(a, b T) bool
	items []T
}

func New[T any](less func(a, b T) bool) *PriorityQueue[T] {
	return &PriorityQueue[T]{less: less}
}

func (p *PriorityQueue[T]) Push(v T) {
	p.items = append(p.items, v)
	i := len(p.items) - 1
	for i > 0 {
		parent := (i - 1) / 2
		if p.less(p.items[i], p.items[parent]) {
			p.items[i], p.items[parent] = p.items[parent], p.items[i]
			i = parent
		} else {
			break
		}
	}
}

func (p *PriorityQueue[T]) Pop() (T, bool) {
	if len(p.items) == 0 {
		var zero T
		return zero, false
	}

	top := p.items[0]
	n := len(p.items)
	last := p.items[n-1]
	var zero T
	p.items[n-1] = zero
	p.items = p.items[:n-1]

	if n > 1 {
		p.items[0] = last
		i := 0
		for {
			left := 2*i + 1
			if left >= len(p.items) {
				break
			}

			best := i
			if p.less(p.items[left], p.items[best]) {
				best = left
			}

			right := left + 1
			if right < len(p.items) && p.less(p.items[right], p.items[best]) {
				best = right
			}

			if best == i {
				break
			}

			p.items[i], p.items[best] = p.items[best], p.items[i]
			i = best
		}
	}

	return top, true
}

func (p *PriorityQueue[T]) Peek() (T, bool) {
	if len(p.items) == 0 {
		var zero T
		return zero, false
	}
	return p.items[0], true
}

func (p *PriorityQueue[T]) Len() int {
	return len(p.items)
}
