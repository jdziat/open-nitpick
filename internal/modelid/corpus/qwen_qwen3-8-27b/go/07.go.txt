package main

import (
	"encoding/json"
	"fmt"
	"os"
	"strconv"
	"strings"
)

type Config struct {
	Host           string `json:"host"`
	Port           int    `json:"port"`
	Database       string `json:"database"`
	TimeoutSeconds int    `json:"timeout_seconds"`
	LogLevel       string `json:"log_level"`
}

const (
	EnvHost           = "CONFIG_HOST"
	EnvPort           = "CONFIG_PORT"
	EnvDatabase       = "CONFIG_DATABASE"
	EnvTimeoutSeconds = "CONFIG_TIMEOUT_SECONDS"
	EnvLogLevel       = "CONFIG_LOG_LEVEL"
)

func defaultConfig() Config {
	return Config{
		Host:           "localhost",
		Port:           8080,
		Database:       "app",
		TimeoutSeconds: 30,
		LogLevel:       "info",
	}
}

func Load(path string) (*Config, error) {
	cfg := defaultConfig()

	if path != "" {
		data, err := os.ReadFile(path)
		if err != nil {
			if !os.IsNotExist(err) {
				return nil, fmt.Errorf("read %s: %w", path, err)
			}
		} else {
			if err := json.Unmarshal(data, &cfg); err != nil {
				return nil, fmt.Errorf("parse %s: %w", path, err)
			}
		}
	}

	if err := applyEnvironment(&cfg); err != nil {
		return nil, err
	}

	if err := cfg.Validate(); err != nil {
		return nil, err
	}

	return &cfg, nil
}

func applyEnvironment(cfg *Config) error {
	if v := os.Getenv(EnvHost); v != "" {
		cfg.Host = v
	}

	if v := os.Getenv(EnvPort); v != "" {
		n, err := strconv.Atoi(v)
		if err != nil {
			return fmt.Errorf("%s must be an integer: %w", EnvPort, err)
		}
		cfg.Port = n
	}

	if v := os.Getenv(EnvDatabase); v != "" {
		cfg.Database = v
	}

	if v := os.Getenv(EnvTimeoutSeconds); v != "" {
		n, err := strconv.Atoi(v)
		if err != nil {
			return fmt.Errorf("%s must be an integer: %w", EnvTimeoutSeconds, err)
		}
		cfg.TimeoutSeconds = n
	}

	if v := os.Getenv(EnvLogLevel); v != "" {
		cfg.LogLevel = v
	}

	return nil
}

func (c Config) Validate() error {
	var missing []string

	if c.Host == "" {
		missing = append(missing, "host")
	}
	if c.Port <= 0 {
		missing = append(missing, "port")
	}
	if c.Database == "" {
		missing = append(missing, "database")
	}
	if c.TimeoutSeconds <= 0 {
		missing = append(missing, "timeout_seconds")
	}
	if c.LogLevel == "" {
		missing = append(missing, "log_level")
	}

	if len(missing) > 0 {
		return fmt.Errorf("missing or invalid required config keys: %s", strings.Join(missing, ", "))
	}

	return nil
}

func main() {
	path := "config.json"
	if len(os.Args) > 1 {
		path = os.Args[1]
	}

	cfg, err := Load(path)
	if err != nil {
		fmt.Fprintln(os.Stderr, "error:", err)
		os.Exit(1)
	}

	out, err := json.MarshalIndent(cfg, "", "  ")
	if err != nil {
		fmt.Fprintln(os.Stderr, "error:", err)
		os.Exit(1)
	}

	fmt.Println(string(out))
}
