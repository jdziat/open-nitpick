from typing import List

__all__ = ["parse_csv"]


def parse_csv(text: str, delimiter: str = ",") -> List[List[str]]:
    """Parse a CSV string with quoted fields, escaped quotes, and a delimiter."""
    if not delimiter:
        raise ValueError("delimiter must be a non-empty string")

    rows: List[List[str]] = []
    row: List[str] = []
    field: List[str] = []
    in_quotes = False
    row_has_field = False

    i = 0
    n = len(text)
    dlen = len(delimiter)

    while i < n:
        if in_quotes:
            ch = text[i]
            if ch == '"':
                if i + 1 < n and text[i + 1] == '"':
                    field.append('"')
                    i += 2
                    continue
                in_quotes = False
                i += 1
                continue
            field.append(ch)
            i += 1
            continue

        ch = text[i]
        if not field and ch == '"':
            in_quotes = True
            row_has_field = True
