import threading
import time
import uuid


class URLShortener:
    def __init__(self):
        self._links = {}
        self._lock = threading.Lock()

    def shorten(self, url, ttl=None):
        with self._lock:
            code = uuid.uuid4().hex[:8]
            while code in self._links:
                code = uuid.uuid4().hex[:8]

            expires_at = None if ttl is None else time.monotonic() + ttl
            self._links[code] = (url, expires_at)
            return code

    def resolve(self, code):
        with self._lock:
            entry = self._links.get(code)
            if entry is None:
                raise KeyError(code)

            url, expires_at = entry
            if expires_at is not None and time.monotonic() >= expires_at:
                del self._links[code]
                raise KeyError(code)

            return url
