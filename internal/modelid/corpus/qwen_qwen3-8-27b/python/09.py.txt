from datetime import timedelta


def date_range(start, end, skip_weekends=False):
    """Yield each day between start and end inclusive.

    If skip_weekends is True, Saturday and Sunday are omitted.
    """
    step = 1 if start <= end else -1
    current = start

    while (step == 1 and current <= end) or (step == -1 and current >= end):
        if not skip_weekends or current.weekday() < 5:
            yield current
        current += timedelta(days=step)
