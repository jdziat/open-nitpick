#!/usr/bin/env python3
"""Count the most frequent words in text."""

import argparse
import re
import sys
from collections import Counter

_WORD_RE = re.compile(r"[^\W_]+(?:['’-][^\W_]+)*")


def normalize_words(text):
    """Return a list of casefolded words with punctuation removed."""
    return _WORD_RE.findall(text.casefold())


def word_frequency_counter(text, k=10):
    """Return the top k (word, count) pairs from text."""
    counts = Counter(normalize_words(text))
    if k is None:
        limit = len(counts)
    else:
        limit = max(0, k)
    return sorted(counts.items(), key=lambda item: (-item[1], item[0]))[:limit]


def _read_text(paths):
    """Read text from the given paths, or stdin if paths is empty."""
    if not paths:
        return sys.stdin.read()
    chunks = []
    for path in paths:
        with open(path, "r", encoding="utf-8", errors="replace") as handle:
            chunks.append(handle.read())
    return "\n".join(chunks)


def main(argv=None):
    parser = argparse.ArgumentParser(description="Print the top K most frequent words.")
    parser.add_argument("input", nargs="*", help="Text files to read. Reads stdin if omitted.")
    parser.add_argument("-k", "--top", type=int, default=10, help="Number of top words to print.")
    args = parser.parse_args(argv)

    for word, count in word_frequency_counter(_read_text(args.input), args.top):
        print(f"{count}\t{word}")
    return 0


if __name__ == "__main__":
    sys.exit(main())
