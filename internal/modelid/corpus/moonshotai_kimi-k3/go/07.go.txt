package main

import (
	"encoding/json"
	"fmt"
	"os"
	"reflect"
	"strconv"
	"strings"
	"time"
)

// envPrefix is prepended to every override variable, e.g. APP_SERVER_PORT.
const envPrefix = "APP"

// Duration is a time.Duration that marshals to/from a JSON string such as
// "250ms" or "5s" (a JSON number is treated as nanoseconds).
type Duration time.Duration

func (d Duration) String() string { return time.Duration(d).String() }

func (d Duration) MarshalJSON() ([]byte, error) {
	return json.Marshal(time.Duration(d).String())
}

func (d *Duration) UnmarshalJSON(data []byte) error {
	if string(data) == "null" {
		return nil
	}
	var raw any
	if err := json.Unmarshal(data, &raw); err != nil {
		return err
	}
	switch v := raw.(type) {
	case float64:
		*d = Duration(time.Duration(v))
		return nil
	case string:
		parsed, err := time.ParseDuration(v)
		if err != nil {
			return fmt.Errorf("invalid duration %q", v)
		}
		*d = Duration(parsed)
		return nil
	default:
		return fmt.Errorf("invalid duration value %v", raw)
	}
}

// Config is the root of the application configuration.
type Config struct {
	Server   ServerConfig   `json:"server"`
	Database DatabaseConfig `json:"database"`
	Log      LogConfig      `json:"log"`
}

type ServerConfig struct {
	Host         string   `json:"host"`
	Port         int      `json:"port"`
	ReadTimeout  Duration `json:"read_timeout"`
	WriteTimeout Duration `json:"write_timeout"`
	Debug        bool     `json:"debug"`
}

type DatabaseConfig struct {
	URL      string `json:"url" validate:"required"`
	Username string `json:"username" validate:"required"`
	Password string `json:"password" validate:"required"`
	MaxConns int    `json:"max_conns"`
}

type LogConfig struct {
	Level  string   `json:"level"`
	Format string   `json:"format"`
	Tags   []string `json:"tags"`
}

// Validator may be implemented for checks beyond `validate:"required"`.
type Validator interface {
	Validate() error
}

// Validate enforces value constraints on the fully merged configuration.
func (c *Config) Validate() error {
	if c.Server.Port < 1 || c.Server.Port > 65535 {
		return fmt.Errorf("server.port must be between 1 and 65535 (got %d)", c.Server.Port)
	}
	if c.Database.MaxConns < 1 {
		return fmt.Errorf("database.max_conns must be at least 1 (got %d)", c.Database.MaxConns)
	}
	switch strings.ToLower(c.Log.Level) {
	case "debug", "info", "warn", "error":
	default:
		return fmt.Errorf("log.level must be debug, info, warn or error (got %q)", c.Log.Level)
	}
	switch strings.ToLower(c.Log.Format) {
	case "json", "text":
	default:
		return fmt.Errorf("log.format must be json or text (got %q)", c.Log.Format)
	}
	return nil
}

// DefaultConfig returns a Config populated with sensible defaults.
func DefaultConfig() *Config {
	return &Config{
		Server: ServerConfig{
			Host:         "0.0.0.0",
			Port:         8080,
			ReadTimeout:  Duration(5 * time.Second),
			WriteTimeout: Duration(10 * time.Second),
			Debug:        false,
		},
		Database: DatabaseConfig{
			MaxConns: 10,
		},
		Log: LogConfig{
			Level:  "info",
			Format: "json",
		},
	}
}

// Load layers configuration from three sources, lowest precedence first:
// built-in defaults, the JSON file at path, then APP_* environment
// variables. A missing file is not an error; the result is validated
// before it is returned.
func Load(path string) (*Config, error) {
	cfg := DefaultConfig()

	if data, err := os.ReadFile(path); err == nil {
		if err := json.Unmarshal(data, cfg); err != nil {
			return nil, fmt.Errorf("parse %s: %w", path, err)
		}
	} else if !os.IsNotExist(err) {
		return nil, fmt.Errorf("read %s: %w", path, err)
	}

	if err := applyEnv(reflect.ValueOf(cfg).Elem(), envPrefix); err != nil {
		return nil, fmt.Errorf("environment override: %w", err)
	}

	if err := validateConfig(cfg); err != nil {
		return nil, err
	}

	return cfg, nil
}

// fieldName resolves a struct field's configuration key from its json tag.
func fieldName(f reflect.StructField) (name string, skip bool) {
	tag := f.Tag.Get("json")
	if tag == "-" {
		return "", true
	}
	if idx := strings.Index(tag, ","); idx >= 0 {
		tag = tag[:idx]
	}
	if tag == "" {
		tag = strings.ToLower(f.Name)
	}
	return tag, false
}

// applyEnv walks the config tree, overriding any leaf field whose
// corresponding environment variable (PREFIX_SECTION_KEY) is set.
func applyEnv(v reflect.Value, prefix string) error {
	t := v.Type()
	for i := 0; i < v.NumField(); i++ {
		field := v.Field(i)
		if !field.CanSet() {
			continue
		}
		name, skip := fieldName(t.Field(i))
		if skip {
			continue
		}
		envName := prefix + "_" + strings.ToUpper(strings.ReplaceAll(name, "-", "_"))

		if field.Kind() == reflect.Struct {
			if err := applyEnv(field, envName); err != nil {
				return err
			}
			continue
		}

		value, ok := os.LookupEnv(envName)
		if !ok {
			continue
		}
		if err := setFieldFromString(field, value); err != nil {
			return fmt.Errorf("%s: %w", envName, err)
		}
	}
	return nil
}

var durationType = reflect.TypeOf(Duration(0))

func setFieldFromString(field reflect.Value, value string) error {
	if field.Type() == durationType {
		d, err := time.ParseDuration(value)
		if err != nil {
			return fmt.Errorf("invalid duration %q", value)
		}
		field.SetInt(int64(d))
		return nil
	}

	switch field.Kind() {
	case reflect.String:
		field.SetString(value)
	case reflect.Bool:
		b, err := strconv.ParseBool(value)
		if err != nil {
			return fmt.Errorf("invalid bool %q", value)
		}
		field.SetBool(b)
	case reflect.Int, reflect.Int8, reflect.Int16, reflect.Int32, reflect.Int64:
		n, err := strconv.ParseInt(value, 10, field.Type().Bits())
		if err != nil {
			return fmt.Errorf("invalid integer %q", value)
		}
		field.SetInt(n)
	case reflect.Uint, reflect.Uint8, reflect.Uint16, reflect.Uint32, reflect.Uint64:
		n, err := strconv.ParseUint(value, 10, field.Type().Bits())
		if err != nil {
			return fmt.Errorf("invalid unsigned integer %q", value)
		}
		field.SetUint(n)
	case reflect.Float32, reflect.Float64:
		f, err := strconv.ParseFloat(value, field.Type().Bits())
		if err != nil {
			return fmt.Errorf("invalid float %q", value)
		}
		field.SetFloat(f)
	case reflect.Slice: // comma-separated values, e.g. APP_LOG_TAGS="api,prod"
		parts := strings.Split(value, ",")
		slice := reflect.MakeSlice(field.Type(), len(parts), len(parts))
		for i, part := range parts {
			if err := setFieldFromString(slice.Index(i), strings.TrimSpace(part)); err != nil {
				return err
			}
		}
		field.Set(slice)
	default:
		return fmt.Errorf("unsupported type %s", field.Type())
	}
	return nil
}

// validateConfig checks `validate:"required"` tags, then runs custom
// validation if the config implements Validator.
func validateConfig(cfg *Config) error {
	var missing []string
	collectRequired(reflect.ValueOf(cfg).Elem(), "", &missing)
	if len(missing) > 0 {
		hints := make([]string, len(missing))
		for i, key := range missing {
			envName := envPrefix + "_" + strings.ToUpper(strings.ReplaceAll(key, ".", "_"))
			hints[i] = fmt.Sprintf("%s (set in the config file or via %s)", key, envName)
		}
		return fmt.Errorf("missing required configuration: %s", strings.Join(hints, ", "))
	}
	if v, ok := any(cfg).(Validator); ok {
		if err := v.Validate(); err != nil {
			return fmt.Errorf("invalid configuration: %w", err)
		}
	}
	return nil
}

func collectRequired(v reflect.Value, path string, missing *[]string) {
	t := v.Type()
	for i := 0; i < v.NumField(); i++ {
		field := v.Field(i)
		sf := t.Field(i)
		if sf.PkgPath != "" {
			continue // unexported
		}
		name, skip := fieldName(sf)
		if skip {
			continue
		}
		full := name
		if path != "" {
			full = path + "." + name
		}
		if field.Kind() == reflect.Struct {
			collectRequired(field, full, missing)
			continue
		}
		if hasRequiredTag(sf) && field.IsZero() {
			*missing = append(*missing, full)
		}
	}
}

func hasRequiredTag(f reflect.StructField) bool {
	for _, opt := range strings.Split(f.Tag.Get("validate"), ",") {
		if strings.TrimSpace(opt) == "required" {
			return true
		}
	}
	return false
}

func main() {
	path := "config.json"
	if len(os.Args) > 1 {
		path = os.Args[1]
	}

	cfg, err := Load(path)
	if err != nil {
		fmt.Fprintf(os.Stderr, "config error: %v\n", err)
		os.Exit(1)
	}

	out, err := json.MarshalIndent(cfg, "", "  ")
	if err != nil {
		fmt.Fprintf(os.Stderr, "marshal config: %v\n", err)
		os.Exit(1)
	}
	fmt.Println(string(out))
}
