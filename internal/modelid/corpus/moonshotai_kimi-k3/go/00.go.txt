package main

import (
	"container/list"
	"fmt"
)

// LRUCache is a fixed-capacity cache that evicts the least recently used
// entry when full. Both Get and Put run in O(1) time.
type LRUCache struct {
	capacity int
	items    map[int]*list.Element
	order    *list.List // front = most recently used, back = least recently used
}

// entry is the key/value pair stored in each list element.
type entry struct {
	key   int
	value int
}

// NewLRUCache returns an LRU cache that holds at most capacity entries.
func NewLRUCache(capacity int) *LRUCache {
	if capacity <= 0 {
		panic("lru: capacity must be positive")
	}
	return &LRUCache{
		capacity: capacity,
		items:    make(map[int]*list.Element, capacity),
		order:    list.New(),
	}
}

// Get returns the value stored for key and marks the entry as most recently
// used. The second return value reports whether the key was present.
func (c *LRUCache) Get(key int) (int, bool) {
	elem, ok := c.items[key]
	if !ok {
		return 0, false
	}
	c.order.MoveToFront(elem)
	return elem.Value.(entry).value, true
}

// Put inserts or updates the value for key, marking it as most recently
// used. If the cache is full, the least recently used entry is evicted.
func (c *LRUCache) Put(key, value int) {
	if elem, ok := c.items[key]; ok {
		elem.Value = entry{key, value}
		c.order.MoveToFront(elem)
		return
	}
	if c.order.Len() == c.capacity {
		oldest := c.order.Back()
		if oldest != nil {
			c.order.Remove(oldest)
			delete(c.items, oldest.Value.(entry).key)
		}
	}
	c.items[key] = c.order.PushFront(entry{key, value})
}

// Len returns the number of entries currently in the cache.
func (c *LRUCache) Len() int {
	return c.order.Len()
}

func main() {
	cache := NewLRUCache(2)

	cache.Put(1, 10)
	cache.Put(2, 20)

	if v, ok := cache.Get(1); ok {
		fmt.Println("get(1):", v) // 10; key 1 is now most recently used
	}

	cache.Put(3, 30) // evicts key 2

	if _, ok := cache.Get(2); !ok {
		fmt.Println("get(2): miss (evicted)")
	}

	cache.Put(4, 40) // evicts key 1

	if _, ok := cache.Get(1); !ok {
		fmt.Println("get(1): miss (evicted)")
	}
	if v, ok := cache.Get(3); ok {
		fmt.Println("get(3):", v)
	}
	if v, ok := cache.Get(4); ok {
		fmt.Println("get(4):", v)
	}
	fmt.Println("len:", cache.Len())
}
