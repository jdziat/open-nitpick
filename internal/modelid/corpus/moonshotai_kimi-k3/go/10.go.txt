package main

import (
	"crypto/rand"
	"encoding/base64"
	"errors"
	"fmt"
	"sync"
	"time"
)

var ErrNotFound = errors.New("short code not found or expired")

type entry struct {
	url       string
	expiresAt time.Time
}

type Shortener struct {
	mu      sync.RWMutex
	entries map[string]entry
	ttl     time.Duration
}

func NewShortener(ttl time.Duration) *Shortener {
	return &Shortener{
		entries: make(map[string]entry),
		ttl:     ttl,
	}
}

func (s *Shortener) Shorten(url string) (string, error) {
	code, err := generateCode(6)
	if err != nil {
		return "", err
	}
	s.mu.Lock()
	s.entries[code] = entry{url: url, expiresAt: time.Now().Add(s.ttl)}
	s.mu.Unlock()
	return code, nil
}

func (s *Shortener) Resolve(code string) (string, error) {
	s.mu.RLock()
	e, ok := s.entries[code]
	s.mu.RUnlock()
	if !ok || time.Now().After(e.expiresAt) {
		return "", ErrNotFound
	}
	return e.url, nil
}

func generateCode(n int) (string, error) {
	b := make([]byte, n)
	if _, err := rand.Read(b); err != nil {
		return "", err
	}
	return base64.RawURLEncoding.EncodeToString(b)[:n], nil
}

func main() {
	s := NewShortener(5 * time.Minute)

	code, err := s.Shorten("https://example.com/some/very/long/url")
	if err != nil {
		panic(err)
	}
	fmt.Println("short code:", code)

	url, err := s.Resolve(code)
	if err != nil {
		panic(err)
	}
	fmt.Println("resolved:", url)
}
