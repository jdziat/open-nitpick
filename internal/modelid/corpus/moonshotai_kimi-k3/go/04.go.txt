package main

import (
	"fmt"
	"strings"
	"unicode"
)

// slugify lowercases text, replaces runs of non-alphanumeric characters
// with single hyphens, trims leading/trailing hyphens, and handles
// Unicode letters and numbers.
func slugify(s string) string {
	var b strings.Builder
	b.Grow(len(s))

	lastWasHyphen := false
	for _, r := range strings.ToLower(s) {
		if unicode.IsLetter(r) || unicode.IsNumber(r) {
			b.WriteRune(r)
			lastWasHyphen = false
		} else if !lastWasHyphen && b.Len() > 0 {
			b.WriteByte('-')
			lastWasHyphen = true
		}
	}

	return strings.TrimRight(b.String(), "-")
}

func main() {
	fmt.Println(slugify("Hello, World!"))
	fmt.Println(slugify("  Déjà Vu — 2024  "))
	fmt.Println(slugify("Café & Restaurant"))
	fmt.Println(slugify("---Multiple---Separators---"))
	fmt.Println(slugify("日本語のタイトル"))
}
