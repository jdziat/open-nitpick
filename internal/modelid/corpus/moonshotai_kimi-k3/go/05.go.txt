package main

import (
	"bufio"
	"flag"
	"fmt"
	"io"
	"os"
	"sort"
	"strings"
	"unicode"
)

// normalize lowercases a token and strips leading/trailing punctuation
// and symbols, leaving internal characters (e.g. apostrophes) intact.
func normalize(token string) string {
	token = strings.TrimFunc(token, func(r rune) bool {
		return unicode.IsPunct(r) || unicode.IsSymbol(r)
	})
	return strings.ToLower(token)
}

// countWords builds a word-frequency map from all text read from r.
func countWords(r io.Reader) (map[string]int, error) {
	counts := make(map[string]int)
	scanner := bufio.NewScanner(r)
	scanner.Split(bufio.ScanWords)
	for scanner.Scan() {
		if w := normalize(scanner.Text()); w != "" {
			counts[w]++
		}
	}
	return counts, scanner.Err()
}

type wordFreq struct {
	word  string
	count int
}

// topK returns the k most frequent words; ties are broken alphabetically.
// A negative k returns every word.
func topK(counts map[string]int, k int) []wordFreq {
	freqs := make([]wordFreq, 0, len(counts))
	for w, c := range counts {
		freqs = append(freqs, wordFreq{w, c})
	}
	sort.Slice(freqs, func(i, j int) bool {
		if freqs[i].count != freqs[j].count {
			return freqs[i].count > freqs[j].count
		}
		return freqs[i].word < freqs[j].word
	})
	if k >= 0 && k < len(freqs) {
		freqs = freqs[:k]
	}
	return freqs
}

func main() {
	k := flag.Int("k", 10, "number of top words to report (negative for all)")
	flag.Parse()

	var in io.Reader = os.Stdin
	if flag.NArg() > 0 {
		f, err := os.Open(flag.Arg(0))
		if err != nil {
			fmt.Fprintln(os.Stderr, "wordfreq:", err)
			os.Exit(1)
		}
		defer f.Close()
		in = f
	}

	counts, err := countWords(in)
	if err != nil {
		fmt.Fprintln(os.Stderr, "wordfreq:", err)
		os.Exit(1)
	}

	for _, wf := range topK(counts, *k) {
		fmt.Printf("%s\t%d\n", wf.word, wf.count)
	}
}
