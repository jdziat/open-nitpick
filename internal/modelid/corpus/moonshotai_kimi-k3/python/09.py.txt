"""Date-range iterator yielding each day between two dates, inclusive."""

from datetime import date, datetime, timedelta


def _to_date(value):
    """Coerce a date or datetime to a plain date."""
    if isinstance(value, datetime):
        return value.date()
    if isinstance(value, date):
        return value
    raise TypeError(
        f"expected datetime.date or datetime.datetime, got {type(value).__name__}"
    )


def date_range(start, end, skip_weekends=False):
    """Yield each day from ``start`` to ``end``, inclusive.

    Args:
        start: The first date in the range (datetime.date or datetime.datetime).
        end: The last date in the range (datetime.date or datetime.datetime).
        skip_weekends: If True, Saturdays and Sundays are not yielded.

    Yields:
        datetime.date objects, one per day in the range.

    Raises:
        TypeError: If start or end is not a date or datetime.
        ValueError: If start is after end.
    """
    start = _to_date(start)
    end = _to_date(end)
    if start > end:
        raise ValueError("start must not be after end")

    one_day = timedelta(days=1)
    current = start
    while current <= end:
        if not skip_weekends or current.weekday() < 5:
            yield current
        current += one_day


if __name__ == "__main__":
    first = date(2024, 6, 1)
    last = date(2024, 6, 10)

    print("All days:")
    for day in date_range(first, last):
        print(f"  {day:%Y-%m-%d %a}")

    print("Weekdays only:")
    for day in date_range(first, last, skip_weekends=True):
        print(f"  {day:%Y-%m-%d %a}")
