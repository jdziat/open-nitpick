"""A simple matrix type with multiplication, transpose, and determinant."""


class MatrixError(ValueError):
    """Raised when a matrix operation fails due to a shape mismatch."""


class Matrix:
    """A 2-D matrix of numbers backed by a list of lists."""

    def __init__(self, rows):
        rows = [list(row) for row in rows]
        if not rows or not rows[0]:
            raise MatrixError("matrix must have at least one row and one column")
        width = len(rows[0])
        for row in rows:
            if len(row) != width:
                raise MatrixError("all rows must have the same length")
        self._rows = rows

    @property
    def shape(self):
        return (len(self._rows), len(self._rows[0]))

    @property
    def rows(self):
        return self.shape[0]

    @property
    def cols(self):
        return self.shape[1]

    def is_square(self):
        return self.rows == self.cols

    def __getitem__(self, index):
        return self._rows[index]

    def __eq__(self, other):
        return isinstance(other, Matrix) and self._rows == other._rows

    def __repr__(self):
        return "Matrix({!r})".format(self._rows)

    def transpose(self):
        """Return a new matrix with rows and columns swapped."""
        return Matrix([list(col) for col in zip(*self._rows)])

    @property
    def T(self):
        return self.transpose()

    def __mul__(self, other):
        if isinstance(other, Matrix):
            return self._matmul(other)
        if isinstance(other, (int, float, complex)):
            return Matrix([[value * other for value in row] for row in self._rows])
        return NotImplemented

    def __rmul__(self, other):
        if isinstance(other, (int, float, complex)):
            return Matrix([[other * value for value in row] for row in self._rows])
        return NotImplemented

    def __matmul__(self, other):
        if isinstance(other, Matrix):
            return self._matmul(other)
        return NotImplemented

    def _matmul(self, other):
        if self.cols != other.rows:
            raise MatrixError(
                "cannot multiply {}x{} matrix by {}x{} matrix".format(
                    self.rows, self.cols, other.rows, other.cols
                )
            )
        other_cols = other.transpose()._rows
        return Matrix(
            [
                [sum(a * b for a, b in zip(row, col)) for col in other_cols]
                for row in self._rows
            ]
        )

    def determinant(self):
        """Return the determinant of a square matrix."""
        if not self.is_square():
            raise MatrixError(
                "determinant is only defined for square matrices, "
                "got {}x{}".format(self.rows, self.cols)
            )
        return _determinant(self._rows)

    det = determinant


def _determinant(rows):
    """Compute a determinant by Laplace expansion along the first row."""
    n = len(rows)
    if n == 1:
        return rows[0][0]
    if n == 2:
        return rows[0][0] * rows[1][1] - rows[0][1] * rows[1][0]
    total = 0
    for col in range(n):
        minor = [row[:col] + row[col + 1:] for row in rows[1:]]
        sign = -1 if col % 2 else 1
        total += sign * rows[0][col] * _determinant(minor)
    return total


if __name__ == "__main__":
    a = Matrix([[1, 2, 3], [4, 5, 6]])
    b = Matrix([[7, 8], [9, 10], [11, 12]])
    print("a =", a)
    print("b =", b)
    print("a * b =", a * b)
    print("a @ b =", a @ b)
    print("2 * a =", 2 * a)
    print("a.T =", a.T)

    c = Matrix([[6, 1, 1], [4, -2, 5], [2, 8, 7]])
    print("det(c) =", c.determinant())

    try:
        a * a
    except MatrixError as exc:
        print("shape error:", exc)

    try:
        a.determinant()
    except MatrixError as exc:
        print("shape error:", exc)
