class PriorityQueue:
    """A priority queue backed by a binary heap.

    Ordering is defined by a caller-supplied comparison function
    ``compare(a, b)`` which must return a negative number if ``a`` has
    higher priority (comes out first), zero if they are equivalent, and
    a positive number if ``b`` has higher priority. If omitted, natural
    ordering is used, producing a min-heap.
    """

    def __init__(self, compare=None, items=None):
        if compare is None:
            compare = lambda a, b: (a > b) - (a < b)
        self._compare = compare
        self._heap = []
        if items:
            self._heap = list(items)
            self._heapify()

    def __len__(self):
        return len(self._heap)

    def __bool__(self):
        return bool(self._heap)

    def __repr__(self):
        return f"{type(self).__name__}({self._heap!r})"

    def is_empty(self):
        """Return True if the queue contains no items."""
        return not self._heap

    def push(self, item):
        """Insert ``item`` into the queue. O(log n)."""
        self._heap.append(item)
        self._sift_up(len(self._heap) - 1)

    def peek(self):
        """Return the highest-priority item without removing it. O(1)."""
        if not self._heap:
            raise IndexError("peek from empty priority queue")
        return self._heap[0]

    def pop(self):
        """Remove and return the highest-priority item. O(log n)."""
        heap = self._heap
        if not heap:
            raise IndexError("pop from empty priority queue")
        top = heap[0]
        last = heap.pop()
        if heap:
            heap[0] = last
            self._sift_down(0)
        return top

    def _heapify(self):
        for index in range(len(self._heap) // 2 - 1, -1, -1):
            self._sift_down(index)

    def _sift_up(self, index):
        heap = self._heap
        compare = self._compare
        item = heap[index]
        while index > 0:
            parent = (index - 1) // 2
            if compare(item, heap[parent]) < 0:
                heap[index] = heap[parent]
                index = parent
            else:
                break
        heap[index] = item

    def _sift_down(self, index):
        heap = self._heap
        compare = self._compare
        size = len(heap)
        item = heap[index]
        while True:
            left = 2 * index + 1
            if left >= size:
                break
            right = left + 1
            child = left
            if right < size and compare(heap[right], heap[left]) < 0:
                child = right
            if compare(heap[child], item) < 0:
                heap[index] = heap[child]
                index = child
            else:
                break
        heap[index] = item
