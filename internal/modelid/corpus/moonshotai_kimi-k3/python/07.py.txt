"""JSON configuration loader with defaults, environment overrides, and validation.

Resolution order, from lowest to highest precedence:

1. ``defaults`` passed to :func:`load_config`
2. The JSON file, if a path is given
3. Environment variables

Environment variables are named ``{PREFIX}{SEPARATOR}KEY`` where nested keys
are joined with the separator. For example, with prefix ``APP`` and the
default separator ``__``, the variable ``APP_DATABASE__HOST=db.local``
overrides ``config["database"]["host"]``. Variable names are lowercased
before being applied, and values are parsed as JSON when possible (so
``true``, ``8080`` and ``[1, 2]`` become native Python values), otherwise
they are kept as strings.

Required keys are given as dotted paths (e.g. ``"database.host"``) and must
be present and not ``None`` after all sources are merged.

Example::

    config = load_config(
        "settings.json",
        defaults={"server": {"host": "127.0.0.1", "port": 8080}},
        required=["server.host", "api_key"],
        env_prefix="MYAPP",
    )
    port = config["server"]["port"]
    api_key = config.require("api_key")
"""

from __future__ import annotations

import json
import os
from collections.abc import Iterable, Mapping
from copy import deepcopy
from typing import Any, Optional

__all__ = [
    "Config",
    "ConfigError",
    "ConfigFileError",
    "ConfigValidationError",
    "load_config",
]

_MISSING = object()


class ConfigError(Exception):
    """Base class for configuration errors."""


class ConfigFileError(ConfigError):
    """The configuration file is missing, unreadable, or not a JSON object."""


class ConfigValidationError(ConfigError):
    """One or more required configuration keys are missing."""


def _deep_merge(base: Mapping[str, Any], override: Mapping[str, Any]) -> dict[str, Any]:
    """Return a recursive merge of *override* onto *base*; neither is mutated."""
    merged: dict[str, Any] = deepcopy(dict(base))
    for key, value in override.items():
        if key in merged and isinstance(merged[key], dict) and isinstance(value, Mapping):
            merged[key] = _deep_merge(merged[key], value)
        else:
            merged[key] = deepcopy(value)
    return merged


def _parse_env_value(raw: str) -> Any:
    """Interpret an environment variable string as JSON, falling back to str."""
    try:
        return json.loads(raw)
    except (ValueError, TypeError):
        return raw


def _set_nested(target: dict[str, Any], path: list[str], value: Any) -> None:
    """Set *value* at the nested *path* in *target*, creating dicts as needed."""
    cursor = target
    for part in path[:-1]:
        existing = cursor.get(part)
        if not isinstance(existing, dict):
            existing = {}
            cursor[part] = existing
        cursor = existing
    cursor[path[-1]] = value


def _lookup(data: Mapping[str, Any], dotted_key: str) -> Any:
    """Resolve a dotted key path, returning the ``_MISSING`` sentinel if absent."""
    cursor: Any = data
    for part in dotted_key.split("."):
        if not isinstance(cursor, Mapping) or part not in cursor:
            return _MISSING
        cursor = cursor[part]
    return cursor


def _read_json_file(path: str) -> dict[str, Any]:
    """Read and parse a JSON file, raising :class:`ConfigFileError` on failure."""
    try:
        with open(path, "r", encoding="utf-8") as handle:
            data = json.load(handle)
    except FileNotFoundError as exc:
        raise ConfigFileError(f"configuration file not found: {path}") from exc
    except json.JSONDecodeError as exc:
        raise ConfigFileError(f"invalid JSON in configuration file {path}: {exc}") from exc
    except OSError as exc:
        raise ConfigFileError(f"cannot read configuration file {path}: {exc}") from exc
    if not isinstance(data, dict):
        raise ConfigFileError(
            f"configuration file {path} must contain a JSON object at the top level"
        )
    return data


def _apply_env_overrides(
    config: dict[str, Any],
    *,
    prefix: str,
    separator: str,
    environ: Mapping[str, str],
) -> dict[str, Any]:
    """Apply matching environment variables as overrides onto *config*."""
    marker = f"{prefix}{separator}"
    for name, raw in environ.items():
        if not name.startswith(marker):
            continue
        suffix = name[len(marker):]
        path = [part.lower() for part in suffix.split(separator) if part]
        if path:
            _set_nested(config, path, _parse_env_value(raw))
    return config


class Config:
    """Immutable-ish wrapper around the resolved configuration mapping.

    Supports item access by top-level key (``config["server"]``) and dotted
    path lookups via :meth:`get` and :meth:`require`.
    """

    def __init__(self, data: Mapping[str, Any]) -> None:
        self._data: dict[str, Any] = deepcopy(dict(data))

    def get(self, dotted_key: str, default: Any = None) -> Any:
        """Return the value at *dotted_key* (e.g. ``"server.host"``) or *default*."""
        value = _lookup(self._data, dotted_key)
        return default if value is _MISSING else value

    def require(self, dotted_key: str) -> Any:
        """Return the value at *dotted_key*, raising if it is missing or None."""
        value = _lookup(self._data, dotted_key)
        if value is _MISSING or value is None:
            raise ConfigValidationError(
                f"missing required configuration key: {dotted_key!r}"
            )
        return value

    def as_dict(self) -> dict[str, Any]:
        """Return a deep copy of the underlying configuration."""
        return deepcopy(self._data)

    def __getitem__(self, key: str) -> Any:
        return self._data[key]

    def __contains__(self, key: str) -> bool:
        return key in self._data

    def __iter__(self):
        return iter(self._data)

    def __len__(self) -> int:
        return len(self._data)

    def __repr__(self) -> str:
        return f"Config({self._data!r})"


def load_config(
    path: Optional[str] = None,
    *,
    defaults: Optional[Mapping[str, Any]] = None,
    required: Iterable[str] = (),
    env_prefix: str = "APP",
    env_separator: str = "__",
    environ: Optional[Mapping[str, str]] = None,
) -> Config:
    """Load configuration from *path*, layered over *defaults*, with env overrides.

    Args:
        path: Path to a JSON file containing a top-level object. Optional.
        defaults: Mapping of default values used when no other source sets a key.
        required: Dotted key paths that must be present and not None after merging.
        env_prefix: Prefix for environment variable overrides.
        env_separator: Separator between the prefix and key path segments.
        environ: Environment mapping to read; defaults to ``os.environ``
            (injectable for testing).

    Returns:
        A :class:`Config` wrapping the merged result.

    Raises:
        ConfigFileError: If the file cannot be read or is not a JSON object.
        ConfigValidationError: If any required key is missing, listing them all.
    """
    config: dict[str, Any] = deepcopy(dict(defaults)) if defaults else {}

    if path is not None:
        config = _deep_merge(config, _read_json_file(path))

    config = _apply_env_overrides(
        config,
        prefix=env_prefix,
        separator=env_separator,
        environ=os.environ if environ is None else environ,
    )

    missing = []
    for key in required:
        value = _lookup(config, key)
        if value is _MISSING or value is None:
            missing.append(key)
    if missing:
        raise ConfigValidationError(
            "missing required configuration keys: " + ", ".join(sorted(missing))
        )

    return Config(config)
