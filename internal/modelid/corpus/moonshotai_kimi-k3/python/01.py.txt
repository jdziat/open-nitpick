"""A parser for a subset of CSV.

Supported features:
  * fields separated by a configurable single-character delimiter
    (comma by default);
  * fields optionally enclosed in double quotes;
  * escaped quotes: two double quotes ("") inside a quoted field stand
    for one literal double quote;
  * quoted fields may contain the delimiter and newline characters;
  * records are separated by LF, CR, or CRLF line endings.

The parser is strict: a quote may only appear at the start of a field,
and only a delimiter or a record separator may follow a closing quote.
Malformed input raises CSVError. An empty line yields a row containing
a single empty string.

parse_csv() returns a list of rows, where each row is a list of strings.
"""

__all__ = ["CSVError", "parse_csv", "load", "load_path"]


class CSVError(ValueError):
    """Raised when the input is not valid for this CSV subset."""


def parse_csv(text, delimiter=","):
    """Parse *text* and return a list of rows (lists of strings)."""
    if len(delimiter) != 1:
        raise ValueError("delimiter must be a single character")
    if delimiter in '"\r\n':
        raise ValueError("delimiter cannot be a quote or a newline")

    rows = []
    row = []
    field = []
    in_quotes = False
    after_quote = False

    def end_field():
        nonlocal field
        row.append("".join(field))
        field = []

    def end_row():
        nonlocal row
        end_field()
        rows.append(row)
        row = []

    i = 0
    n = len(text)
    while i < n:
        ch = text[i]

        if in_quotes:
            if ch == '"':
                if i + 1 < n and text[i + 1] == '"':
                    field.append('"')
                    i += 1  # consume the second quote as well
                else:
                    in_quotes = False
                    after_quote = True
            else:
                field.append(ch)
        elif after_quote:
            if ch == delimiter:
                end_field()
                after_quote = False
            elif ch in "\r\n":
                end_row()
                after_quote = False
                if ch == "\r" and i + 1 < n and text[i + 1] == "\n":
                    i += 1  # consume the LF of a CRLF pair
            else:
                raise CSVError(
                    "expected %r or a newline after closing quote, "
                    "got %r at offset %d" % (delimiter, ch, i)
                )
        elif ch == delimiter:
            end_field()
        elif ch in "\r\n":
            end_row()
            if ch == "\r" and i + 1 < n and text[i + 1] == "\n":
                i += 1  # consume the LF of a CRLF pair
        elif ch == '"':
            if field:
                raise CSVError(
                    "unexpected quote in unquoted field at offset %d" % i
                )
            in_quotes = True
        else:
            field.append(ch)

        i += 1

    if in_quotes:
        raise CSVError("unterminated quoted field at end of input")
    if after_quote or field or row:
        end_row()

    return rows


def load(fp, delimiter=","):
    """Read CSV text from a file-like object and parse it."""
    return parse_csv(fp.read(), delimiter=delimiter)


def load_path(path, delimiter=",", encoding="utf-8"):
    """Open the file at *path* as text and parse its contents."""
    with open(path, "r", encoding=encoding, newline="") as fp:
        return parse_csv(fp.read(), delimiter=delimiter)
