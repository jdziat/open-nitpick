"""Word-frequency counter.

Reads text, normalises case and punctuation, and reports the top K
most frequent words together with their counts.

Usage:
    python wordfreq.py [FILE] [-k N]
    cat book.txt | python wordfreq.py -k 20
"""

import argparse
import re
import sys
from collections import Counter

# Matches runs of letters/digits, allowing internal apostrophes
# (e.g. "don't", "rock’n’roll"). Underscores are treated as punctuation.
_WORD_RE = re.compile(r"[^\W_]+(?:['’][^\W_]+)?", re.UNICODE)


def tokenize(text):
    """Split *text* into normalised word tokens.

    Case is folded to lower-case and surrounding punctuation is
    discarded, so 'Hello,', 'HELLO' and 'hello!' all become 'hello'.
    """
    return _WORD_RE.findall(text.lower())


def count_words(text):
    """Return a Counter mapping each normalised word to its frequency."""
    return Counter(tokenize(text))


def top_k(text, k):
    """Return the *k* most frequent words in *text*.

    The result is a list of (word, count) tuples ordered by descending
    count; ties are broken alphabetically so output is deterministic.
    """
    if k <= 0:
        return []
    counts = count_words(text)
    return sorted(counts.items(), key=lambda item: (-item[1], item[0]))[:k]


def main(argv=None):
    parser = argparse.ArgumentParser(
        description="Count word frequencies and print the top K words."
    )
    parser.add_argument(
        "source",
        nargs="?",
        help="Text file to read (omit or pass '-' to read from stdin).",
    )
    parser.add_argument(
        "-k",
        "--top",
        type=int,
        default=10,
        help="Number of top words to report (default: 10).",
    )
    args = parser.parse_args(argv)

    if args.source and args.source != "-":
        with open(args.source, "r", encoding="utf-8") as handle:
            text = handle.read()
    else:
        text = sys.stdin.read()

    for word, count in top_k(text, args.top):
        print(f"{word}\t{count}")


if __name__ == "__main__":
    main()
