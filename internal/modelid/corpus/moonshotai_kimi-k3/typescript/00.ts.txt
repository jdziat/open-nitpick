/**
 * A fixed-capacity LRU (Least Recently Used) cache.
 *
 * `get` and `put` both run in O(1) time. The implementation relies on the
 * insertion-order guarantee of JavaScript's `Map`: the first key in the map
 * is always the least recently used entry.
 */
class LRUCache<K, V> {
  private readonly capacity: number;
  private readonly entries: Map<K, V>;

  constructor(capacity: number) {
    if (!Number.isInteger(capacity) || capacity <= 0) {
      throw new Error("LRUCache capacity must be a positive integer");
    }
    this.capacity = capacity;
    this.entries = new Map<K, V>();
  }

  /**
   * Returns the value stored under `key`, or `undefined` if the key is not
   * present. A successful lookup marks the entry as most recently used.
   */
  get(key: K): V | undefined {
    if (!this.entries.has(key)) {
      return undefined;
    }
    const value = this.entries.get(key) as V;
    // Re-insert so the key moves to the most-recently-used position.
    this.entries.delete(key);
    this.entries.set(key, value);
    return value;
  }

  /**
   * Inserts or updates the entry for `key`, marking it as most recently used.
   * If the cache is full, the least recently used entry is evicted first.
   */
  put(key: K, value: V): void {
    if (this.entries.has(key)) {
      this.entries.delete(key);
    } else if (this.entries.size >= this.capacity) {
      const lruKey = this.entries.keys().next().value as K;
      this.entries.delete(lruKey);
    }
    this.entries.set(key, value);
  }

  has(key: K): boolean {
    return this.entries.has(key);
  }

  delete(key: K): boolean {
    return this.entries.delete(key);
  }

  clear(): void {
    this.entries.clear();
  }

  get size(): number {
    return this.entries.size;
  }

  /** Current keys ordered from least to most recently used. */
  keys(): K[] {
    return Array.from(this.entries.keys());
  }
}

// --- Demo ---

const cache = new LRUCache<string, number>(3);

cache.put("a", 1);
cache.put("b", 2);
cache.put("c", 3);
console.log(cache.keys()); // [ "a", "b", "c" ]

cache.get("a"); // "a" is now the most recently used entry
cache.put("d", 4); // cache is full, so "b" (the LRU entry) is evicted

console.log(cache.keys()); // [ "c", "a", "d" ]
console.log(cache.get("b")); // undefined
console.log(cache.get("a")); // 1
console.log(cache.size); // 3
