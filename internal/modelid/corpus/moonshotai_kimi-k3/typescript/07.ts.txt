/**
 * config-loader.ts
 *
 * A small, typed loader for JSON configuration files featuring:
 *   - default values (deep-merged, lowest precedence)
 *   - environment-variable overrides (highest precedence)
 *   - validation of required keys
 *
 * Precedence: defaults < config file < environment variables.
 *
 * Environment variables map onto nested keys with "__" as the separator:
 *   prefix "APP"  ->  APP_SERVER__PORT=8080  overrides  config.server.port
 * Values are coerced: "true"/"false" -> boolean, numeric -> number,
 * "null" -> null, JSON objects/arrays are parsed, everything else stays
 * a string. Path segments match existing keys case-insensitively.
 */

import { readFileSync } from "node:fs";
import { resolve } from "node:path";

/* ------------------------------------------------------------------ types */

export type DeepPartial<T> = {
  [K in keyof T]?: T[K] extends readonly unknown[]
    ? T[K]
    : T[K] extends object
      ? DeepPartial<T[K]>
      : T[K];
};

/** All dotted key paths of T, e.g. "server" | "server.host" | "server.port". */
export type ConfigPath<T> = T extends readonly unknown[]
  ? never
  : T extends object
    ? {
        [K in keyof T & string]: T[K] extends readonly unknown[]
          ? K
          : T[K] extends object
            ? K | `${K}.${ConfigPath<T[K]>}`
            : K;
      }[keyof T & string]
    : never;

export class ConfigError extends Error {
  constructor(message: string) {
    super(message);
    this.name = "ConfigError";
  }
}
