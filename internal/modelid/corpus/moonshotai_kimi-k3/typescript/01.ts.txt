export interface CsvParseOptions {
  delimiter?: string;
}

export class CsvParseError extends Error {
  constructor(
    message: string,
    public readonly row: number,
    public readonly column: number
  ) {
    super(`${message} (row ${row}, column ${column})`);
    this.name = "CsvParseError";
  }
}

export function parseCsv(input: string, options: CsvParseOptions = {}): string[][] {
  const delimiter = options.delimiter ?? ",";
  if (delimiter.length !== 1) {
    throw new CsvParseError("Delimiter must be a single character", 1, 1);
  }
  if (delimiter === '"' || delimiter === "\r" || delimiter === "\n") {
    throw new CsvParseError("Delimiter cannot be a quote or newline", 1, 1);
  }

  const rows: string[][] = [];
  let row: string[] = [];
  let field = "";
  let inQuotes = false;
  let fieldStarted = false;
  let i = 0;
  let rowNumber = 1;
  let columnNumber = 1;

  const endField = (): void => {
    row.push(field);
    field = "";
    fieldStarted = false;
  };

  const endRow = (): void => {
    endField();
    rows.push(row);
    row = [];
    rowNumber += 1;
    columnNumber = 0;
  };

  while (i < input.length) {
    const ch = input[i];

    if (inQuotes) {
      if (ch === '"') {
        if (i + 1 < input.length && input[i + 1] === '"') {
          field += '"';
          i += 2;
          columnNumber += 2;
          continue;
        }
        inQuotes = false;
        i += 1;
        columnNumber += 1;
        continue;
      }
      field += ch;
      if (ch === "\n") {
        rowNumber += 1;
        columnNumber = 1;
      } else {
        columnNumber += 1;
      }
      i += 1;
      continue;
    }

    if (ch === '"' && !fieldStarted) {
      inQuotes = true;
      fieldStarted = true;
      i += 1;
      columnNumber += 1;
      continue;
    }

    if (ch === delimiter) {
      endField();
      i += 1;
      columnNumber += 1;
      continue;
    }

    if (ch === "\n") {
      endRow();
      i += 1;
      columnNumber = 1;
      continue;
    }

    if (ch === "\r") {
      if (i + 1 < input.length && input[i + 1] === "\n") {
        i += 1;
      }
      endRow();
      i += 1;
      columnNumber = 1;
      continue;
    }

    field += ch;
    fieldStarted = true;
    i += 1;
    columnNumber += 1;
  }

  if (inQuotes) {
    throw new CsvParseError("Unterminated quoted field", rowNumber, columnNumber);
  }

  if (field.length > 0 || fieldStarted || row.length > 0) {
    endRow();
  }

  return rows;
}

export default parseCsv;
