/**
 * retry — call a function up to `attempts` times with exponential backoff
 * and full jitter, returning either the value or the last error.
 *
 * Example:
 *   const result = await retry(fetchJson, { attempts: 5, baseDelayMs: 200 });
 *   if (result.ok) console.log(result.value);
 *   else console.error("failed after", result.attempts, "attempts:", result.error);
 */

export interface RetryOptions {
  /** Total number of times to invoke the function, including the first try. Default: 5. */
  attempts?: number;
  /** Delay before the first retry, in milliseconds. Default: 250. */
  baseDelayMs?: number;
  /** Upper bound for a single backoff delay, in milliseconds. Default: 10_000. */
  maxDelayMs?: number;
  /** Randomize each wait across the full range [0, computed delay). Default: true. */
  jitter?: boolean;
  /** Return false to stop retrying on a particular error. Default: retry on everything. */
  shouldRetry?: (error: unknown, attempt: number) => boolean;
  /** Invoked before each retry wait with the error, the upcoming attempt number, and the wait. */
  onRetry?: (error: unknown, attempt: number, delayMs: number) => void;
}

export type RetryResult<T> =
  | { readonly ok: true; readonly value: T; readonly attempts: number }
  | { readonly ok: false; readonly error: unknown; readonly attempts: number };

const sleep = (ms: number): Promise<void> =>
  new Promise((resolve) => setTimeout(resolve, ms));

function backoffDelay(
  attempt: number,
  baseDelayMs: number,
  maxDelayMs: number,
  jitter: boolean,
): number {
  const exponential = Math.min(maxDelayMs, baseDelayMs * 2 ** attempt);
  return jitter ? Math.random() * exponential : Math.max(0, exponential);
}

export async function retry<T>(
  fn: (attempt: number) => Promise<T> | T,
  options: RetryOptions = {},
): Promise<RetryResult<T>> {
  const {
    attempts = 5,
    baseDelayMs = 250,
    maxDelayMs = 10_000,
    jitter = true,
    shouldRetry = () => true,
    onRetry,
  } = options;

  const maxAttempts = Math.max(1, Math.floor(attempts));
  let lastError: unknown;
  let attempt = 0;

  for (; attempt < maxAttempts; attempt++) {
    try {
      const value = await fn(attempt);
      return { ok: true, value, attempts: attempt + 1 };
    } catch (error) {
      lastError = error;
      if (attempt === maxAttempts - 1 || !shouldRetry(error, attempt + 1)) break;
      const delayMs = backoffDelay(attempt, baseDelayMs, maxDelayMs, jitter);
      onRetry?.(error, attempt + 1, delayMs);
      await sleep(delayMs);
    }
  }

  return { ok: false, error: lastError, attempts: attempt + 1 };
}

/** Like `retry`, but throws the last error instead of returning a result object. */
export async function retryOrThrow<T>(
  fn: (attempt: number) => Promise<T> | T,
  options: RetryOptions = {},
): Promise<T> {
  const result = await retry(fn, options);
  if (!result.ok) throw result.error;
  return result.value;
}
