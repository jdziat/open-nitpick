#!/usr/bin/env node
/**
 * wordfreq.ts — a word-frequency counter.
 *
 * Reads text (from a file or stdin), normalises case and punctuation,
 * and prints the top K words with their counts.
 *
 * Usage:
 *   tsx wordfreq.ts [K] [file]
 *   cat file.txt | tsx wordfreq.ts [K]
 *
 * Can also be imported: topWords(text, k) returns WordCount[].
 */

import { readFileSync } from "node:fs";
import { pathToFileURL } from "node:url";

/** A word paired with the number of times it occurs. */
export interface WordCount {
  word: string;
  count: number;
}

/**
 * Normalise raw text: lowercase it and canonicalise curly quotes so
 * apostrophes/quotes compare equal across sources.
 */
export function normalizeText(text: string): string {
  return text
    .toLowerCase()
    .replace(/\u2018|\u2019/g, "'")
    .replace(/\u201c|\u201d/g, '"');
}

/**
 * A word is a run of Unicode letters/digits, optionally containing
 * internal apostrophes ("don't") or hyphens ("state-of-the-art").
 */
const WORD_RE = /[\p{L}\p{N}]+(?:['\u2019-][\p{L}\p{N}]+)*/gu;

/** Split text into normalised word tokens. */
export function tokenize(text: string): string[] {
  return normalizeText(text).match(WORD_RE) ?? [];
}

/** Count occurrences of each word. */
export function countWords(words: readonly string[]): Map<string, number> {
  const counts = new Map<string, number>();
  for (const word of words) {
    counts.set(word, (counts.get(word) ?? 0) + 1);
  }
  return counts;
}

/**
 * Return the top K most frequent words in `text`, most frequent first.
 * Ties are broken alphabetically so output is deterministic.
 */
export function topWords(text: string, k: number): WordCount[] {
  return [...countWords(tokenize(text)).entries()]
    .map(([word, count]) => ({ word, count }))
    .sort((a, b) => b.count - a.count || a.word.localeCompare(b.word))
    .slice(0, Math.max(0, k));
}

async function readAllStdin(): Promise<string> {
  const chunks: Buffer[] = [];
  for await (const chunk of process.stdin) {
    chunks.push(chunk as Buffer);
  }
  return Buffer.concat(chunks).toString("utf8");
}

async function main(argv: readonly string[]): Promise<void> {
  const usage =
    "usage: wordfreq [K] [file]   (K defaults to 10; reads stdin if no file given)";

  const k = argv[0] !== undefined ? Number(argv[0]) : 10;
  if (!Number.isInteger(k) || k < 1) {
    console.error(usage);
    process.exitCode = 1;
    return;
  }

  const file = argv[1];
  let text: string;
  try {
    text = file !== undefined ? readFileSync(file, "utf8") : await readAllStdin();
  } catch (err) {
    console.error(`wordfreq: cannot read ${file}: ${(err as Error).message}`);
    process.exitCode = 1;
    return;
  }

  for (const { word, count } of topWords(text, k)) {
    console.log(`${count}\t${word}`);
  }
}

// Only run the CLI when executed directly, so the functions can be imported.
const invokedDirectly =
  process.argv[1] !== undefined &&
  import.meta.url === pathToFileURL(process.argv[1]).href;

if (invokedDirectly) {
  main(process.argv.slice(2)).catch((err: unknown) => {
    console.error(err);
    process.exitCode = 1;
  });
}
