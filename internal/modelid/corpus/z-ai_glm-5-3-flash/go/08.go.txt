package main

import "fmt"

// PriorityQueue is a priority queue backed by a binary heap. Elements are
// ordered by the comparison function supplied at construction: less(a, b)
// reports whether a must be popped before b.
type PriorityQueue[T any] struct {
	heap []T
	less func(a, b T) bool
}

// New returns an empty priority queue ordered by less.
func New[T any](less func(a, b T) bool) *PriorityQueue[T] {
	return &PriorityQueue[T]{less: less}
}

// Push adds an element to the queue.
func (pq *PriorityQueue[T]) Push(v T) {
	pq.heap = append(pq.heap, v)
	pq.siftUp(len(pq.heap) - 1)
}

// Pop removes and returns the highest-priority element. The second result is
// false if the queue is empty.
func (pq *PriorityQueue[T]) Pop() (T, bool) {
	if len(pq.heap) == 0 {
		var zero T
		return zero, false
	}
	top := pq.heap[0]
	last := len(pq.heap) - 1
	pq.heap[0] = pq.heap[last]
	pq.heap = pq.heap[:last]
	if last > 0 {
		pq.siftDown(0)
	}
	return top, true
}

// Peek returns the highest-priority element without removing it. The second
// result is false if the queue is empty.
func (pq *PriorityQueue[T]) Peek() (T, bool) {
	if len(pq.heap) == 0 {
		var zero T
		return zero, false
	}
	return pq.heap[0], true
}

// Len reports the number of elements in the queue.
func (pq *PriorityQueue[T]) Len() int { return len(pq.heap) }

func (pq *PriorityQueue[T]) siftUp(i int) {
	for i > 0 {
		parent := (i - 1) / 2
		if !pq.less(pq.heap[i], pq.heap[parent]) {
			break
		}
		pq.heap[i], pq.heap[parent] = pq.heap[parent], pq.heap[i]
		i = parent
	}
}

func (pq *PriorityQueue[T]) siftDown(i int) {
	n := len(pq.heap)
	for {
		left := 2*i + 1
		if left >= n {
			return
		}
		child := left
		if right := left + 1; right < n && pq.less(pq.heap[right], pq.heap[left]) {
			child = right
		}
		if !pq.less(pq.heap[child], pq.heap[i]) {
			return
		}
		pq.heap[i], pq.heap[child] = pq.heap[child], pq.heap[i]
		i = child
	}
}

func main() {
	nums := New[int](func(a, b int) bool { return a > b }) // max-heap: larger pops first
	for _, n := range []int{5, 1, 9, 3, 7} {
		nums.Push(n)
	}
	fmt.Println("top:", func() int { n, _ := nums.Peek(); return n }())
	for nums.Len() > 0 {
		n, _ := nums.Pop()
		fmt.Println(n)
	}

	type task struct {
		name string
		prio int
	}
	jobs := New[task](func(a, b task) bool { return a.prio < b.prio })
	jobs.Push(task{"reply to email", 5})
	jobs.Push(task{"fix outage", 1})
	jobs.Push(task{"write report", 3})
	for jobs.Len() > 0 {
		j, _ := jobs.Pop()
		fmt.Printf("%s (priority %d)\n", j.name, j.prio)
	}
}
