package main

import (
	"errors"
	"fmt"
	"math"
)

// Matrix is a dense matrix of float64 values.
type Matrix struct {
	rows int
	cols int
	data [][]float64
}

// New returns a zero-initialized rows x cols matrix.
func New(rows, cols int) (*Matrix, error) {
	if rows <= 0 || cols <= 0 {
		return nil, errors.New("matrix: dimensions must be positive")
	}
	data := make([][]float64, rows)
	for i := range data {
		data[i] = make([]float64, cols)
	}
	return &Matrix{rows: rows, cols: cols, data: data}, nil
}

// NewFromRows returns a matrix built from row slices.
// All rows must have the same length.
func NewFromRows(rows [][]float64) (*Matrix, error) {
	if len(rows) == 0 {
		return nil, errors.New("matrix: at least one row required")
	}
	cols := len(rows[0])
	if cols == 0 {
		return nil, errors.New("matrix: at least one column required")
	}
	for i, row := range rows {
		if len(row) != cols {
			return nil, fmt.Errorf("matrix: row %d has %d columns, expected %d", i, len(row), cols)
		}
	}
	data := make([][]float64, len(rows))
	for i := range rows {
		data[i] = append([]float64(nil), rows[i]...)
	}
	return &Matrix{rows: len(rows), cols: cols, data: data}, nil
}

// Rows returns the number of rows.
func (m *Matrix) Rows() int { return m.rows }

// Cols returns the number of columns.
func (m *Matrix) Cols() int { return m.cols }

// Multiply returns m*n. It reports an error if the inner dimensions differ.
func (m *Matrix) Multiply(n *Matrix) (*Matrix, error) {
	if m == nil || n == nil {
		return nil, errors.New("matrix: nil operand")
	}
	if m.cols != n.rows {
		return nil, fmt.Errorf("matrix: cannot multiply %dx%d by %dx%d: inner dimensions differ (%d != %d)",
			m.rows, m.cols, n.rows, n.cols, m.cols, n.rows)
	}
	out, err := New(m.rows, n.cols)
	if err != nil {
		return nil, err
	}
	for i := 0; i < m.rows; i++ {
		for k := 0; k < m.cols; k++ {
			a := m.data[i][k]
			if a == 0 {
				continue
			}
			rowN := n.data[k]
			rowOut := out.data[i]
			for j := 0; j < n.cols; j++ {
				rowOut[j] += a * rowN[j]
			}
		}
	}
	return out, nil
}

// Transpose returns m^T.
func (m *Matrix) Transpose() (*Matrix, error) {
	if m == nil {
		return nil, errors.New("matrix: nil receiver")
	}
	out, err := New(m.cols, m.rows)
	if err != nil {
		return nil, err
	}
	for i := 0; i < m.rows; i++ {
		for j := 0; j < m.cols; j++ {
			out.data[j][i] = m.data[i][j]
		}
	}
	return out, nil
}

// Determinant returns the determinant of a square matrix, computed with
// Gaussian elimination and partial pivoting. It reports an error if the
// matrix is not square.
func (m *Matrix) Determinant() (float64, error) {
	if m == nil {
		return 0, errors.New("matrix: nil receiver")
	}
	if m.rows != m.cols {
		return 0, fmt.Errorf("matrix: determinant requires a square matrix, got %dx%d", m.rows, m.cols)
	}
	n := m.rows
	a := make([][]float64, n)
	for i := range a {
		a[i] = append([]float64(nil), m.data[i]...)
	}
	det := 1.0
	for col := 0; col < n; col++ {
		pivot := col
		for row := col + 1; row < n; row++ {
			if math.Abs(a[row][col]) > math.Abs(a[pivot][col]) {
				pivot = row
			}
		}
		if a[pivot][col] == 0 {
			return 0, nil // singular matrix
		}
		if pivot != col {
			a[col], a[pivot] = a[pivot], a[col]
			det = -det
		}
		det *= a[col][col]
		for row := col + 1; row < n; row++ {
			f := a[row][col] / a[col][col]
			if f == 0 {
				continue
			}
			for j := col; j < n; j++ {
				a[row][j] -= f * a[col][j]
			}
		}
	}
	return det, nil
}

// String renders the matrix row by row.
func (m *Matrix) String() string {
	if m == nil {
		return "<nil matrix>"
	}
	var buf []byte
	for _, row := range m.data {
		buf = append(buf, '[')
		for j, v := range row {
			if j > 0 {
				buf = append(buf, ' ')
			}
			buf = append(buf, fmt.Sprintf("%g", v)...)
		}
		buf = append(buf, "]\n"...)
	}
	return string(buf)
}

func main() {
	a, err := NewFromRows([][]float64{
		{1, 2, 3},
		{4, 5, 6},
	})
	if err != nil {
		fmt.Println("error:", err)
		return
	}
	b, err := NewFromRows([][]float64{
		{7, 8},
		{9, 10},
		{11, 12},
	})
	if err != nil {
		fmt.Println("error:", err)
		return
	}

	prod, err := a.Multiply(b)
	if err != nil {
		fmt.Println("error:", err)
		return
	}
	fmt.Print("A * B =\n", prod)

	tr, err := prod.Transpose()
	if err != nil {
		fmt.Println("error:", err)
		return
	}
	fmt.Print("(A * B)^T =\n", tr)

	det, err := prod.Determinant()
	if err != nil {
		fmt.Println("error:", err)
	} else {
		fmt.Println("det(A * B) =", det)
	}

	// Shape mismatch errors.
	if _, err := b.Multiply(a); err != nil {
		fmt.Println("multiply mismatch:", err)
	}
	rect, err := NewFromRows([][]float64{{1, 2, 3}, {4, 5, 6}})
	if err != nil {
		fmt.Println("error:", err)
		return
	}
	if _, err := rect.Determinant(); err != nil {
		fmt.Println("non-square determinant:", err)
	}
}
