package main

import (
	"fmt"
	"sync"
	"time"
)

// TokenBucket is a token-bucket rate limiter safe for concurrent use.
// It holds at most `capacity` tokens and refills at `rate` tokens per second.
type TokenBucket struct {
	mu       sync.Mutex
	tokens   float64
	capacity float64
	rate     float64 // tokens per second
	lastFill time.Time
}

// NewTokenBucket returns a bucket that starts full.
func NewTokenBucket(capacity, rate float64) *TokenBucket {
	return &TokenBucket{
		tokens:   capacity,
		capacity: capacity,
		rate:     rate,
		lastFill: time.Now(),
	}
}

// refill adds tokens accrued since the last fill, capping at capacity.
// Caller must hold tb.mu.
func (tb *TokenBucket) refill(now time.Time) {
	elapsed := now.Sub(tb.lastFill).Seconds()
	if elapsed <= 0 {
		return
	}
	tb.tokens += elapsed * tb.rate
	if tb.tokens > tb.capacity {
		tb.tokens = tb.capacity
	}
	tb.lastFill = now
}

// Allow reports whether a single token is available, consuming it if so.
func (tb *TokenBucket) Allow() bool {
	tb.mu.Lock()
	defer tb.mu.Unlock()
	tb.refill(time.Now())
	if tb.tokens >= 1 {
		tb.tokens--
		return true
	}
	return false
}

// AllowN reports whether n tokens are available, consuming them if so.
func (tb *TokenBucket) AllowN(n float64) bool {
	tb.mu.Lock()
	defer tb.mu.Unlock()
	tb.refill(time.Now())
	if tb.tokens >= n {
		tb.tokens -= n
		return true
	}
	return false
}

// Wait blocks until a token is available, polling at a small interval.
func (tb *TokenBucket) Wait() {
	const pollInterval = 5 * time.Millisecond
	for {
		if tb.Allow() {
			return
		}
		time.Sleep(pollInterval)
	}
}

// Tokens returns the current number of available tokens.
// The value is approximate if other goroutines are consuming tokens.
func (tb *TokenBucket) Tokens() float64 {
	tb.mu.Lock()
	defer tb.mu.Unlock()
	tb.refill(time.Now())
	return tb.tokens
}

func main() {
	// Bucket of 5 tokens refilling at 2 tokens per second.
	limiter := NewTokenBucket(5, 2)

	var wg sync.WaitGroup
	var mu sync.Mutex
	allowed := 0

	for i := 0; i < 20; i++ {
		wg.Add(1)
		go func() {
			defer wg.Done()
			if limiter.Allow() {
				mu.Lock()
				allowed++
				mu.Unlock()
			}
		}()
	}
	wg.Wait()
	fmt.Printf("allowed %d of 20 concurrent burst requests\n", allowed)

	start := time.Now()
	for i := 0; i < 3; i++ {
		limiter.Wait()
	}
	fmt.Printf("3 Wait() calls took %v (limited by refill rate)\n",
		time.Since(start).Round(10*time.Millisecond))
	fmt.Printf("tokens now: %.2f\n", limiter.Tokens())
}
