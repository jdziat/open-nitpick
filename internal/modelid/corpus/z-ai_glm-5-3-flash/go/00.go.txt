package main

import "fmt"

// entry is a node in the doubly linked list that tracks recency order.
type entry[K comparable, V any] struct {
	key   K
	value V
	prev  *entry[K, V]
	next  *entry[K, V]
}

// LRUCache is a fixed-capacity cache that evicts the least recently
// used entry when the capacity is exceeded.
type LRUCache[K comparable, V any] struct {
	capacity int
	items    map[K]*entry[K, V]
	head     *entry[K, V] // sentinel: most recently used side
	tail     *entry[K, V] // sentinel: least recently used side
}

// NewLRUCache creates a cache with the given fixed capacity.
func NewLRUCache[K comparable, V any](capacity int) *LRUCache[K, V] {
	if capacity < 1 {
		panic("lru: capacity must be positive")
	}
	h, t := &entry[K, V]{}, &entry[K, V]{}
	h.next, t.prev = t, h
	return &LRUCache[K, V]{
		capacity: capacity,
		items:    make(map[K]*entry[K, V], capacity),
		head:     h,
		tail:     t,
	}
}

// Len returns the number of entries currently in the cache.
func (c *LRUCache[K, V]) Len() int {
	return len(c.items)
}

// Get returns the value for key and marks it as most recently used.
func (c *LRUCache[K, V]) Get(key K) (V, bool) {
	e, ok := c.items[key]
	if !ok {
		var zero V
		return zero, false
	}
	c.moveToFront(e)
	return e.value, true
}

// Put inserts or updates key, marking it as most recently used, and
// evicts the least recently used entry if the cache is at capacity.
func (c *LRUCache[K, V]) Put(key K, value V) {
	if e, ok := c.items[key]; ok {
		e.value = value
		c.moveToFront(e)
		return
	}
	if len(c.items) == c.capacity {
		lru := c.tail.prev
		c.remove(lru)
		delete(c.items, lru.key)
	}
	e := &entry[K, V]{key: key, value: value}
	c.items[key] = e
	c.addToFront(e)
}

func (c *LRUCache[K, V]) addToFront(e *entry[K, V]) {
	e.prev = c.head
	e.next = c.head.next
	c.head.next.prev = e
	c.head.next = e
}

func (c *LRUCache[K, V]) remove(e *entry[K, V]) {
	e.prev.next = e.next
	e.next.prev = e.prev
	e.prev, e.next = nil, nil
}

func (c *LRUCache[K, V]) moveToFront(e *entry[K, V]) {
	c.remove(e)
	c.addToFront(e)
}

func main() {
	cache := NewLRUCache[int, string](3)

	cache.Put(1, "one")
	cache.Put(2, "two")
	cache.Put(3, "three")

	if v, ok := cache.Get(1); ok {
		fmt.Println("get(1):", v) // promotes key 1
	}

	cache.Put(4, "four") // evicts key 2, the least recently used

	if _, ok := cache.Get(2); !ok {
		fmt.Println("key 2 evicted")
	}
	if v, ok := cache.Get(4); ok {
		fmt.Println("get(4):", v)
	}
	fmt.Println("size:", cache.Len())
}
