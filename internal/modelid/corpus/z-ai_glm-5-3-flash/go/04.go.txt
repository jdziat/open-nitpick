package main

import (
	"fmt"
	"strings"
	"unicode"
)

// Slugify lowercases text, replaces each run of non-alphanumeric
// characters with a single hyphen, and trims leading and trailing
// hyphens. Unicode letters and digits are preserved.
func Slugify(text string) string {
	var b strings.Builder
	b.Grow(len(text))

	prevHyphen := true // treat the start of the string like a preceding hyphen

	for _, r := range text {
		if unicode.IsLetter(r) || unicode.IsDigit(r) {
			b.WriteRune(unicode.ToLower(r))
			prevHyphen = false
		} else if !prevHyphen {
			b.WriteByte('-')
			prevHyphen = true
		}
		// Skip remaining runes in a run of separators (and leading ones).
	}

	return strings.TrimSuffix(b.String(), "-")
}

func main() {
	examples := []string{
		"Hello, World!",
		"  Multiple   spaces & symbols!!!  ",
		"Héllo, Wörld — Ünïcode 123",
		"日本語のテキスト too",
		"---already--slugified---",
		"!!!",
		"",
	}

	for _, s := range examples {
		fmt.Printf("%q -> %q\n", s, Slugify(s))
	}
}
