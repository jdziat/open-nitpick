"""date_range.py

A date-range iterator that yields each day between two dates, inclusive,
with an option to skip weekends.

Usage:
    from date_range import date_range, DateRange

    for day in date_range(date(2024, 1, 1), date(2024, 1, 7)):
        print(day)

    for day in date_range(date(2024, 1, 1), date(2024, 1, 7), skip_weekends=True):
        print(day)  # Mondays through Fridays only
"""

from __future__ import annotations

import datetime as _dt
from typing import Iterator, Union

DateLike = Union[_dt.date, _dt.datetime, str]

_WEEKDAY_SATURDAY = 5


def _coerce_to_date(value: DateLike) -> _dt.date:
    """Normalize *value* to a datetime.date.

    Accepts a datetime.date, a datetime.datetime (its date part is used),
    or an ISO-8601 string such as "2024-01-15".
    """
    if isinstance(value, _dt.datetime):
        return value.date()
    if isinstance(value, _dt.date):
        return value
    if isinstance(value, str):
        return _dt.date.fromisoformat(value)
    raise TypeError(
        f"expected date, datetime, or ISO date string, got {type(value).__name__}"
    )


def date_range(
    start: DateLike,
    end: DateLike,
    *,
    skip_weekends: bool = False,
) -> Iterator[_dt.date]:
    """Yield each day from *start* through *end*, inclusive.

    Args:
        start: First day of the range (date, datetime, or ISO string).
        end: Last day of the range (inclusive).
        skip_weekends: When True, Saturdays and Sundays are omitted.

    Yields:
        datetime.date objects in ascending order.

    Raises:
        TypeError: If *start* or *end* is not a recognized date type.
        ValueError: If a date string is malformed or *start* is after *end*.
    """
    start = _coerce_to_date(start)
    end = _coerce_to_date(end)
    if start > end:
        raise ValueError(f"start ({start}) must not be after end ({end})")

    current = start
    one_day = _dt.timedelta(days=1)
    while current <= end:
        if not skip_weekends or current.weekday() < _WEEKDAY_SATURDAY:
            yield current
        current += one_day


class DateRange:
    """A reusable, iterable representation of an inclusive date range.

    Each call to iter() returns a fresh iterator, so the same DateRange
    object can be traversed multiple times. Also supports len() and
    membership tests with `in`.
    """

    __slots__ = ("_start", "_end", "_skip_weekends")

    def __init__(
        self,
        start: DateLike,
        end: DateLike,
        *,
        skip_weekends: bool = False,
    ) -> None:
        self._start = _coerce_to_date(start)
        self._end = _coerce_to_date(end)
        self._skip_weekends = bool(skip_weekends)
        if self._start > self._end:
            raise ValueError(
                f"start ({self._start}) must not be after end ({self._end})"
            )

    @property
    def start(self) -> _dt.date:
        return self._start

    @property
    def end(self) -> _dt.date:
        return self._end

    @property
    def skip_weekends(self) -> bool:
        return self._skip_weekends

    def __iter__(self) -> Iterator[_dt.date]:
        return date_range(self._start, self._end, skip_weekends=self._skip_weekends)

    def __len__(self) -> int:
        total_days = (self._end - self._start).days + 1
        if not self._skip_weekends:
            return total_days
        whole_weeks, extra_days = divmod(total_days, 7)
        weekend_days = whole_weeks * 2
        first_extra = self._start + _dt.timedelta(days=whole_weeks * 7)
        for offset in range(extra_days):
            if (first_extra + _dt.timedelta(days=offset)).weekday() >= _WEEKDAY_SATURDAY:
                weekend_days += 1
        return total_days - weekend_days

    def __contains__(self, item: DateLike) -> bool:
        try:
            day = _coerce_to_date(item)
        except (TypeError, ValueError):
            return False
        if not (self._start <= day <= self._end):
            return False
        return not (self._skip_weekends and day.weekday() >= _WEEKDAY_SATURDAY)

    def __repr__(self) -> str:
        flag = ", skip_weekends=True" if self._skip_weekends else ""
        return (
            f"DateRange({self._start.isoformat()}, {self._end.isoformat()}{flag})"
        )


if __name__ == "__main__":
    demo_start = _dt.date.today() - _dt.timedelta(days=13)
    demo_end = _dt.date.today()

    print(f"All days from {demo_start} to {demo_end}:")
    for day in date_range(demo_start, demo_end):
        print(" ", day.strftime("%a %Y-%m-%d"))

    weekdays = DateRange(demo_start, demo_end, skip_weekends=True)
    print(
        f"\nWeekdays only from {demo_start} to {demo_end} "
        f"({len(weekdays)} days):"
    )
    for day in weekdays:
        print(" ", day.strftime("%a %Y-%m-%d"))
