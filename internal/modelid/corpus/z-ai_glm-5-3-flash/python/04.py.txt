"""slugify — convert arbitrary text into URL-friendly slugs."""

import re
import unicodedata

# Runs of characters that are not Unicode letters or digits.  Underscore is
# treated as a separator too, since it is not alphanumeric.
_SEPARATORS = re.compile(r"[\W_]+", re.UNICODE)


def slugify(text):
    """Return a slug for *text*.

    The text is lowercased, runs of non-alphanumeric characters are replaced
    with a single hyphen, and leading/trailing hyphens are trimmed.  Unicode
    letters and digits are preserved.

        slugify("  Hello, World!  ")       -> "hello-world"
        slugify("Crème brûlée & café")     -> "crème-brûlée-café"
        slugify("multiple___under__scores") -> "multiple-under-scores"
        slugify("日本語 テスト")             -> "日本語-テスト"
    """
    text = unicodedata.normalize("NFKC", str(text))
    text = text.lower()
    return _SEPARATORS.sub("-", text).strip("-")


if __name__ == "__main__":
    examples = [
        ("  Hello, World!  ", "hello-world"),
        ("Crème brûlée & café", "crème-brûlée-café"),
        ("multiple___under__scores", "multiple-under-scores"),
        ("--already--slugged--", "already-slugged"),
        ("日本語 テスト", "日本語-テスト"),
        ("", ""),
        ("!!!", ""),
    ]
    for source, expected in examples:
        result = slugify(source)
        assert result == expected, f"{source!r}: got {result!r}, want {expected!r}"
        print(f"{source!r:35} -> {result!r}")
    print("all examples passed")
