#!/usr/bin/env python3
"""wordfreq.py — a simple word-frequency counter.

Reads text (from a file or stdin), normalises case and punctuation, and
prints/returns the top K words with their counts.

Usage:
    python wordfreq.py [-k K] [file]

If no file is given (or the file is "-"), text is read from stdin.

Example:
    $ echo "The cat, the dog. The CAT!" | python wordfreq.py -k 2
           3  the
           2  cat
"""

from __future__ import annotations

import argparse
import re
import sys
from collections import Counter
from typing import Iterable, List, Optional, Tuple

# A "word" is a run of unicode letters/digits, optionally containing
# apostrophes inside it (so "don't" stays one word). Hyphens and all other
# punctuation act as separators ("well-known" -> "well", "known").
_TOKEN_RE = re.compile(r"[^\W_]+(?:'[^\W_]+)*", re.UNICODE)


def normalise(text: str) -> str:
    """Return *text* lower-cased (remaining punctuation is handled when tokenising)."""
    return text.lower()


def tokenize(text: str) -> List[str]:
    """Split *text* into normalised word tokens.

    Case is folded to lower-case, punctuation is treated as a separator,
    and apostrophes are kept only when they appear inside a word
    ("don't" -> "don't", "dogs'" -> "dogs", "'tis" -> "tis").
    """
    tokens = []
    for raw in _TOKEN_RE.findall(normalise(text)):
        token = raw.strip("'")
        if token:
            tokens.append(token)
    return tokens


def count_words(text: str) -> Counter:
    """Return a Counter mapping each normalised word to its frequency."""
    return Counter(tokenize(text))


def top_words(text: str, k: int = 10) -> List[Tuple[str, int]]:
    """Return the top *k* (word, count) pairs found in *text*.

    Ordering: by count (descending), then alphabetically for ties, so the
    result is deterministic. *k* is clamped to be non-negative.
    """
    k = max(k, 0)
    counts = count_words(text)
    return sorted(counts.items(), key=lambda item: (-item[1], item[0]))[:k]


def read_text(path: Optional[str]) -> str:
    """Read all text from *path*, or from stdin if *path* is None or '-'."""
    if path is None or path == "-":
        return sys.stdin.read()
    with open(path, "r", encoding="utf-8", errors="replace") as handle:
        return handle.read()


def main(argv: Optional[Iterable[str]] = None) -> int:
    parser = argparse.ArgumentParser(
        description="Count word frequencies and print the top K words."
    )
    parser.add_argument(
        "file",
        nargs="?",
        default=None,
        help="text file to read (default: stdin; '-' also means stdin)",
    )
    parser.add_argument(
        "-k",
        metavar="K",
        type=int,
        default=10,
        help="number of top words to show (default: 10)",
    )
    args = parser.parse_args(list(argv) if argv is not None else None)

    text = read_text(args.file)
    for word, count in top_words(text, args.k):
        print(f"{count:>8}  {word}")
    return 0


if __name__ == "__main__":
    sys.exit(main())
