class _Node:
    __slots__ = ("key", "value", "prev", "next")

    def __init__(self, key=None, value=None):
        self.key = key
        self.value = value
        self.prev = None
        self.next = None


class LRUCache:
    """A fixed-capacity LRU cache with O(1) get and put.

    When the cache is full, put() evicts the least recently used entry.
    Both get() and put() count as "use" and refresh an entry's recency.
    """

    def __init__(self, capacity):
        if capacity <= 0:
            raise ValueError("capacity must be a positive integer")
        self.capacity = capacity
        self._map = {}          # key -> _Node
        self._head = _Node()    # sentinel: most recently used side
        self._tail = _Node()    # sentinel: least recently used side
        self._head.next = self._tail
        self._tail.prev = self._head

    # --- internal doubly linked list helpers ---

    def _unlink(self, node):
        node.prev.next = node.next
        node.next.prev = node.prev
        node.prev = node.next = None

    def _push_front(self, node):
        node.next = self._head.next
        node.prev = self._head
        self._head.next.prev = node
        self._head.next = node

    def _pop_back(self):
        lru = self._tail.prev
        if lru is self._head:
            return None
        self._unlink(lru)
        return lru

    # --- public API ---

    def get(self, key, default=None):
        """Return the value for key, or default if absent. Marks key as used."""
        node = self._map.get(key)
        if node is None:
            return default
        self._unlink(node)
        self._push_front(node)
        return node.value

    def put(self, key, value):
        """Insert or update key. Evicts the least recently used entry if full."""
        node = self._map.get(key)
        if node is not None:
            node.value = value
            self._unlink(node)
            self._push_front(node)
            return
        if len(self._map) >= self.capacity:
            lru = self._pop_back()
            if lru is not None:
                del self._map[lru.key]
        node = _Node(key, value)
        self._map[key] = node
        self._push_front(node)

    # --- conveniences ---

    def __len__(self):
        return len(self._map)

    def __contains__(self, key):
        return key in self._map

    def keys(self):
        """Keys ordered from most to least recently used."""
        out = []
        node = self._head.next
        while node is not self._tail:
            out.append(node.key)
            node = node.next
        return out

    def __repr__(self):
        items = []
        node = self._head.next
        while node is not self._tail:
            items.append(f"{node.key!r}: {node.value!r}")
            node = node.next
        return "{" + ", ".join(items) + "}"


if __name__ == "__main__":
    cache = LRUCache(2)
    cache.put(1, "one")
    cache.put(2, "two")
    assert cache.get(1) == "one"      # 1 is now most recently used
    cache.put(3, "three")             # evicts key 2 (least recently used)
    assert cache.get(2) is None
    cache.put(1, "uno")               # update existing key, no eviction
    assert len(cache) == 2
    assert cache.get(1) == "uno"
    assert cache.keys() == [1, 3]
    print(cache)
    print("all assertions passed")
