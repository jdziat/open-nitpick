/**
 * A parser for a practical subset of CSV (RFC 4180 style):
 *
 * - Fields may be enclosed in double quotes (`"..."`).
 * - Inside a quoted field, `""` denotes a literal double quote.
 * - The field delimiter is configurable and must be a single character
 *   other than `"` or a newline (default: `,`).
 * - Records are terminated by `\r\n`, `\n`, or `\r`.
 * - Delimiters, escaped quotes, and newlines may appear inside quoted
 *   fields as data.
 * - A trailing record terminator at the end of the input does not
 *   produce an extra empty record; empty input produces no records.
 *
 * The parser is intentionally lenient about some malformed inputs
 * (e.g., an unterminated quoted field, or stray quotes inside unquoted
 * fields): it does its best rather than throwing.
 */

export interface CsvParseOptions {
  /**
   * Single-character field delimiter. Must not be `"` or a newline.
   * Defaults to `","`.
   */
  delimiter?: string;
}

/**
