#!/usr/bin/env node
/**
 * Word-frequency counter.
 *
 * Library use:
 *   import { topWords } from "./wordfreq";
 *   topWords("The the THE cat.", 2); // [{ word: "the", count: 3 }, { word: "cat", count: 1 }]
 *
 * CLI use (reads stdin when no files are given, "-" means stdin):
 *   cat article.txt | npx ts-node wordfreq.ts 10
 *   npx ts-node wordfreq.ts 10 article.txt
 */

import { readFileSync } from "node:fs";

export interface WordCount {
  word: string;
  count: number;
}

/**
 * Normalise text: lowercase, fold curly apostrophes to straight ones, turn
 * every run of non-letter/non-digit characters into a space (so punctuation
 * and hyphens split words), then drop stray apostrophes at word edges.
 */
export function tokenise(text: string): string[] {
  return text
    .toLowerCase()
    .replace(/[\u2018\u2019]/g, "'")
    .replace(/[^\p{L}\p{N}']+/gu, " ")
    .split(/\s+/)
    .map((w) => w.replace(/^'+|'+$/g, ""))
    .filter((w) => w.length > 0);
}

/** Count occurrences of every normalised word. */
export function countWords(text: string): Map<string, number> {
  const counts = new Map<string, number>();
  for (const word of tokenise(text)) {
    counts.set(word, (counts.get(word) ?? 0) + 1);
  }
  return counts;
}

/** Top K words by count; ties are broken alphabetically. */
export function topWords(text: string, k: number): WordCount[] {
  const limit = Number.isFinite(k) ? Math.max(0, Math.trunc(k)) : 0;
  return Array.from(countWords(text), ([word, count]) => ({ word, count }))
    .sort((a, b) => b.count - a.count || a.word.localeCompare(b.word))
    .slice(0, limit);
}

function readInput(paths: string[]): string {
  const chunks = paths.length > 0 ? paths : ["-"];
  return chunks.map((p) => readFileSync(p === "-" ? 0 : p, "utf8")).join("\n");
}

function main(argv: string[]): void {
  const args = argv.slice(2);
  const kIsPositional = args.length > 0 && /^\d+$/.test(args[0]);
  const k = kIsPositional ? Number.parseInt(args[0], 10) : 10;
  const files = kIsPositional ? args.slice(1) : args;

  try {
    for (const { word, count } of topWords(readInput(files), k)) {
      console.log(`${String(count).padStart(8)}  ${word}`);
    }
  } catch (err) {
    console.error((err as Error).message);
    process.exitCode = 1;
  }
}

if (typeof require !== "undefined" && require.main === module) {
  main(process.argv);
}
