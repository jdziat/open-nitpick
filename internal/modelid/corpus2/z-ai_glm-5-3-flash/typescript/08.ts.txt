/**
 * PriorityQueue<T>
 *
 * A priority queue backed by a binary heap. Ordering is defined by a
 * caller-supplied comparison function using the same convention as
 * Array.prototype.sort:
 *
 *   compare(a, b) <  0  ->  a is dequeued before b
 *   compare(a, b) >  0  ->  b is dequeued before a
 *   compare(a, b) === 0 ->  a and b are equivalent (order unspecified)
 *
 * Example:
 *   const q = new PriorityQueue<number>((a, b) => a - b); // min-heap
 *   q.push(3);
 *   q.push(1);
 *   q.peek(); // 1
 *   q.pop();  // 1
 *
 * push and pop are O(log n); peek is O(1).
 */
export class PriorityQueue<T> {
  private heap: T[] = [];

  constructor(private readonly compare: (a: T, b: T) => number) {}

  /** Number of elements currently in the queue. */
  get size(): number {
    return this.heap.length;
  }

  /** True when the queue contains no elements. */
  isEmpty(): boolean {
    return this.heap.length === 0;
  }

  /** Highest-priority element without removing it, or undefined if empty. */
  peek(): T | undefined {
    return this.heap[0];
  }

  /** Add an element to the queue. O(log n). */
  push(item: T): void {
    this.heap.push(item);
    this.siftUp(this.heap.length - 1);
  }

  /** Remove and return the highest-priority element, or undefined if empty. */
  pop(): T | undefined {
    if (this.heap.length === 0) return undefined;

    const top = this.heap[0];
    const last = this.heap.pop()!;
    if (this.heap.length > 0) {
      this.heap[0] = last;
      this.siftDown(0);
    }
    return top;
  }

  private siftUp(index: number): void {
    const item = this.heap[index];
    while (index > 0) {
      const parent = (index - 1) >> 1;
      if (this.compare(this.heap[parent], item) <= 0) break;
      this.heap[index] = this.heap[parent];
      index = parent;
    }
    this.heap[index] = item;
  }

  private siftDown(index: number): void {
    const n = this.heap.length;
    const item = this.heap[index];
    for (;;) {
      const left = 2 * index + 1;
      if (left >= n) break;
      const right = left + 1;
      const child =
        right < n && this.compare(this.heap[right], this.heap[left]) < 0
          ? right
          : left;
      if (this.compare(this.heap[child], item) >= 0) break;
      this.heap[index] = this.heap[child];
      index = child;
    }
    this.heap[index] = item;
  }
}
