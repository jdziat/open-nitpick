interface LRUNode<K, V> {
  key: K;
  value: V;
  prev: LRUNode<K, V>;
  next: LRUNode<K, V>;
}

function createNode<K, V>(key: K, value: V): LRUNode<K, V> {
  return { key, value, prev: undefined!, next: undefined! };
}

export class LRUCache<K, V> {
  private readonly capacity: number;
  private readonly map = new Map<K, LRUNode<K, V>>();

  // Sentinel nodes: head.next is most recently used, tail.prev is least recently used.
  private readonly head: LRUNode<K, V>;
  private readonly tail: LRUNode<K, V>;

  constructor(capacity: number) {
    if (!Number.isInteger(capacity) || capacity < 1) {
      throw new RangeError(`capacity must be a positive integer, got ${capacity}`);
    }
    this.capacity = capacity;
    this.head = createNode<K, V>(undefined as unknown as K, undefined as unknown as V);
    this.tail = createNode<K, V>(undefined as unknown as K, undefined as unknown as V);
    this.head.next = this.tail;
    this.tail.prev = this.head;
  }

  get size(): number {
    return this.map.size;
  }

  has(key: K): boolean {
    return this.map.has(key);
  }

  get(key: K): V | undefined {
    const node = this.map.get(key);
    if (node === undefined) {
      return undefined;
    }
    this.moveToFront(node);
    return node.value;
  }

  put(key: K, value: V): void {
    const existing = this.map.get(key);
    if (existing !== undefined) {
      existing.value = value;
      this.moveToFront(existing);
      return;
    }

    if (this.map.size >= this.capacity) {
      this.evictLRU();
    }

    const node = createNode(key, value);
    this.map.set(key, node);
    this.insertAfter(this.head, node);
  }

  peek(key: K): V | undefined {
    const node = this.map.get(key);
    return node === undefined ? undefined : node.value;
  }

  delete(key: K): boolean {
    const node = this.map.get(key);
    if (node === undefined) {
      return false;
    }
    this.unlink(node);
    this.map.delete(key);
    return true;
  }

  clear(): void {
    this.map.clear();
    this.head.next = this.tail;
    this.tail.prev = this.head;
  }

  keys(): K[] {
    const result: K[] = [];
    for (let node = this.head.next; node !== this.tail; node = node.next) {
      result.push(node.key);
    }
    return result;
  }

  private insertAfter(target: LRUNode<K, V>, node: LRUNode<K, V>): void {
    node.prev = target;
    node.next = target.next;
    target.next.prev = node;
    target.next = node;
  }

  private unlink(node: LRUNode<K, V>): void {
    node.prev.next = node.next;
    node.next.prev = node.prev;
    node.prev = node;
    node.next = node;
  }

  private moveToFront(node: LRUNode<K, V>): void {
    if (this.head.next === node) {
      return;
    }
    this.unlink(node);
    this.insertAfter(this.head, node);
  }

  private evictLRU(): void {
    const lru = this.tail.prev;
    if (lru === this.head) {
      return;
    }
    this.unlink(lru);
    this.map.delete(lru.key);
  }
}
