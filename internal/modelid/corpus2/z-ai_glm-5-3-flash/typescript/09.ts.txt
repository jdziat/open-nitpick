/**
 * dateRange.ts
 *
 * A date-range iterator that yields each calendar day between two dates,
 * inclusive of both endpoints, with an option to skip weekends.
 *
 * Example:
 *
 *   for (const day of dateRange("2024-01-27", "2024-02-02", { skipWeekends: true })) {
 *     console.log(day.toDateString());
 *   }
 *   // Mon Jan 29 2024
 *   // Tue Jan 30 2024
 *   // Wed Jan 31 2024
 *   // Thu Feb 01 2024
 *   // Fri Feb 02 2024
 */

export type DateInput = Date | string | number;

export interface DateRangeOptions {
  /** When true, Saturdays and Sundays are omitted from the iteration. Default: false. */
  skipWeekends?: boolean;
}

/**
 * Yields each day from `start` through `end`, inclusive, in ascending order.
 *
 * @param start   First day of the range (any value accepted by the Date constructor).
 * @param end     Last day of the range (inclusive).
 * @param options Optional behaviour flags, e.g. `{ skipWeekends: true }`.
 * @throws {TypeError}  If either input cannot be interpreted as a valid date.
 * @throws {RangeError} If `start` falls after `end`.
 */
export function* dateRange(
  start: DateInput,
  end: DateInput,
  options: DateRangeOptions = {},
): Generator<Date, void, undefined> {
  const first = startOfDay(toValidDate(start));
  const last = startOfDay(toValidDate(end));

  if (first.getTime() > last.getTime()) {
    throw new RangeError(
      `Start date (${first.toDateString()}) must not be after end date (${last.toDateString()}).`,
    );
  }

  const current = first;
  while (current.getTime() <= last.getTime()) {
    if (!options.skipWeekends || !isWeekend(current)) {
      yield new Date(current.getTime());
    }
    current.setDate(current.getDate() + 1); // Handles month/year boundaries and DST correctly.
  }
}

/**
 * An iterable, class-based wrapper around {@link dateRange}.
 *
 * Example:
 *
 *   const range = new DateRange(new Date(2024, 0, 1), new Date(2024, 0, 7), { skipWeekends: true });
 *   console.log(range.count());   // 5
 *   console.log(range.toArray()); // [Mon Jan 01 2024, ..., Fri Jan 05 2024]
 */
export class DateRange implements Iterable<Date> {
  readonly start: Date;
  readonly end: Date;
  readonly options: Readonly<Required<DateRangeOptions>>;

  constructor(start: DateInput, end: DateInput, options: DateRangeOptions = {}) {
    this.start = toValidDate(start);
    this.end = toValidDate(end);
    this.options = { skipWeekends: options.skipWeekends ?? false };

    if (startOfDay(this.start).getTime() > startOfDay(this.end).getTime()) {
      throw new RangeError(
        `Start date (${this.start.toDateString()}) must not be after end date (${this.end.toDateString()}).`,
      );
    }
  }

  [Symbol.iterator](): Generator<Date, void, undefined> {
    return dateRange(this.start, this.end, this.options);
  }

  /** Materializes the range into an array of Date objects. */
  toArray(): Date[] {
    return Array.from(this);
  }

  /** Number of days the iteration will yield. */
  count(): number {
    const iterator = this[Symbol.iterator]();
    let n = 0;
    while (!iterator.next().done) n += 1;
    return n;
  }
}

/** True if the date falls on a Saturday or Sunday. */
export function isWeekend(date: Date): boolean {
  const day = date.getDay(); // 0 = Sunday, 6 = Saturday
  return day === 0 || day === 6;
}

/** Returns a copy of `date` with the time components zeroed out. */
function startOfDay(date: Date): Date {
  const copy = new Date(date.getTime());
  copy.setHours(0, 0, 0, 0);
  return copy;
}

/** Parses any accepted input into a Date, throwing on invalid values. */
function toValidDate(input: DateInput): Date {
  const date = input instanceof Date ? new Date(input.getTime()) : new Date(input);
  if (Number.isNaN(date.getTime())) {
    throw new TypeError(`Invalid date input: ${String(input)}`);
  }
  return date;
}
