export type Matrix = number[][];

export interface Shape {
  readonly rows: number;
  readonly cols: number;
}

export class MatrixError extends Error {
  constructor(message: string) {
    super(message);
    this.name = "MatrixError";
  }
}

function validate(m: Matrix): void {
  if (!Array.isArray(m)) {
    throw new MatrixError("Matrix must be an array of rows.");
  }
  for (let i = 0; i < m.length; i++) {
    if (!Array.isArray(m[i])) {
      throw new MatrixError(`Row ${i} is not an array.`);
    }
    if (i > 0 && m[i].length !== m[0].length) {
      throw new MatrixError(
        `Ragged matrix: row ${i} has ${m[i].length} columns, expected ${m[0].length}.`
      );
    }
  }
}

function assertValid(m: Matrix, label: string): Shape {
  validate(m);
  const rows = m.length;
  const cols = rows > 0 ? m[0].length : 0;
  if (rows === 0 || cols === 0) {
    throw new MatrixError(`${label}: matrix must have at least one row and one column.`);
  }
  return { rows, cols };
}

export function shape(m: Matrix): Shape {
  validate(m);
  return { rows: m.length, cols: m.length > 0 ? m[0].length : 0 };
}

export function multiply(a: Matrix, b: Matrix): Matrix {
  const sa = assertValid(a, "multiply left operand");
  const sb = assertValid(b, "multiply right operand");
  if (sa.cols !== sb.rows) {
    throw new MatrixError(
      `Cannot multiply: left matrix is ${sa.rows}x${sa.cols}, right matrix is ${sb.rows}x${sb.cols}; ` +
        `inner dimensions (${sa.cols} and ${sb.rows}) must match.`
    );
  }
  const result: Matrix = [];
  for (let i = 0; i < sa.rows; i++) {
    const row: number[] = [];
    for (let j = 0; j < sb.cols; j++) {
      let sum = 0;
      for (let k = 0; k < sa.cols; k++) {
        sum += a[i][k] * b[k][j];
      }
      row.push(sum);
    }
    result.push(row);
  }
  return result;
}

export function transpose(m: Matrix): Matrix {
  const { rows, cols } = assertValid(m, "transpose");
  const result: Matrix = [];
  for (let j = 0; j < cols; j++) {
    const row: number[] = [];
    for (let i = 0; i < rows; i++) {
      row.push(m[i][j]);
    }
    result.push(row);
  }
  return result;
}

export function determinant(m: Matrix): number {
  const s = assertValid(m, "determinant");
  if (s.rows !== s.cols) {
    throw new MatrixError(
      `Determinant requires a square matrix, got ${s.rows}x${s.cols}.`
    );
  }
  const n = s.rows;
  if (n === 1) return m[0][0];

  // Gaussian elimination with partial pivoting.
  const a: number[][] = m.map((row) => row.slice());
  let det = 1;

  for (let col = 0; col < n; col++) {
    let pivot = col;
    for (let row = col + 1; row < n; row++) {
      if (Math.abs(a[row][col]) > Math.abs(a[pivot][col])) {
        pivot = row;
      }
    }
    if (a[pivot][col] === 0) {
      return 0;
    }
    if (pivot !== col) {
      const tmp = a[pivot];
      a[pivot] = a[col];
      a[col] = tmp;
      det = -det;
    }
    det *= a[col][col];
    for (let row = col + 1; row < n; row++) {
      const factor = a[row][col] / a[col][col];
      for (let j = col; j < n; j++) {
        a[row][j] -= factor * a[col][j];
      }
    }
  }

  return det;
}
