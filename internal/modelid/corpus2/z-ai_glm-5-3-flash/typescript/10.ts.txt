const ALPHABET =
  "0123456789abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ";
const CODE_LENGTH = 8;
const DEFAULT_TTL_SECONDS = 60 * 60; // 1 hour

interface Entry {
  url: string;
  expiresAt: number; // epoch milliseconds
}

export class UrlShortener {
  private entries = new Map<string, Entry>();

  /** Store `url` and return a short code that expires after `ttlSeconds`. */
  shorten(url: string, ttlSeconds: number = DEFAULT_TTL_SECONDS): string {
    const expiresAt = Date.now() + ttlSeconds * 1000;
    let code: string;
    do {
      code = this.randomCode();
    } while (this.entries.has(code));
    this.entries.set(code, { url, expiresAt });
    return code;
  }

  /** Return the original URL for `code`, or undefined if unknown or expired. */
  resolve(code: string): string | undefined {
    const entry = this.entries.get(code);
    if (entry === undefined) return undefined;
    if (Date.now() >= entry.expiresAt) {
      this.entries.delete(code);
      return undefined;
    }
    return entry.url;
  }

  /** Remove all expired entries from the store. */
  purgeExpired(): void {
    const now = Date.now();
    for (const [code, entry] of this.entries) {
      if (now >= entry.expiresAt) this.entries.delete(code);
    }
  }

  private randomCode(): string {
    let code = "";
    for (let i = 0; i < CODE_LENGTH; i++) {
      code += ALPHABET[Math.floor(Math.random() * ALPHABET.length)];
    }
    return code;
  }
}

// Example usage:
//
//   const shortener = new UrlShortener();
//   const code = shortener.shorten("https://example.com", 60); // expires in 60s
//   console.log(shortener.resolve(code)); // "https://example.com"
//
//   setTimeout(() => {
//     console.log(shortener.resolve(code)); // undefined (expired)
//   }, 61_000);
