/**
 * Convert arbitrary text into a URL-friendly slug.
 *
 * - Lowercases the input.
 * - Replaces runs of non-alphanumeric characters with a single hyphen.
 * - Trims leading and trailing hyphens.
 * - Preserves Unicode letters and numbers (e.g. "café" -> "café",
 *   "日本語 テキスト" -> "日本語-テキスト"), including combining marks
 *   so decomposed accented characters stay intact.
 */
export function slugify(text: string): string {
  return text
    .toLowerCase()
    .replace(/[^\p{L}\p{M}\p{N}]+/gu, "-")
    .replace(/^-+/, "")
    .replace(/-+$/, "");
}
