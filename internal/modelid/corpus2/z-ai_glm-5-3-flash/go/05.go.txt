// Command wordfreq reads text, normalises it (lower-case, punctuation
// stripped), and prints the top K words with their counts.
//
// Usage:
//
//	wordfreq [-k N] [files...]
//
// Text is read from the given files, or from standard input when no files
// are given. Every character that is not a Unicode letter or digit is
// treated as a word separator, so "Hello, WORLD!" yields "hello" and
// "world". Output lines are: rank, count, word (tab separated), ordered by
// descending count with ties broken alphabetically.
package main

import (
	"flag"
	"fmt"
	"io"
	"os"
	"sort"
	"strings"
	"unicode"
)

// wordCount pairs a word with the number of times it occurred.
type wordCount struct {
	word  string
	count int
}

// countWords reads r, splits it into normalised words, and increments the
// corresponding entries in counts.
func countWords(r io.Reader, counts map[string]int) error {
	data, err := io.ReadAll(r)
	if err != nil {
		return err
	}

	var b strings.Builder
	flush := func() {
		if b.Len() > 0 {
			counts[b.String()]++
			b.Reset()
		}
	}

	for _, rn := range string(data) {
		if unicode.IsLetter(rn) || unicode.IsDigit(rn) {
			b.WriteRune(unicode.ToLower(rn))
		} else {
			flush()
		}
	}
	flush()

	return nil
}

// topWords returns up to k of the most frequent words, most frequent first,
// with ties broken alphabetically for deterministic output.
func topWords(counts map[string]int, k int) []wordCount {
	words := make([]wordCount, 0, len(counts))
	for w, c := range counts {
		words = append(words, wordCount{word: w, count: c})
	}

	sort.Slice(words, func(i, j int) bool {
		if words[i].count != words[j].count {
			return words[i].count > words[j].count
		}
		return words[i].word < words[j].word
	})

	if k > len(words) {
		k = len(words)
	}
	if k < 0 {
		k = 0
	}
	return words[:k]
}

func main() {
	k := flag.Int("k", 10, "how many top words to print")
	flag.Parse()

	if *k < 0 {
		fmt.Fprintln(os.Stderr, "wordfreq: -k must not be negative")
		os.Exit(2)
	}

	counts := make(map[string]int)

	if args := flag.Args(); len(args) == 0 {
		if err := countWords(os.Stdin, counts); err != nil {
			fmt.Fprintln(os.Stderr, "wordfreq:", err)
			os.Exit(1)
		}
	} else {
		for _, path := range args {
			f, err := os.Open(path)
			if err != nil {
				fmt.Fprintln(os.Stderr, "wordfreq:", err)
				os.Exit(1)
			}
			err = countWords(f, counts)
			f.Close()
			if err != nil {
				fmt.Fprintln(os.Stderr, "wordfreq:", err)
				os.Exit(1)
			}
		}
	}

	for i, wc := range topWords(counts, *k) {
		fmt.Printf("%d\t%d\t%s\n", i+1, wc.count, wc.word)
	}
}
