package main

import (
	"fmt"
	"sync"
	"time"
)

// TokenBucket is a token-bucket rate limiter.
//
// The bucket starts full, holds at most capacity tokens, and is refilled
// lazily at refillRate tokens per second as time passes. All methods are
// safe for concurrent use by multiple goroutines.
type TokenBucket struct {
	mu         sync.Mutex
	tokens     float64
	capacity   float64
	refillRate float64 // tokens added per second
	lastRefill time.Time
}

// NewTokenBucket returns a full bucket with the given capacity and refill
// rate (tokens per second).
func NewTokenBucket(capacity int, refillRate float64) *TokenBucket {
	if capacity <= 0 {
		panic("tokenbucket: capacity must be positive")
	}
	if refillRate <= 0 {
		panic("tokenbucket: refill rate must be positive")
	}
	return &TokenBucket{
		tokens:     float64(capacity),
		capacity:   float64(capacity),
		refillRate: refillRate,
		lastRefill: time.Now(),
	}
}

// refill brings the token count up to date for the time elapsed since the
// last refill. The caller must hold t.mu.
func (t *TokenBucket) refill(now time.Time) {
	elapsed := now.Sub(t.lastRefill).Seconds()
	if elapsed <= 0 {
		return
	}
	t.tokens += elapsed * t.refillRate
	if t.tokens > t.capacity {
		t.tokens = t.capacity
	}
	t.lastRefill = now
}

// Allow reports whether one token is available and consumes it if so.
func (t *TokenBucket) Allow() bool {
	return t.AllowN(1)
}

// AllowN reports whether n tokens are available and consumes them if so.
// If fewer than n tokens are available, nothing is consumed.
func (t *TokenBucket) AllowN(n int) bool {
	if n <= 0 {
		return true
	}
	t.mu.Lock()
	defer t.mu.Unlock()
	t.refill(time.Now())
	if t.tokens >= float64(n) {
		t.tokens -= float64(n)
		return true
	}
	return false
}

// Wait blocks until one token is available, then consumes it.
func (t *TokenBucket) Wait() {
	interval := time.Duration(float64(time.Second) / t.refillRate)
	for {
		if t.Allow() {
			return
		}
		time.Sleep(interval)
	}
}

// Tokens returns the number of tokens currently in the bucket.
func (t *TokenBucket) Tokens() float64 {
	t.mu.Lock()
	defer t.mu.Unlock()
	t.refill(time.Now())
	return t.tokens
}

func main() {
	// Up to 5 immediate requests, then 2 more per second.
	bucket := NewTokenBucket(5, 2)

	const (
		workers  = 4
		attempts = 25
	)

	var (
		wg      sync.WaitGroup
		mu      sync.Mutex
		allowed int
	)

	for w := 0; w < workers; w++ {
		wg.Add(1)
		go func(id int) {
			defer wg.Done()
			for i := 0; i < attempts; i++ {
				if bucket.Allow() {
					mu.Lock()
					allowed++
					mu.Unlock()
					fmt.Printf("worker %d: allowed\n", id)
				} else {
					fmt.Printf("worker %d: rate limited\n", id)
				}
				time.Sleep(50 * time.Millisecond)
			}
		}(w)
	}

	wg.Wait()
	fmt.Printf("%d of %d requests allowed; tokens remaining: %.2f\n",
		allowed, workers*attempts, bucket.Tokens())
}
