package main

import "fmt"

// Node is a single node in the binary search tree.
type Node struct {
	key   int
	left  *Node
	right *Node
}

// BST is a binary search tree rooted at root.
type BST struct {
	root *Node
}

// Insert adds key to the tree. Duplicate keys are ignored.
func (t *BST) Insert(key int) {
	t.root = insert(t.root, key)
}

func insert(n *Node, key int) *Node {
	if n == nil {
		return &Node{key: key}
	}
	if key < n.key {
		n.left = insert(n.left, key)
	} else if key > n.key {
		n.right = insert(n.right, key)
	}
	return n
}

// Search reports whether key is present in the tree.
func (t *BST) Search(key int) bool {
	n := t.root
	for n != nil {
		switch {
		case key < n.key:
			n = n.left
		case key > n.key:
			n = n.right
		default:
			return true
		}
	}
	return false
}

// Delete removes key from the tree if present.
func (t *BST) Delete(key int) {
	t.root = deleteNode(t.root, key)
}

func deleteNode(n *Node, key int) *Node {
	if n == nil {
		return nil
	}
	if key < n.key {
		n.left = deleteNode(n.left, key)
		return n
	}
	if key > n.key {
		n.right = deleteNode(n.right, key)
		return n
	}

	// Found the node to delete.
	if n.left == nil {
		return n.right
	}
	if n.right == nil {
		return n.left
	}

	// Two children: replace key with in-order successor
	// (smallest key in the right subtree), then delete it.
	successor := n.right
	for successor.left != nil {
		successor = successor.left
	}
	n.key = successor.key
	n.right = deleteNode(n.right, successor.key)
	return n
}

// InOrder returns all keys in ascending order.
func (t *BST) InOrder() []int {
	out := []int{}
	var walk func(n *Node)
	walk = func(n *Node) {
		if n == nil {
			return
		}
		walk(n.left)
		out = append(out, n.key)
		walk(n.right)
	}
	walk(t.root)
	return out
}

func main() {
	bst := &BST{}

	for _, k := range []int{50, 30, 70, 20, 40, 60, 80} {
		bst.Insert(k)
	}

	fmt.Println("In-order traversal:", bst.InOrder())

	fmt.Println("Search 40:", bst.Search(40))
	fmt.Println("Search 45:", bst.Search(45))

	bst.Delete(20) // leaf node
	bst.Delete(30) // node with one child (after 20 was removed)
	bst.Delete(50) // node with two children

	fmt.Println("After deleting 20, 30, 50:", bst.InOrder())
}
