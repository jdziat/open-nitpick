package main

import (
	"fmt"
	"iter"
	"time"
)

// DateRange iterates over each day between Start and End, inclusive.
type DateRange struct {
	Start        time.Time
	End          time.Time
	SkipWeekends bool
}

// NewDateRange creates a range from start to end inclusive. Both dates are
// normalized to midnight in their own location. If end is before start,
// the two are swapped so iteration always moves forward.
func NewDateRange(start, end time.Time, skipWeekends bool) DateRange {
	start = time.Date(start.Year(), start.Month(), start.Day(), 0, 0, 0, 0, start.Location())
	end = time.Date(end.Year(), end.Month(), end.Day(), 0, 0, 0, 0, end.Location())
	if end.Before(start) {
		start, end = end, start
	}
	return DateRange{Start: start, End: end, SkipWeekends: skipWeekends}
}

// Days returns an iterator yielding each day in the range, inclusive of both
// endpoints. If SkipWeekends is true, Saturdays and Sundays are omitted.
// Iteration stops early if the consumer breaks out of the range loop.
func (r DateRange) Days() iter.Seq[time.Time] {
	return func(yield func(time.Time) bool) {
		for d := r.Start; !d.After(r.End); d = d.AddDate(0, 0, 1) {
			if r.SkipWeekends {
				switch d.Weekday() {
				case time.Saturday, time.Sunday:
					continue
				}
			}
			if !yield(d) {
				return
			}
		}
	}
}

func main() {
	start := time.Date(2025, time.May, 23, 0, 0, 0, 0, time.UTC)
	end := time.Date(2025, time.June, 2, 0, 0, 0, 0, time.UTC)

	fmt.Println("All days:")
	for day := range NewDateRange(start, end, false).Days() {
		fmt.Println(day.Format("Mon 2006-01-02"))
	}

	fmt.Println("\nWeekdays only:")
	for day := range NewDateRange(start, end, true).Days() {
		fmt.Println(day.Format("Mon 2006-01-02"))
	}

	fmt.Println("\nEarly exit after first 3 weekdays:")
	count := 0
	for day := range NewDateRange(start, end, true).Days() {
		fmt.Println(day.Format("Mon 2006-01-02"))
		count++
		if count == 3 {
			break
		}
	}
}
