package main

import (
	"encoding/json"
	"errors"
	"fmt"
	"os"
	"reflect"
	"strconv"
	"strings"
	"time"
)

// Config is the application configuration.
//
// Loading precedence (lowest to highest):
//
//  1. `default` struct tags
//  2. values from the JSON file
//  3. environment variables named by the `env` tag
//
// Example config.json:
//
//	{
//	  "app_name": "myservice",
//	  "log_level": "debug",
//	  "listen_addr": ":9000",
//	  "read_timeout": "10s",
//	  "workers": 8,
//	  "verbose": true,
//	  "database_url": "postgres://user:pass@localhost:5432/mydb"
//	}
type Config struct {
	AppName     string        `json:"app_name"     env:"APP_NAME"     default:"myservice"`
	LogLevel    string        `json:"log_level"    env:"LOG_LEVEL"    default:"info"  oneof:"debug,info,warn,error"`
	ListenAddr  string        `json:"listen_addr"  env:"LISTEN_ADDR"  default:":8080"`
	ReadTimeout time.Duration `json:"read_timeout" env:"READ_TIMEOUT" default:"5s"`
	Workers     int           `json:"workers"      env:"WORKERS"      default:"4"`
	Verbose     bool          `json:"verbose"      env:"VERBOSE"      default:"false"`
	DatabaseURL string        `json:"database_url" env:"DATABASE_URL"                 required:"true"`
}

// Load builds a Config from defaults, the JSON file at path (if it exists),
// and environment variables, then validates required fields.
func Load(path string) (*Config, error) {
	cfg := &Config{}

	if err := applyDefaults(cfg); err != nil {
		return nil, fmt.Errorf("apply defaults: %w", err)
	}

	data, err := os.ReadFile(path)
	switch {
	case err == nil:
		dec := json.NewDecoder(strings.NewReader(string(data)))
		dec.DisallowUnknownFields() // catch typos in the config file
		if err := dec.Decode(cfg); err != nil {
			return nil, fmt.Errorf("parse %s: %w", path, err)
		}
	case errors.Is(err, os.ErrNotExist):
		// A missing file is acceptable: defaults plus environment are enough.
	default:
		return nil, fmt.Errorf("read %s: %w", path, err)
	}

	if err := applyEnvOverrides(cfg); err != nil {
		return nil, fmt.Errorf("env overrides: %w", err)
	}

	if err := validate(cfg); err != nil {
		return nil, err
	}
	return cfg, nil
}

// forEachField calls fn for every exported field of the struct pointed to by cfg.
func forEachField(cfg *Config, fn func(sf reflect.StructField, fv reflect.Value) error) error {
	rv := reflect.ValueOf(cfg).Elem()
	rt := rv.Type()
	for i := 0; i < rt.NumField(); i++ {
		if err := fn(rt.Field(i), rv.Field(i)); err != nil {
			return fmt.Errorf("%s: %w", rt.Field(i).Name, err)
		}
	}
	return nil
}

func applyDefaults(cfg *Config) error {
	return forEachField(cfg, func(sf reflect.StructField, fv reflect.Value) error {
		def := sf.Tag.Get("default")
		if def == "" || !fv.IsZero() {
			return nil
		}
		return setValue(fv, def)
	})
}

func applyEnvOverrides(cfg *Config) error {
	return forEachField(cfg, func(sf reflect.StructField, fv reflect.Value) error {
		name := sf.Tag.Get("env")
		if name == "" {
			return nil
		}
		raw, ok := os.LookupEnv(name)
		if !ok || strings.TrimSpace(raw) == "" {
			return nil
		}
		return setValue(fv, strings.TrimSpace(raw))
	})
}

func validate(cfg *Config) error {
	return forEachField(cfg, func(sf reflect.StructField, fv reflect.Value) error {
		jsonName := strings.Split(sf.Tag.Get("json"), ",")[0]

		if sf.Tag.Get("required") == "true" && fv.IsZero() {
			return fmt.Errorf(
				"required field %q is empty (set it in the config file or the %s environment variable)",
				jsonName, sf.Tag.Get("env"),
			)
		}

		if oneof := sf.Tag.Get("oneof"); oneof != "" && fv.Kind() == reflect.String && !fv.IsZero() {
			val := fv.String()
			for _, allowed := range strings.Split(oneof, ",") {
				if val == allowed {
					return nil
				}
			}
			return fmt.Errorf("field %q has value %q; must be one of: %s", jsonName, val, oneof)
		}
		return nil
	})
}

// setValue parses s according to the field's type and stores it.
func setValue(fv reflect.Value, s string) error {
	switch fv.Kind() {
	case reflect.String:
		fv.SetString(s)
	case reflect.Bool:
		b, err := strconv.ParseBool(s)
		if err != nil {
			return fmt.Errorf("invalid bool %q: %w", s, err)
		}
		fv.SetBool(b)
	case reflect.Int, reflect.Int8, reflect.Int16, reflect.Int32, reflect.Int64:
		if fv.Type() == reflect.TypeOf(time.Duration(0)) {
			d, err := time.ParseDuration(s)
			if err != nil {
				return fmt.Errorf("invalid duration %q: %w", s, err)
			}
			fv.SetInt(int64(d))
			return nil
		}
		n, err := strconv.ParseInt(s, 10, fv.Type().Bits())
		if err != nil {
			return fmt.Errorf("invalid int %q: %w", s, err)
		}
		fv.SetInt(n)
	case reflect.Uint, reflect.Uint8, reflect.Uint16, reflect.Uint32, reflect.Uint64:
		n, err := strconv.ParseUint(s, 10, fv.Type().Bits())
		if err != nil {
			return fmt.Errorf("invalid uint %q: %w", s, err)
		}
		fv.SetUint(n)
	case reflect.Float32, reflect.Float64:
		f, err := strconv.ParseFloat(s, fv.Type().Bits())
		if err != nil {
			return fmt.Errorf("invalid float %q: %w", s, err)
		}
		fv.SetFloat(f)
	default:
		return fmt.Errorf("unsupported field type %s", fv.Type())
	}
	return nil
}

// describe renders the final configuration field by field.
func describe(cfg *Config) string {
	var b strings.Builder
	rv := reflect.ValueOf(cfg).Elem()
	rt := rv.Type()
	for i := 0; i < rt.NumField(); i++ {
		fv := rv.Field(i)
		name := strings.Split(rt.Field(i).Tag.Get("json"), ",")[0]
		fmt.Fprintf(&b, "  %-14s = %v\n", name, fv.Interface())
	}
	return b.String()
}

func main() {
	path := "config.json"
	if len(os.Args) > 1 {
		path = os.Args[1]
	}

	cfg, err := Load(path)
	if err != nil {
		fmt.Fprintln(os.Stderr, "configuration error:", err)
		os.Exit(1)
	}

	fmt.Printf("configuration loaded (file: %s):\n%s", path, describe(cfg))
}
