from collections import OrderedDict


class LRUCache:
    """A fixed-capacity LRU (Least Recently Used) cache.

    Provides O(1) get and put operations. When inserting a new entry
    would exceed the cache's capacity, the least recently used entry
    is evicted.
    """

    def __init__(self, capacity):
        if capacity <= 0:
            raise ValueError("capacity must be a positive integer")
        self._capacity = capacity
        self._data = OrderedDict()  # order: oldest (front) -> newest (back)

    def get(self, key, default=None):
        """Return the value for key, or default if the key is absent.

        If present, the entry is marked as most recently used.
        """
        if key not in self._data:
            return default
        self._data.move_to_end(key)
        return self._data[key]

    def put(self, key, value):
        """Insert or update key with value, marking it most recently used.

        Evicts the least recently used entry if the cache is over capacity.
        """
        if key in self._data:
            self._data.move_to_end(key)
        self._data[key] = value
        if len(self._data) > self._capacity:
            self._data.popitem(last=False)

    def __contains__(self, key):
        return key in self._data

    def __len__(self):
        return len(self._data)

    def __repr__(self):
        items = ", ".join(f"{k!r}: {v!r}" for k, v in self._data.items())
        return f"LRUCache({{{items}}})"


if __name__ == "__main__":
    cache = LRUCache(3)
    cache.put("a", 1)
    cache.put("b", 2)
    cache.put("c", 3)
    print(cache)            # LRUCache({'a': 1, 'b': 2, 'c': 3})

    print(cache.get("a"))   # 1  ('a' is now most recently used)
    cache.put("d", 4)       # evicts 'b', the least recently used entry
    print(cache)            # LRUCache({'c': 3, 'a': 1, 'd': 4})
    print(cache.get("b"))   # None ('b' was evicted)

    cache.put("a", 10)      # update existing key, refresh its recency
    print(cache.get("a"))   # 10
    print(len(cache))       # 3
