"""date_range.py

A date-range iterator that yields each day between two dates, inclusive,
with an option to skip weekends.

Library usage:

    import datetime as dt
    from date_range import date_range

    for day in date_range(dt.date(2024, 1, 1), dt.date(2024, 1, 7)):
        print(day)

    for business_day in date_range(
        dt.date(2024, 1, 1), dt.date(2024, 1, 31), skip_weekends=True
    ):
        print(business_day)

Command-line usage:

    python date_range.py 2024-01-01 2024-01-07
    python date_range.py 2024-01-01 2024-01-31 --skip-weekends
"""

from __future__ import annotations

import argparse
import datetime as dt
from collections.abc import Iterator

__all__ = ["date_range"]

# datetime.date.weekday(): Monday=0 ... Saturday=5, Sunday=6
_WEEKEND = frozenset({5, 6})


def date_range(
    start: dt.date,
    end: dt.date,
    *,
    skip_weekends: bool = False,
) -> Iterator[dt.date]:
    """Yield every date from ``start`` through ``end``, inclusive.

    If ``start`` is later than ``end``, the days are yielded in reverse
    order. When ``skip_weekends`` is True, Saturdays and Sundays are
    omitted.

    Args:
        start: The first date in the range.
        end: The last date in the range.
        skip_weekends: If True, exclude Saturday and Sunday.

    Yields:
        Each date in the range, one day at a time.

    Examples:
        >>> list(date_range(dt.date(2024, 1, 5), dt.date(2024, 1, 8),
        ...                 skip_weekends=True))
        [datetime.date(2024, 1, 5), datetime.date(2024, 1, 8)]
    """
    going_forward = start <= end
    one_day = dt.timedelta(days=1)

    current = start
    if going_forward:
        while current <= end:
            if not skip_weekends or current.weekday() not in _WEEKEND:
                yield current
            current += one_day
    else:
        while current >= end:
            if not skip_weekends or current.weekday() not in _WEEKEND:
                yield current
            current -= one_day


def _parse_date(text: str) -> dt.date:
    """Parse a CLI date argument in ISO format (YYYY-MM-DD)."""
    try:
        return dt.date.fromisoformat(text)
    except ValueError:
        raise argparse.ArgumentTypeError(
            f"invalid date {text!r}: expected YYYY-MM-DD"
        ) from None


def main(argv: list[str] | None = None) -> int:
    """Command-line entry point."""
    parser = argparse.ArgumentParser(
        description="Print each day between two dates, inclusive.",
    )
    parser.add_argument("start", type=_parse_date, help="start date (YYYY-MM-DD)")
    parser.add_argument("end", type=_parse_date, help="end date (YYYY-MM-DD)")
    parser.add_argument(
        "-w",
        "--skip-weekends",
        action="store_true",
        help="exclude Saturdays and Sundays from the output",
    )
    args = parser.parse_args(argv)

    for day in date_range(args.start, args.end, skip_weekends=args.skip_weekends):
        print(day.isoformat())
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
