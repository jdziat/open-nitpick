#!/usr/bin/env python3
"""wordfreq.py - a word-frequency counter.

Reads text from files given on the command line (or stdin if none are
given), normalises case and punctuation, and prints the top K words
with their counts.

Usage:
    python wordfreq.py [-k K] [FILE ...]
    some_command | python wordfreq.py -k 5
"""

import argparse
import re
import sys
from collections import Counter

# A word is a run of letters/digits (unicode-aware), optionally joined by
# internal apostrophes so "don't" stays one word. Surrounding punctuation
# such as quotes, commas, or periods is dropped.
TOKEN_RE = re.compile(r"[^\W_]+(?:'[^\W_]+)*", re.UNICODE)


def normalise(text):
    """Lowercase `text` and split it into word tokens, ignoring punctuation."""
    return TOKEN_RE.findall(text.lower())


def count_words(tokens):
    """Return a Counter mapping each word to its number of occurrences."""
    return Counter(tokens)


def top_k(counts, k):
    """Return the k most common words as (word, count) tuples.

    Ties are broken alphabetically so the output is deterministic.
    """
    return sorted(counts.items(), key=lambda item: (-item[1], item[0]))[:k]


def top_words(text, k=10):
    """Normalise, count, and return the top k words found in `text`."""
    return top_k(count_words(normalise(text)), k)


def _read_one(path):
    with open(path, "r", encoding="utf-8") as handle:
        return handle.read()


def read_text(paths):
    """Read and concatenate the given files, or stdin if none are given."""
    if not paths:
        return sys.stdin.read()
    return "\n".join(_read_one(path) for path in paths)


def main(argv=None):
    parser = argparse.ArgumentParser(
        description="Print the top K words in a text."
    )
    parser.add_argument(
        "-k", type=int, default=10, metavar="K",
        help="number of top words to show (default: 10)",
    )
    parser.add_argument(
        "files", nargs="*", metavar="FILE",
        help="input files (reads stdin if omitted)",
    )
    args = parser.parse_args(argv)

    if args.k < 1:
        parser.error("K must be a positive integer")

    for word, count in top_words(read_text(args.files), args.k):
        print(f"{count:6d}  {word}")
    return 0


if __name__ == "__main__":
    sys.exit(main())
