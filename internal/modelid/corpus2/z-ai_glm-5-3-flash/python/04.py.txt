"""slugify.py — convert arbitrary text into URL-friendly slugs."""

import re
import unicodedata


def slugify(text: str) -> str:
    """Return a URL-friendly slug for *text*.

    The text is lowercased, Unicode-decomposed so that accented letters
    reduce to their base form (e.g. "é" -> "e"), runs of non-alphanumeric
    characters are replaced with a single hyphen, and leading/trailing
    hyphens are trimmed. Non-ASCII Unicode letters (Cyrillic, Greek,
    CJK, etc.) are preserved rather than stripped.

    Examples:
        slugify("Hello, World!")         -> "hello-world"
        slugify("  Héllo   Wörld!  ")    -> "hello-world"
        slugify("Пример -- заголовок!")  -> "пример-заголовок"
    """
    if not isinstance(text, str):
        text = str(text)

    # Decompose accents into base letter + combining mark (NFKD also
    # expands things like "²" into "2" and "½" into "1/2").
    text = unicodedata.normalize("NFKD", text)

    # Drop combining marks so accented letters become their base form.
    text = "".join(ch for ch in text if not unicodedata.combining(ch))

    # Unicode-aware lowercasing.
    text = text.lower()

    # Map every non-alphanumeric character to a hyphen. `str.isalnum()`
    # is Unicode-aware, so letters and digits in any script are kept.
    mapped = "".join(ch if ch.isalnum() else "-" for ch in text)

    # Collapse runs of hyphens into a single hyphen.
    mapped = re.sub(r"-+", "-", mapped)

    # Trim hyphens from both ends.
    return mapped.strip("-")


if __name__ == "__main__":
    samples = [
        "Hello, World!",
        "  Multiple   spaces -- and   dashes!  ",
        "Héllo Wörld & Çafé",
        "Пример заголовка",
        "Ελληνικός τίτλος",
        "日本語のタイトル",
        "___under_scores___ and 123 numbers",
        "",
        "---",
    ]
    for s in samples:
        print(f"{s!r:45} -> {slugify(s)!r}")
