class Matrix:
    def __init__(self, data):
        if not data or not data[0]:
            raise ValueError("Matrix data cannot be empty")
        self.rows = len(data)
        self.cols = len(data[0])
        for row in data:
            if len(row) != self.cols:
                raise ValueError("All rows must have the same number of columns")
        self.data = [list(row) for row in data]

    def __repr__(self):
        rows_str = '\n'.join(['  ' + str(row) for row in self.data])
        return f"Matrix([\n{rows_str}\n])"

    def __eq__(self, other):
        if not isinstance(other, Matrix):
            return False
        return self.data == other.data

    def __mul__(self, other):
        if isinstance(other, (int, float)):
            result = [[self.data[i][j] * other for j in range(self.cols)]
                      for i in range(self.rows)]
            return Matrix(result)
        if isinstance(other, Matrix):
            if self.cols != other.rows:
                raise ValueError(
                    f"Shape mismatch for multiplication: "
                    f"({self.rows}x{self.cols}) and ({other.rows}x{other.cols})"
                )
            result = []
            for i in range(self.rows):
                row = []
                for j in range(other.cols):
                    val = sum(self.data[i][k] * other.data[k][j]
                              for k in range(self.cols))
                    row.append(val)
                result.append(row)
            return Matrix(result)
        return NotImplemented

    def __rmul__(self, other):
        if isinstance(other, (int, float)):
            return self.__mul__(other)
        return NotImplemented

    def transpose(self):
        result = [[self.data[i][j] for i in range(self.rows)]
                  for j in range(self.cols)]
        return Matrix(result)

    def determinant(self):
        if self.rows != self.cols:
            raise ValueError(
                f"Determinant requires a square matrix, got {self.rows}x{self.cols}"
            )
        n = self.rows
        # Work on a copy using LU decomposition (Gaussian elimination)
        mat = [row[:] for row in self.data]
        det = 1.0
        for col in range(n):
            # Find pivot
            max_row = max(range(col, n), key=lambda r: abs(mat[r][col]))
            if abs(mat[max_row][col]) < 1e-12:
                return 0
            if max_row != col:
                mat[col], mat[max_row] = mat[max_row], mat[col]
                det *= -1
            det *= mat[col][col]
            pivot = mat[col][col]
            for row in range(col + 1, n):
                factor = mat[row][col] / pivot
                for k in range(col, n):
                    mat[row][k] -= factor * mat[col][k]
        # Round to avoid floating point noise for integer matrices
        if abs(det - round(det)) < 1e-9:
            return int(round(det))
        return det
