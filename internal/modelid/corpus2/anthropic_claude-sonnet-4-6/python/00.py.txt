class LRUCache:
    def __init__(self, capacity: int):
        if capacity <= 0:
            raise ValueError("Capacity must be a positive integer")
        self.capacity = capacity
        self.cache = {}  # key -> [value, prev_key, next_key]
        # Doubly linked list with sentinel head and tail
        self.head = 'HEAD'  # Most recently used side
        self.tail = 'TAIL'  # Least recently used side
        # head <-> tail (initially empty)
        self.prev = {self.tail: self.head}
        self.next = {self.head: self.tail}

    def _remove(self, key):
        """Remove a node from the doubly linked list."""
        p = self.prev[key]
        n = self.next[key]
        self.next[p] = n
        self.prev[n] = p
        del self.prev[key]
        del self.next[key]

    def _insert_after_head(self, key):
        """Insert a node right after head (most recently used position)."""
        n = self.next[self.head]
        self.next[self.head] = key
        self.prev[key] = self.head
        self.next[key] = n
        self.prev[n] = key

    def get(self, key: int) -> int:
        """Return the value of the key if it exists, otherwise return -1."""
        if key not in self.cache:
            return -1
        # Move to most recently used
        self._remove(key)
        self._insert_after_head(key)
        return self.cache[key]

    def put(self, key: int, value: int) -> None:
        """Insert or update the key-value pair."""
        if key in self.cache:
            self._remove(key)
        elif len(self.cache) >= self.capacity:
            # Evict least recently used (node before tail)
            lru_key = self.prev[self.tail]
            self._remove(lru_key)
            del self.cache[lru_key]
        self.cache[key] = value
        self._insert_after_head(key)

    def __repr__(self):
        items = []
        current = self.next[self.head]
        while current != self.tail:
            items.append(f"{current}:{self.cache[current]}")
            current = self.next[current]
        return f"LRUCache(capacity={self.capacity}, items=[{', '.join(items)}])"


def main():
    print("=== LRU Cache Demo ===\n")

    cache = LRUCache(3)
    print(f"Created cache with capacity 3")

    cache.put(1, 10)
    print(f"put(1, 10) -> {cache}")

    cache.put(2, 20)
    print(f"put(2, 20) -> {cache}")

    cache.put(3, 30)
    print(f"put(3, 30) -> {cache}")

    print(f"get(1) = {cache.get(1)}")  # Returns 10, moves 1 to front
    print(f"After get(1): {cache}")

    cache.put(4, 40)  # Evicts key 2 (LRU)
    print(f"put(4, 40) -> {cache}  (evicts key 2)")

    print(f"get(2) = {cache.get(2)}")  # Returns -1 (evicted)
    print(f"get(3) = {cache.get(3)}")  # Returns 30
    print(f"get(4) = {cache.get(4)}")  # Returns 40

    print(f"\nAfter gets: {cache}")

    cache.put(5, 50)  # Evicts key 1 (LRU now)
    print(f"put(5, 50) -> {cache}  (evicts key 1)")

    print(f"get(1) = {cache.get(1)}")  # Returns -1 (evicted)

    print("\n=== Update existing key ===")
    cache.put(3, 300)
    print(f"put(3, 300) -> {cache}")
    print(f"get(3) = {cache.get(3)}")  # Returns 300

    print("\n=== Capacity 1 edge case ===")
    small = LRUCache(1)
    small.put(1, 1)
    print(f"put(1,1) -> {small}")
    small.put(2, 2)
    print(f"put(2,2) -> {small}  (evicts key 1)")
    print(f"get(1) = {small.get(1)}")  # -1
    print(f"get(2) = {small.get(2)}")  # 2


if __name__ == "__main__":
    main()
