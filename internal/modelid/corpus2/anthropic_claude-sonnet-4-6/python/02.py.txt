import threading
import time


class TokenBucket:
    """
    A thread-safe token-bucket rate limiter.

    Parameters
    ----------
    capacity : float
        Maximum number of tokens the bucket can hold.
    refill_rate : float
        Number of tokens added to the bucket per second.
    """

    def __init__(self, capacity: float, refill_rate: float) -> None:
        if capacity <= 0:
            raise ValueError("capacity must be positive")
        if refill_rate <= 0:
            raise ValueError("refill_rate must be positive")

        self._capacity = float(capacity)
        self._refill_rate = float(refill_rate)
        self._tokens = float(capacity)          # start full
        self._last_refill = time.monotonic()
        self._lock = threading.Lock()

    # ------------------------------------------------------------------
    # Internal helpers (must be called with self._lock held)
    # ------------------------------------------------------------------

    def _refill(self) -> None:
        """Add tokens that have accumulated since the last call."""
        now = time.monotonic()
        elapsed = now - self._last_refill
        added = elapsed * self._refill_rate
        self._tokens = min(self._capacity, self._tokens + added)
        self._last_refill = now

    # ------------------------------------------------------------------
    # Public API
    # ------------------------------------------------------------------

    def consume(self, tokens: float = 1.0) -> bool:
        """
        Try to consume *tokens* tokens without blocking.

        Returns
        -------
        bool
            ``True`` if the tokens were available and have been consumed,
            ``False`` otherwise (the bucket is unchanged on failure).
        """
        if tokens <= 0:
            raise ValueError("tokens must be positive")

        with self._lock:
            self._refill()
            if self._tokens >= tokens:
                self._tokens -= tokens
                return True
            return False

    def consume_blocking(self, tokens: float = 1.0, timeout: float | None = None) -> bool:
        """
        Block until *tokens* tokens are available, then consume them.

        Parameters
        ----------
        tokens : float
            Number of tokens to consume.
        timeout : float | None
            Maximum number of seconds to wait.  ``None`` means wait forever.

        Returns
        -------
        bool
            ``True`` if tokens were consumed, ``False`` if *timeout* expired.
        """
        if tokens <= 0:
            raise ValueError("tokens must be positive")
        if tokens > self._capacity:
            raise ValueError("tokens exceeds bucket capacity and can never be satisfied")

        deadline = None if timeout is None else time.monotonic() + timeout

        while True:
            with self._lock:
                self._refill()
                if self._tokens >= tokens:
                    self._tokens -= tokens
                    return True

                # How long until enough tokens accumulate?
                deficit = tokens - self._tokens
                wait = deficit / self._refill_rate

            # Clamp wait to the remaining timeout (if any).
            if deadline is not None:
                remaining = deadline - time.monotonic()
                if remaining <= 0:
                    return False
                wait = min(wait, remaining)

            time.sleep(wait)

    @property
    def capacity(self) -> float:
        """Maximum token capacity."""
        return self._capacity

    @property
    def refill_rate(self) -> float:
        """Tokens added per second."""
        return self._refill_rate

    @property
    def tokens(self) -> float:
        """Current token count (snapshot; may be stale immediately)."""
        with self._lock:
            self._refill()
            return self._tokens

    def __repr__(self) -> str:
        return (
            f"TokenBucket(capacity={self._capacity}, "
            f"refill_rate={self._refill_rate}, "
            f"tokens={self.tokens:.3f})"
        )


# ---------------------------------------------------------------------------
# Quick demonstration / smoke-test
# ---------------------------------------------------------------------------

def _demo() -> None:
    import random

    print("=== Token Bucket Demo ===\n")

    # 1. Basic non-blocking usage
    bucket = TokenBucket(capacity=5, refill_rate=2)
    print(f"Initial state : {bucket}")

    for i in range(7):
        ok = bucket.consume(1)
        print(f"  consume(1) attempt {i + 1}: {'OK' if ok else 'DENIED'}  | {bucket}")

    print()

    # 2. Concurrent producers / consumers
    bucket2 = TokenBucket(capacity=10, refill_rate=5)
    results = {"ok": 0, "denied": 0}
    results_lock = threading.Lock()

    def worker(n: int) -> None:
        for _ in range(n):
            time.sleep(random.uniform(0.0, 0.05))
            granted = bucket2.consume(1)
            with results_lock:
                if granted:
                    results["ok"] += 1
                else:
                    results["denied"] += 1

    threads = [threading.Thread(target=worker, args=(20,)) for _ in range(5)]
    for t in threads:
        t.start()
    for t in threads:
        t.join()

    print(f"Concurrent test (5 threads × 20 attempts):")
    print(f"  Granted : {results['ok']}")
    print(f"  Denied  : {results['denied']}")
    print()

    # 3. Blocking usage with timeout
    bucket3 = TokenBucket(capacity=3, refill_rate=1)
    bucket3.consume(3)          # drain completely
    print("Blocking consume (up to 2.5 s) after draining bucket …")
    start = time.monotonic()
    ok = bucket3.consume_blocking(tokens=1, timeout=2.5)
    elapsed = time.monotonic() - start
    print(f"  Result: {'OK' if ok else 'TIMED OUT'}  (waited {elapsed:.2f} s)\n")


if __name__ == "__main__":
    _demo()
