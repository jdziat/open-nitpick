import unicodedata
import re


def slugify(text: str) -> str:
    """
    Convert text to a URL-friendly slug.

    Steps:
    1. Normalize Unicode to NFC form.
    2. Lowercase the text.
    3. Transliterate/decompose accented characters (NFKD) and drop combining marks.
    4. Replace runs of non-alphanumeric characters with a single hyphen.
    5. Strip leading and trailing hyphens.
    """
    # Normalize to NFC first
    text = unicodedata.normalize("NFC", text)

    # Lowercase
    text = text.lower()

    # Decompose characters (NFKD) so accented letters become base + combining mark
    text = unicodedata.normalize("NFKD", text)

    # Keep only characters that are not combining marks (i.e., drop diacritics)
    # but keep letters, digits, and other characters
    text = "".join(
        ch for ch in text if not unicodedata.combining(ch)
    )

    # Replace any sequence of characters that are not Unicode letters or digits
    # with a single hyphen
    text = re.sub(r"[^\w]+", "-", text, flags=re.UNICODE)

    # re \w includes underscore, so replace underscores with hyphens
    text = re.sub(r"_+", "-", text)

    # Collapse multiple hyphens into one
    text = re.sub(r"-+", "-", text)

    # Strip leading/trailing hyphens
    text = text.strip("-")

    return text


if __name__ == "__main__":
    examples = [
        "Hello, World!",
        "  Spaced   Out  ",
        "Héllo Wörld",
        "Ångström",
        "日本語テスト",
        "Привет мир",
        "C'est la vie!",
        "under_score_test",
        "multiple---hyphens",
        "UPPER CASE TEXT",
        "café au lait",
        "naïve résumé",
        "100% Python!",
        "foo@bar.com",
        "  ---leading and trailing---  ",
        "",
        "αβγδ Greek letters",
        "Ελληνικά",
    ]

    for example in examples:
        result = slugify(example)
        print(f"{example!r:40s} -> {result!r}")
