import time
import random
import functools
from typing import Callable, Optional, Tuple, Type, TypeVar

T = TypeVar("T")


class RetryError(Exception):
    """Raised when all retry attempts have been exhausted."""

    def __init__(self, message: str, last_exception: Exception):
        super().__init__(message)
        self.last_exception = last_exception


def retry(
    func: Callable[..., T],
    *args,
    max_attempts: int = 3,
    base_delay: float = 1.0,
    max_delay: float = 60.0,
    exponential_base: float = 2.0,
    jitter: bool = True,
    exceptions: Tuple[Type[Exception], ...] = (Exception,),
    on_retry: Optional[Callable[[int, Exception, float], None]] = None,
    **kwargs,
) -> T:
    """
    Call *func* up to *max_attempts* times with exponential back-off and jitter.

    Parameters
    ----------
    func            : callable to invoke
    *args           : positional arguments forwarded to func
    max_attempts    : maximum number of calls (must be >= 1)
    base_delay      : initial delay in seconds before the second attempt
    max_delay       : upper bound on the computed delay
    exponential_base: multiplier applied each round (default 2 → 1 s, 2 s, 4 s …)
    jitter          : when True, adds uniform random noise in [0, current_delay]
    exceptions      : tuple of exception types that trigger a retry
    on_retry        : optional callback(attempt_number, exception, next_delay)
    **kwargs        : keyword arguments forwarded to func

    Returns
    -------
    The return value of *func* on success.

    Raises
    ------
    RetryError      : when all attempts fail; `.last_exception` holds the final error.
    ValueError      : when max_attempts < 1.
    """
    if max_attempts < 1:
        raise ValueError("max_attempts must be >= 1")

    last_exc: Optional[Exception] = None

    for attempt in range(1, max_attempts + 1):
        try:
            return func(*args, **kwargs)
        except exceptions as exc:
            last_exc = exc

            if attempt == max_attempts:
                break

            delay = min(base_delay * (exponential_base ** (attempt - 1)), max_delay)
            if jitter:
                delay = random.uniform(0, delay)

            if on_retry is not None:
                on_retry(attempt, exc, delay)

            time.sleep(delay)

    raise RetryError(
        f"All {max_attempts} attempt(s) failed. "
        f"Last error: {type(last_exc).__name__}: {last_exc}",
        last_exception=last_exc,
    )


def retry_decorator(
    max_attempts: int = 3,
    base_delay: float = 1.0,
    max_delay: float = 60.0,
    exponential_base: float = 2.0,
    jitter: bool = True,
    exceptions: Tuple[Type[Exception], ...] = (Exception,),
    on_retry: Optional[Callable[[int, Exception, float], None]] = None,
):
    """
    Decorator version of *retry*.

    Usage
    -----
    @retry_decorator(max_attempts=5, base_delay=0.5)
    def unstable_function():
        ...
    """

    def decorator(func: Callable[..., T]) -> Callable[..., T]:
        @functools.wraps(func)
        def wrapper(*args, **kwargs) -> T:
            return retry(
                func,
                *args,
                max_attempts=max_attempts,
                base_delay=base_delay,
                max_delay=max_delay,
                exponential_base=exponential_base,
                jitter=jitter,
                exceptions=exceptions,
                on_retry=on_retry,
                **kwargs,
            )

        return wrapper

    return decorator


# ---------------------------------------------------------------------------
# Self-contained demo / smoke-test
# ---------------------------------------------------------------------------
if __name__ == "__main__":
    import sys

    call_count = 0

    def flaky(succeed_on: int) -> str:
        global call_count
        call_count += 1
        if call_count < succeed_on:
            raise ValueError(f"Simulated failure on call {call_count}")
        return f"Success on call {call_count}"

    def log_retry(attempt: int, exc: Exception, delay: float) -> None:
        print(f"  [retry] attempt {attempt} failed ({exc}); retrying in {delay:.3f}s")

    print("=== Test 1: succeeds on 3rd attempt ===")
    call_count = 0
    result = retry(
        flaky,
        3,
        max_attempts=5,
        base_delay=0.1,
        max_delay=1.0,
        jitter=False,
        on_retry=log_retry,
    )
    print(f"Result: {result}\n")

    print("=== Test 2: all attempts fail ===")
    call_count = 0
    try:
        retry(
            flaky,
            999,
            max_attempts=3,
            base_delay=0.1,
            max_delay=1.0,
            jitter=False,
            on_retry=log_retry,
        )
    except RetryError as err:
        print(f"Caught RetryError: {err}")
        print(f"Last exception type: {type(err.last_exception).__name__}\n")

    print("=== Test 3: decorator usage ===")
    call_count = 0

    @retry_decorator(max_attempts=4, base_delay=0.1, max_delay=1.0, jitter=True, on_retry=log_retry)
    def decorated_flaky():
        global call_count
        call_count += 1
        if call_count < 3:
            raise RuntimeError(f"Decorator failure #{call_count}")
        return "Decorator success!"

    result = decorated_flaky()
    print(f"Result: {result}\n")

    print("=== Test 4: only retry on specific exceptions ===")
    call_count = 0

    def type_error_raiser():
        global call_count
        call_count += 1
        raise TypeError("wrong type")

    try:
        retry(
            type_error_raiser,
            max_attempts=3,
            base_delay=0.05,
            exceptions=(ValueError,),  # TypeError should NOT be retried
            on_retry=log_retry,
        )
    except TypeError as exc:
        print(f"TypeError propagated immediately (calls={call_count}): {exc}")
    except RetryError as exc:
        print(f"Unexpected RetryError: {exc}")

    print("\nAll tests passed.")
    sys.exit(0)
