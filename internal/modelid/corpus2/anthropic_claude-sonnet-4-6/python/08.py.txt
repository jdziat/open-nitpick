class BinaryHeap:
    def __init__(self, comparator=None):
        """
        Initialize the binary heap priority queue.
        
        :param comparator: A function that takes two elements and returns True if
                          the first element should have higher priority than the second.
                          Defaults to min-heap behavior (smaller values have higher priority).
        """
        self._heap = []
        if comparator is None:
            self._comparator = lambda a, b: a < b
        else:
            self._comparator = comparator

    def push(self, item):
        """Push an item onto the heap."""
        self._heap.append(item)
        self._sift_up(len(self._heap) - 1)

    def pop(self):
        """Remove and return the highest priority item from the heap."""
        if not self._heap:
            raise IndexError("pop from an empty priority queue")
        
        if len(self._heap) == 1:
            return self._heap.pop()
        
        # Swap the root with the last element
        top = self._heap[0]
        self._heap[0] = self._heap.pop()
        self._sift_down(0)
        return top

    def peek(self):
        """Return the highest priority item without removing it."""
        if not self._heap:
            raise IndexError("peek at an empty priority queue")
        return self._heap[0]

    def __len__(self):
        return len(self._heap)

    def __bool__(self):
        return bool(self._heap)

    def __repr__(self):
        return f"BinaryHeap({self._heap!r})"

    def _sift_up(self, index):
        """Move the element at index up to its correct position."""
        while index > 0:
            parent = (index - 1) // 2
            if self._comparator(self._heap[index], self._heap[parent]):
                self._heap[index], self._heap[parent] = self._heap[parent], self._heap[index]
                index = parent
            else:
                break

    def _sift_down(self, index):
        """Move the element at index down to its correct position."""
        size = len(self._heap)
        while True:
            left = 2 * index + 1
            right = 2 * index + 2
            highest = index

            if left < size and self._comparator(self._heap[left], self._heap[highest]):
                highest = left

            if right < size and self._comparator(self._heap[right], self._heap[highest]):
                highest = right

            if highest == index:
                break

            self._heap[index], self._heap[highest] = self._heap[highest], self._heap[index]
            index = highest


def main():
    print("=== Min-Heap (default) ===")
    min_heap = BinaryHeap()
    for value in [5, 3, 8, 1, 9, 2, 7]:
        min_heap.push(value)
        print(f"  Pushed {value}, peek = {min_heap.peek()}")

    print(f"  Size: {len(min_heap)}")
    print("  Popping all elements:", end=" ")
    while min_heap:
        print(min_heap.pop(), end=" ")
    print()

    print("\n=== Max-Heap (custom comparator) ===")
    max_heap = BinaryHeap(comparator=lambda a, b: a > b)
    for value in [5, 3, 8, 1, 9, 2, 7]:
        max_heap.push(value)
        print(f"  Pushed {value}, peek = {max_heap.peek()}")

    print(f"  Size: {len(max_heap)}")
    print("  Popping all elements:", end=" ")
    while max_heap:
        print(max_heap.pop(), end=" ")
    print()

    print("\n=== Priority Queue with tuples (priority, task) ===")
    task_heap = BinaryHeap(comparator=lambda a, b: a[0] < b[0])
    tasks = [(3, "low priority task"), (1, "high priority task"), (2, "medium priority task")]
    for task in tasks:
        task_heap.push(task)
        print(f"  Added: {task}")

    print("  Processing tasks by priority:")
    while task_heap:
        priority, task = task_heap.pop()
        print(f"  [{priority}] {task}")

    print("\n=== Error handling ===")
    empty_heap = BinaryHeap()
    try:
        empty_heap.pop()
    except IndexError as e:
        print(f"  pop error: {e}")

    try:
        empty_heap.peek()
    except IndexError as e:
        print(f"  peek error: {e}")


if __name__ == "__main__":
    main()
