import csv
import io
from typing import Iterator


def parse_csv(
    source: str,
    delimiter: str = ",",
    quotechar: str = '"',
    skipinitialspace: bool = False,
) -> list[list[str]]:
    """
    Parse a CSV string and return a list of rows, where each row is a list of strings.

    Supports:
    - Quoted fields (fields wrapped in quotechar)
    - Escaped quotes (doubled quotechar inside a quoted field, e.g. "" -> ")
    - Configurable delimiter

    Parameters
    ----------
    source        : The raw CSV text to parse.
    delimiter     : Field separator character (default ',').
    quotechar     : Quote character used to wrap fields (default '"').
    skipinitialspace : Skip whitespace after the delimiter (default False).

    Returns
    -------
    A list of rows; each row is a list of string field values.
    """
    reader = csv.reader(
        io.StringIO(source),
        delimiter=delimiter,
        quotechar=quotechar,
        doublequote=True,          # "" inside a quoted field becomes "
        skipinitialspace=skipinitialspace,
        strict=True,
    )
    return [row for row in reader]


# ---------------------------------------------------------------------------
# Manual / reference implementation (no stdlib csv) – used for illustration
# and as a cross-check in the tests below.
# ---------------------------------------------------------------------------

def _parse_csv_manual(
    source: str,
    delimiter: str = ",",
    quotechar: str = '"',
) -> list[list[str]]:
    """
    A hand-written CSV parser that supports quoted fields and escaped
    (doubled) quote characters.  Line endings \r\n, \r, and \n are all
    treated as record separators.
    """
    rows: list[list[str]] = []
    current_row: list[str] = []
    field_chars: list[str] = []

    IN_FIELD = "IN_FIELD"
    IN_QUOTED = "IN_QUOTED"
    AFTER_QUOTE = "AFTER_QUOTE"
    START = "START"

    state = START
    i = 0
    n = len(source)

    def flush_field() -> None:
        current_row.append("".join(field_chars))
        field_chars.clear()

    def flush_row() -> None:
        flush_field()
        rows.append(list(current_row))
        current_row.clear()

    while i < n:
        ch = source[i]

        # ---- normalise line endings ----------------------------------------
        if ch == "\r":
            # treat \r\n as a single newline
            if i + 1 < n and source[i + 1] == "\n":
                i += 1
            ch = "\n"

        # ---- state machine ---------------------------------------------------
        if state == START:
            if ch == quotechar:
                state = IN_QUOTED
            elif ch == delimiter:
                flush_field()          # empty field
            elif ch == "\n":
                flush_row()
            else:
                field_chars.append(ch)
                state = IN_FIELD

        elif state == IN_FIELD:
            if ch == delimiter:
                flush_field()
                state = START
            elif ch == "\n":
                flush_row()
                state = START
            else:
                field_chars.append(ch)

        elif state == IN_QUOTED:
            if ch == quotechar:
                state = AFTER_QUOTE
            else:
                field_chars.append(ch)

        elif state == AFTER_QUOTE:
            if ch == quotechar:
                # escaped quote: "" -> "
                field_chars.append(quotechar)
                state = IN_QUOTED
            elif ch == delimiter:
                flush_field()
                state = START
            elif ch == "\n":
                flush_row()
                state = START
            else:
                # character after closing quote that is not a delimiter or
                # newline – treat as unquoted continuation (lenient)
                field_chars.append(ch)
                state = IN_FIELD

        i += 1

    # flush any trailing content that was not terminated by a newline
    if field_chars or current_row:
        flush_row()

    # remove a single trailing empty row that results from a terminal newline
    if rows and rows[-1] == [""]:
        rows.pop()

    return rows


# ---------------------------------------------------------------------------
# Tests
# ---------------------------------------------------------------------------

def _run_tests() -> None:
    # --- helpers ---
    def eq(got, expected, label=""):
        assert got == expected, f"FAIL [{label}]\n  got:      {got!r}\n  expected: {expected!r}"

    # 1. simple row
    eq(parse_csv("a,b,c"), [["a", "b", "c"]], "simple")

    # 2. multiple rows
    eq(parse_csv("a,b\nc,d"), [["a", "b"], ["c", "d"]], "multi-row")

    # 3. quoted field
    eq(parse_csv('"hello world",b'), [["hello world", "b"]], "quoted")

    # 4. quoted field containing delimiter
    eq(parse_csv('"a,b",c'), [["a,b", "c"]], "quoted-delimiter")

    # 5. escaped quote inside quoted field
    eq(parse_csv('"say ""hi""",end'), [['say "hi"', "end"]], "escaped-quote")

    # 6. empty fields
    eq(parse_csv(",b,"), [["", "b", ""]], "empty-fields")

    # 7. custom delimiter
    eq(parse_csv("a|b|c", delimiter="|"), [["a", "b", "c"]], "custom-delimiter")

    # 8. custom delimiter with quoted field containing default comma
    eq(parse_csv('"a,b"|c', delimiter="|"), [["a,b", "c"]], "custom-delim-quoted")

    # 9. Windows line endings
    eq(parse_csv("a,b\r\nc,d"), [["a", "b"], ["c", "d"]], "crlf")

    # 10. trailing newline should not produce extra empty row
    eq(parse_csv("a,b\n"), [["a", "b"]], "trailing-newline")

    # 11. quoted field with newline inside
    eq(parse_csv('"line1\nline2",b'), [["line1\nline2", "b"]], "newline-in-quoted")

    # 12. single field
    eq(parse_csv("hello"), [["hello"]], "single-field")

    # 13. all empty
    eq(parse_csv(""), [], "empty-string")

    # 14. cross-check manual parser against stdlib-based parser on a richer sample
    sample = 'id,name,description\n1,"Alice","She said ""hello"""\n2,"Bob, Jr.","Multi\nline"\n'
    stdlib_result = parse_csv(sample)
    manual_result = _parse_csv_manual(sample)
    eq(manual_result, stdlib_result, "cross-check manual vs stdlib")

    print("All tests passed.")


if __name__ == "__main__":
    import sys

    if len(sys.argv) > 1 and sys.argv[1] == "--test":
        _run_tests()
    else:
        # Read from stdin and print parsed rows
        data = sys.stdin.read()
        delimiter = ","
        if len(sys.argv) > 1:
            delimiter = sys.argv[1]
        for row in parse_csv(data, delimiter=delimiter):
            print(row)
