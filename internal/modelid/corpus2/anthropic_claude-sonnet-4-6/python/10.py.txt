import time
import random
import string


class URLShortener:
    def __init__(self):
        self._store = {}

    def _generate_code(self, length=6):
        chars = string.ascii_letters + string.digits
        while True:
            code = "".join(random.choices(chars, k=length))
            if code not in self._store:
                return code

    def shorten(self, url: str, ttl: int = 300) -> str:
        """
        Shorten a URL with an optional time-to-live (TTL) in seconds.
        Returns the short code.
        """
        if not url:
            raise ValueError("URL must not be empty.")
        code = self._generate_code()
        expires_at = time.time() + ttl if ttl > 0 else None
        self._store[code] = {"url": url, "expires_at": expires_at}
        return code

    def resolve(self, code: str) -> str:
        """
        Resolve a short code to the original URL.
        Raises KeyError if not found or expired.
        """
        entry = self._store.get(code)
        if entry is None:
            raise KeyError(f"Short code '{code}' not found.")
        if entry["expires_at"] is not None and time.time() > entry["expires_at"]:
            del self._store[code]
            raise KeyError(f"Short code '{code}' has expired.")
        return entry["url"]

    def delete(self, code: str) -> bool:
        """
        Manually delete a short code.
        Returns True if deleted, False if not found.
        """
        if code in self._store:
            del self._store[code]
            return True
        return False

    def purge_expired(self) -> int:
        """
        Remove all expired entries. Returns the count of removed entries.
        """
        now = time.time()
        expired = [
            code
            for code, entry in self._store.items()
            if entry["expires_at"] is not None and now > entry["expires_at"]
        ]
        for code in expired:
            del self._store[code]
        return len(expired)


def main():
    shortener = URLShortener()

    url1 = "https://www.example.com/some/very/long/path?query=1&other=2"
    url2 = "https://www.python.org/docs/3.11/library/time.html"

    code1 = shortener.shorten(url1, ttl=10)
    code2 = shortener.shorten(url2, ttl=1)

    print(f"Shortened '{url1}' -> '{code1}'")
    print(f"Shortened '{url2}' -> '{code2}'")

    print(f"\nResolving '{code1}': {shortener.resolve(code1)}")
    print(f"Resolving '{code2}': {shortener.resolve(code2)}")

    print("\nWaiting 2 seconds for code2 to expire...")
    time.sleep(2)

    try:
        shortener.resolve(code2)
    except KeyError as e:
        print(f"Expected error: {e}")

    print(f"Resolving '{code1}' (still valid): {shortener.resolve(code1)}")

    code3 = shortener.shorten("https://no-expiry.example.com", ttl=0)
    print(f"\nNo-expiry code: '{code3}' -> {shortener.resolve(code3)}")

    shortener.delete(code3)
    try:
        shortener.resolve(code3)
    except KeyError as e:
        print(f"Expected error after delete: {e}")

    code4 = shortener.shorten("https://expire-soon.example.com", ttl=1)
    time.sleep(2)
    purged = shortener.purge_expired()
    print(f"\nPurged {purged} expired entries.")


if __name__ == "__main__":
    main()
