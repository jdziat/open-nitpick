import re
import sys
from collections import Counter


def count_words(text: str) -> Counter:
    """Normalise text and count word frequencies."""
    text = text.lower()
    words = re.findall(r"[a-z']+", text)
    # Strip leading/trailing apostrophes from each word
    words = [word.strip("'") for word in words]
    # Remove empty strings
    words = [word for word in words if word]
    return Counter(words)


def top_k_words(text: str, k: int) -> list[tuple[str, int]]:
    """Return the top K words with their counts."""
    counter = count_words(text)
    return counter.most_common(k)


def read_text_from_file(filepath: str) -> str:
    """Read text from a file."""
    with open(filepath, "r", encoding="utf-8") as f:
        return f.read()


def display_results(results: list[tuple[str, int]], k: int) -> None:
    """Display the top K words and their counts."""
    print(f"\n{'Rank':<6} {'Word':<20} {'Count':<10}")
    print("-" * 36)
    for rank, (word, count) in enumerate(results, start=1):
        print(f"{rank:<6} {word:<20} {count:<10}")


def main() -> None:
    import argparse

    parser = argparse.ArgumentParser(
        description="Word-frequency counter: find the top K words in a text."
    )
    parser.add_argument(
        "source",
        nargs="?",
        help="Path to a text file. If omitted, reads from stdin.",
    )
    parser.add_argument(
        "-k",
        "--top",
        type=int,
        default=10,
        help="Number of top words to display (default: 10).",
    )
    args = parser.parse_args()

    if args.source:
        try:
            text = read_text_from_file(args.source)
        except FileNotFoundError:
            print(f"Error: file '{args.source}' not found.", file=sys.stderr)
            sys.exit(1)
        except OSError as exc:
            print(f"Error reading file: {exc}", file=sys.stderr)
            sys.exit(1)
    else:
        print("Reading from stdin (press Ctrl+D / Ctrl+Z to finish):")
        text = sys.stdin.read()

    k = args.top
    if k <= 0:
        print("Error: -k/--top must be a positive integer.", file=sys.stderr)
        sys.exit(1)

    results = top_k_words(text, k)

    if not results:
        print("No words found in the input.")
        return

    actual_k = len(results)
    print(f"\nTop {actual_k} word(s):")
    display_results(results, actual_k)


if __name__ == "__main__":
    main()
