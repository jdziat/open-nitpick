import * as fs from "fs";
import * as path from "path";

// ---------------------------------------------------------------------------
// Types
// ---------------------------------------------------------------------------

export interface AppConfig {
  // Server
  host: string;
  port: number;

  // Database
  dbHost: string;
  dbPort: number;
  dbName: string;
  dbUser: string;
  dbPassword: string;

  // Application
  logLevel: "debug" | "info" | "warn" | "error";
  nodeEnv: "development" | "production" | "test";
  secretKey: string;

  // Optional
  corsOrigin?: string;
  maxRequestBodySize?: string;
}

// Keys that MUST be present (non-empty) in the final resolved config.
const REQUIRED_KEYS: (keyof AppConfig)[] = [
  "host",
  "port",
  "dbHost",
  "dbPort",
  "dbName",
  "dbUser",
  "dbPassword",
  "secretKey",
];

// ---------------------------------------------------------------------------
// Defaults
// ---------------------------------------------------------------------------

const DEFAULTS: AppConfig = {
  host: "0.0.0.0",
  port: 3000,
  dbHost: "localhost",
  dbPort: 5432,
  dbName: "app_db",
  dbUser: "postgres",
  dbPassword: "",
  logLevel: "info",
  nodeEnv: "development",
  secretKey: "",
  corsOrigin: "*",
  maxRequestBodySize: "1mb",
};

// ---------------------------------------------------------------------------
// Environment-variable mapping
// Each entry maps an env-var name → config key (+ optional transform).
// ---------------------------------------------------------------------------

interface EnvMapping {
  envVar: string;
  key: keyof AppConfig;
  transform?: (raw: string) => unknown;
}

const ENV_MAPPINGS: EnvMapping[] = [
  { envVar: "HOST", key: "host" },
  { envVar: "PORT", key: "port", transform: (v) => parseInt(v, 10) },
  { envVar: "DB_HOST", key: "dbHost" },
  { envVar: "DB_PORT", key: "dbPort", transform: (v) => parseInt(v, 10) },
  { envVar: "DB_NAME", key: "dbName" },
  { envVar: "DB_USER", key: "dbUser" },
  { envVar: "DB_PASSWORD", key: "dbPassword" },
  { envVar: "LOG_LEVEL", key: "logLevel" },
  { envVar: "NODE_ENV", key: "nodeEnv" },
  { envVar: "SECRET_KEY", key: "secretKey" },
  { envVar: "CORS_ORIGIN", key: "corsOrigin" },
  { envVar: "MAX_REQUEST_BODY_SIZE", key: "maxRequestBodySize" },
];

// ---------------------------------------------------------------------------
// Helpers
// ---------------------------------------------------------------------------

function isValidLogLevel(v: unknown): v is AppConfig["logLevel"] {
  return ["debug", "info", "warn", "error"].includes(v as string);
}

function isValidNodeEnv(v: unknown): v is AppConfig["nodeEnv"] {
  return ["development", "production", "test"].includes(v as string);
}

function mergeFromEnv(config: AppConfig): AppConfig {
  const result: AppConfig = { ...config };

  for (const mapping of ENV_MAPPINGS) {
    const raw = process.env[mapping.envVar];
    if (raw !== undefined && raw.trim() !== "") {
      const value = mapping.transform ? mapping.transform(raw) : raw;
      // eslint-disable-next-line @typescript-eslint/no-explicit-any
      (result as any)[mapping.key] = value;
    }
  }

  return result;
}

function loadJsonFile(filePath: string): Partial<AppConfig> {
  const resolved = path.resolve(filePath);

  if (!fs.existsSync(resolved)) {
    console.warn(`[config] File not found: ${resolved}. Using defaults only.`);
    return {};
  }

  const raw = fs.readFileSync(resolved, "utf-8");

  let parsed: unknown;
  try {
    parsed = JSON.parse(raw);
  } catch (err) {
    throw new Error(`[config] Failed to parse JSON from ${resolved}: ${(err as Error).message}`);
  }

  if (typeof parsed !== "object" || parsed === null || Array.isArray(parsed)) {
    throw new Error(`[config] Expected a JSON object in ${resolved}.`);
  }

  return parsed as Partial<AppConfig>;
}

function validateConfig(config: AppConfig): void {
  const errors: string[] = [];

  // Required keys must be non-empty.
  for (const key of REQUIRED_KEYS) {
    const value = config[key];
    if (value === undefined || value === null || value === "") {
      errors.push(`Required config key "${key}" is missing or empty.`);
    }
  }

  // Numeric sanity checks.
  if (!Number.isInteger(config.port) || config.port < 1 || config.port > 65535) {
    errors.push(`"port" must be an integer between 1 and 65535 (got ${config.port}).`);
  }

  if (!Number.isInteger(config.dbPort) || config.dbPort < 1 || config.dbPort > 65535) {
    errors.push(`"dbPort" must be an integer between 1 and 65535 (got ${config.dbPort}).`);
  }

  // Enum checks.
  if (!isValidLogLevel(config.logLevel)) {
    errors.push(`"logLevel" must be one of debug|info|warn|error (got "${config.logLevel}").`);
  }

  if (!isValidNodeEnv(config.nodeEnv)) {
    errors.push(
      `"nodeEnv" must be one of development|production|test (got "${config.nodeEnv}").`
    );
  }

  if (errors.length > 0) {
    throw new Error(`[config] Validation failed:\n  - ${errors.join("\n  - ")}`);
  }
}

// ---------------------------------------------------------------------------
// Public API
// ---------------------------------------------------------------------------

export interface LoadConfigOptions {
  /** Path to the JSON config file. Defaults to "config.json". */
  filePath?: string;
  /** When true, validation errors are thrown. Defaults to true. */
  validate?: boolean;
}

/**
 * Loads configuration with the following priority (highest wins):
 *   1. Environment variables
 *   2. JSON config file values
 *   3. Built-in defaults
 */
export function loadConfig(options: LoadConfigOptions = {}): AppConfig {
  const { filePath = "config.json", validate = true } = options;

  // 1. Start with defaults.
  let config: AppConfig = { ...DEFAULTS };

  // 2. Overlay with JSON file values.
  const fileValues = loadJsonFile(filePath);
  config = { ...config, ...fileValues } as AppConfig;

  // 3. Overlay with environment variables.
  config = mergeFromEnv(config);

  // 4. Validate.
  if (validate) {
    validateConfig(config);
  }

  return config;
}

// ---------------------------------------------------------------------------
// CLI entry-point (node config-loader.ts)
// ---------------------------------------------------------------------------

if (require.main === module) {
  try {
    const configFilePath = process.argv[2] ?? "config.json";
    const config = loadConfig({ filePath: configFilePath });
    console.log("[config] Successfully loaded configuration:");
    console.log(
      JSON.stringify(
        {
          ...config,
          // Mask sensitive fields in output.
          dbPassword: config.dbPassword ? "***" : "(empty)",
          secretKey: config.secretKey ? "***" : "(empty)",
        },
        null,
        2
      )
    );
  } catch (err) {
    console.error((err as Error).message);
    process.exit(1);
  }
}
