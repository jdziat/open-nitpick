import * as fs from "fs";
import * as readline from "readline";

interface WordFrequency {
  word: string;
  count: number;
}

function normalizeText(text: string): string[] {
  return text
    .toLowerCase()
    .replace(/[^a-z0-9\s'-]/g, " ")
    .replace(/\s+/g, " ")
    .trim()
    .split(" ")
    .map((word) => word.replace(/^['-]+|['-]+$/g, ""))
    .filter((word) => word.length > 0);
}

function countWordFrequencies(words: string[]): Map<string, number> {
  const frequencyMap = new Map<string, number>();

  for (const word of words) {
    if (word) {
      frequencyMap.set(word, (frequencyMap.get(word) ?? 0) + 1);
    }
  }

  return frequencyMap;
}

function getTopKWords(
  frequencyMap: Map<string, number>,
  k: number
): WordFrequency[] {
  return Array.from(frequencyMap.entries())
    .map(([word, count]) => ({ word, count }))
    .sort((a, b) => b.count - a.count || a.word.localeCompare(b.word))
    .slice(0, k);
}

function processText(text: string, k: number): WordFrequency[] {
  const words = normalizeText(text);
  const frequencyMap = countWordFrequencies(words);
  return getTopKWords(frequencyMap, k);
}

async function readFromStdin(): Promise<string> {
  const rl = readline.createInterface({
    input: process.stdin,
    crlfDelay: Infinity,
  });

  const lines: string[] = [];

  for await (const line of rl) {
    lines.push(line);
  }

  return lines.join("\n");
}

async function main(): Promise<void> {
  const args = process.argv.slice(2);

  let k = 10;
  let inputFile: string | null = null;

  for (let i = 0; i < args.length; i++) {
    if (args[i] === "--top" || args[i] === "-k") {
      const parsed = parseInt(args[i + 1], 10);
      if (!isNaN(parsed) && parsed > 0) {
        k = parsed;
        i++;
      } else {
        console.error(`Invalid value for ${args[i]}: ${args[i + 1]}`);
        process.exit(1);
      }
    } else if (args[i] === "--file" || args[i] === "-f") {
      inputFile = args[i + 1];
      i++;
    } else if (args[i] === "--help" || args[i] === "-h") {
      console.log("Usage: ts-node solution.ts [options]");
      console.log("Options:");
      console.log("  --top, -k <number>   Number of top words to display (default: 10)");
      console.log("  --file, -f <path>    Input file path (default: stdin)");
      console.log("  --help, -h           Show this help message");
      process.exit(0);
    }
  }

  let text: string;

  if (inputFile) {
    if (!fs.existsSync(inputFile)) {
      console.error(`File not found: ${inputFile}`);
      process.exit(1);
    }
    text = fs.readFileSync(inputFile, "utf-8");
  } else {
    if (process.stdin.isTTY) {
      const sampleText = `
        The quick brown fox jumps over the lazy dog.
        The dog barked at the fox, and the fox ran away quickly.
        A quick brown dog outpaces a lazy fox. The fox was not amused.
        Dogs and foxes are both quick animals. The lazy dog slept all day.
        Over the hills and far away, the fox made its escape.
      `;
      console.log("No input provided. Using sample text:\n");
      console.log(sampleText.trim());
      console.log("\n" + "=".repeat(50) + "\n");
      text = sampleText;
    } else {
      text = await readFromStdin();
    }
  }

  const topWords = processText(text, k);

  if (topWords.length === 0) {
    console.log("No words found in the input text.");
    return;
  }

  console.log(`Top ${Math.min(k, topWords.length)} words by frequency:\n`);

  const maxWordLength = Math.max(...topWords.map((wf) => wf.word.length));
  const maxCountLength = Math.max(
    ...topWords.map((wf) => wf.count.toString().length)
  );

  const header = `${"Rank".padEnd(6)}${"Word".padEnd(maxWordLength + 4)}${"Count".padStart(maxCountLength)}`;
  const separator = "-".repeat(header.length);

  console.log(header);
  console.log(separator);

  topWords.forEach((wf, index) => {
    const rank = `${index + 1}.`.padEnd(6);
    const word = wf.word.padEnd(maxWordLength + 4);
    const count = wf.count.toString().padStart(maxCountLength);
    console.log(`${rank}${word}${count}`);
  });

  console.log(separator);

  const totalWords = Array.from(
    countWordFrequencies(normalizeText(text)).values()
  ).reduce((sum, count) => sum + count, 0);
  const uniqueWords = countWordFrequencies(normalizeText(text)).size;

  console.log(`\nTotal words: ${totalWords}`);
  console.log(`Unique words: ${uniqueWords}`);
}

main().catch((error) => {
  console.error("Error:", error.message);
  process.exit(1);
});
