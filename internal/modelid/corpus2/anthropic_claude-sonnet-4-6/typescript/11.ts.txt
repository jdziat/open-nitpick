type MatrixData = number[][];

class Matrix {
  private data: MatrixData;
  public readonly rows: number;
  public readonly cols: number;

  constructor(data: MatrixData) {
    if (data.length === 0) {
      throw new Error("Matrix must have at least one row.");
    }
    const cols = data[0].length;
    if (cols === 0) {
      throw new Error("Matrix must have at least one column.");
    }
    for (let i = 1; i < data.length; i++) {
      if (data[i].length !== cols) {
        throw new Error(
          `All rows must have the same number of columns. Row 0 has ${cols} columns, but row ${i} has ${data[i].length} columns.`
        );
      }
    }
    this.data = data.map((row) => [...row]);
    this.rows = data.length;
    this.cols = cols;
  }

  get(row: number, col: number): number {
    if (row < 0 || row >= this.rows || col < 0 || col >= this.cols) {
      throw new Error(
        `Index out of bounds: (${row}, ${col}) for matrix of size ${this.rows}x${this.cols}.`
      );
    }
    return this.data[row][col];
  }

  toArray(): MatrixData {
    return this.data.map((row) => [...row]);
  }

  multiply(other: Matrix): Matrix {
    if (this.cols !== other.rows) {
      throw new Error(
        `Shape mismatch for multiplication: (${this.rows}x${this.cols}) * (${other.rows}x${other.cols}). Inner dimensions must match.`
      );
    }
    const result: MatrixData = [];
    for (let i = 0; i < this.rows; i++) {
      result[i] = [];
      for (let j = 0; j < other.cols; j++) {
        let sum = 0;
        for (let k = 0; k < this.cols; k++) {
          sum += this.data[i][k] * other.data[k][j];
        }
        result[i][j] = sum;
      }
    }
    return new Matrix(result);
  }

  transpose(): Matrix {
    const result: MatrixData = [];
    for (let j = 0; j < this.cols; j++) {
      result[j] = [];
      for (let i = 0; i < this.rows; i++) {
        result[j][i] = this.data[i][j];
      }
    }
    return new Matrix(result);
  }

  determinant(): number {
    if (this.rows !== this.cols) {
      throw new Error(
        `Determinant is only defined for square matrices. Got ${this.rows}x${this.cols}.`
      );
    }
    return computeDeterminant(this.data, this.rows);
  }

  toString(): string {
    const rows = this.data.map((row) => `[ ${row.join(", ")} ]`);
    return rows.join("\n");
  }

  static identity(size: number): Matrix {
    if (size <= 0) {
      throw new Error("Identity matrix size must be positive.");
    }
    const data: MatrixData = [];
    for (let i = 0; i < size; i++) {
      data[i] = [];
      for (let j = 0; j < size; j++) {
        data[i][j] = i === j ? 1 : 0;
      }
    }
    return new Matrix(data);
  }

  static zeros(rows: number, cols: number): Matrix {
    if (rows <= 0 || cols <= 0) {
      throw new Error("Matrix dimensions must be positive.");
    }
    const data: MatrixData = [];
    for (let i = 0; i < rows; i++) {
      data[i] = new Array(cols).fill(0);
    }
    return new Matrix(data);
  }
}

function computeDeterminant(data: MatrixData, n: number): number {
  if (n === 1) {
    return data[0][0];
  }
  if (n === 2) {
    return data[0][0] * data[1][1] - data[0][1] * data[1][0];
  }

  // LU decomposition for larger matrices
  // Using Gaussian elimination with partial pivoting
  const mat: number[][] = data.map((row) => [...row]);
  let det = 1;
  let swaps = 0;

  for (let col = 0; col < n; col++) {
    // Find pivot
    let maxRow = col;
    let maxVal = Math.abs(mat[col][col]);
    for (let row = col + 1; row < n; row++) {
      if (Math.abs(mat[row][col]) > maxVal) {
        maxVal = Math.abs(mat[row][col]);
        maxRow = row;
      }
    }

    if (maxVal === 0) {
      return 0;
    }

    if (maxRow !== col) {
      [mat[col], mat[maxRow]] = [mat[maxRow], mat[col]];
      swaps++;
    }

    det *= mat[col][col];

    for (let row = col + 1; row < n; row++) {
      const factor = mat[row][col] / mat[col][col];
      for (let k = col; k < n; k++) {
        mat[row][k] -= factor * mat[col][k];
      }
    }
  }

  if (swaps % 2 !== 0) {
    det = -det;
  }

  return det;
}

// ---- Demo / Tests ----

function assert(condition: boolean, message: string): void {
  if (!condition) {
    throw new Error(`Assertion failed: ${message}`);
  }
}

function approxEqual(a: number, b: number, eps = 1e-9): boolean {
  return Math.abs(a - b) < eps;
}

// Test 1: Basic multiplication
const A = new Matrix([
  [1, 2],
  [3, 4],
]);
const B = new Matrix([
  [5, 6],
  [7, 8],
]);
const AB = A.multiply(B);
assert(AB.get(0, 0) === 19, "AB[0][0] should be 19");
assert(AB.get(0, 1) === 22, "AB[0][1] should be 22");
assert(AB.get(1, 0) === 43, "AB[1][0] should be 43");
assert(AB.get(1, 1) === 50, "AB[1][1] should be 50");
console.log("Test 1 passed: Basic 2x2 multiplication");

// Test 2: Transpose
const C = new Matrix([
  [1, 2, 3],
  [4, 5, 6],
]);
const CT = C.transpose();
assert(CT.rows === 3 && CT.cols === 2, "Transposed shape should be 3x2");
assert(CT.get(0, 0) === 1, "CT[0][0] should be 1");
assert(CT.get(1, 0) === 2, "CT[1][0] should be 2");
assert(CT.get(2, 0) === 3, "CT[2][0] should be 3");
assert(CT.get(0, 1) === 4, "CT[0][1] should be 4");
assert(CT.get(1, 1) === 5, "CT[1][1] should be 5");
assert(CT.get(2, 1) === 6, "CT[2][1] should be 6");
console.log("Test 2 passed: Transpose of 2x3 matrix");

// Test 3: Determinant 2x2
const det2 = A.determinant();
assert(approxEqual(det2, -2), `Determinant of A should be -2, got ${det2}`);
console.log("Test 3 passed: 2x2 determinant");

// Test 4: Determinant 3x3
const D = new Matrix([
  [1, 2, 3],
  [4, 5, 6],
  [7, 8, 10],
]);
const det3 = D.determinant();
assert(approxEqual(det3, -3), `Determinant of D should be -3, got ${det3}`);
console.log("Test 4 passed: 3x3 determinant");

// Test 5: Determinant of singular matrix
const singular = new Matrix([
  [1, 2, 3],
  [4, 5, 6],
  [7, 8, 9],
]);
const detSingular = singular.determinant();
assert(
  approxEqual(detSingular, 0),
  `Determinant of singular matrix should be 0, got ${detSingular}`
);
console.log("Test 5 passed: Singular matrix determinant is 0");

// Test 6: Identity matrix
const I3 = Matrix.identity(3);
const AI = A.multiply(Matrix.identity(2));
assert(AI.get(0, 0) === A.get(0, 0), "A * I should equal A");
assert(AI.get(1, 1) === A.get(1, 1), "A * I should equal A");
console.log("Test 6 passed: Identity matrix multiplication");

// Test 7: Shape mismatch error for multiplication
try {
  const E = new Matrix([[1, 2, 3]]);
  const F = new Matrix([[1, 2]]);
  E.multiply(F);
  assert(false, "Should have thrown a shape mismatch error");

  );

}

// Test 9: Larger matrix multiplication
const G = new Matrix([
  [1, 0, 0],
  [0, 1, 0],
  [0, 0, 1],
]);
const H = new Matrix([
  [1, 2, 3],
  [4, 5, 6],
  [7, 8, 9],
]);
const GH = G.multiply(H);
for (let i = 0; i < 3; i++) {
  for (let j = 0; j < 3; j++) {
    assert(GH.get(i, j) === H.get(i, j), `GH[${i}][${j}] should equal H[${i}][${j}]`);
  }
}
console.log("Test 9 passed: Identity * H = H for 3x3");

// Test 10: 4x4 determinant
