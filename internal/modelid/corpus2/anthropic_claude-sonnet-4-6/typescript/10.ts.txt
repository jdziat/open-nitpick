import crypto from "crypto";

interface ShortenedURL {
  originalUrl: string;
  expiresAt: number;
}

const store = new Map<string, ShortenedURL>();

function generateCode(length = 6): string {
  return crypto.randomBytes(length).toString("base64url").slice(0, length);
}

function shorten(originalUrl: string, ttlMs = 24 * 60 * 60 * 1000): string {
  let code: string;
  do {
    code = generateCode();
  } while (store.has(code));

  store.set(code, {
    originalUrl,
    expiresAt: Date.now() + ttlMs,
  });

  return code;
}

function resolve(code: string): string | null {
  const entry = store.get(code);

  if (!entry) {
    return null;
  }

  if (Date.now() > entry.expiresAt) {
    store.delete(code);
    return null;
  }

  return entry.originalUrl;
}

function deleteCode(code: string): boolean {
  return store.delete(code);
}

function purgeExpired(): number {
  const now = Date.now();
  let count = 0;
  for (const [code, entry] of store.entries()) {
    if (now > entry.expiresAt) {
      store.delete(code);
      count++;
    }
  }
  return count;
}

// --- Demo ---

const url1 = "https://www.example.com/some/very/long/path?query=1&foo=bar";
const url2 = "https://openai.com/research/gpt-4";

const code1 = shorten(url1);
const code2 = shorten(url2, 5000); // expires in 5 seconds
const code3 = shorten("https://typescript.org", 1); // expires almost immediately

console.log(`Shortened "${url1}" -> code: ${code1}`);
console.log(`Shortened "${url2}" -> code: ${code2} (5s TTL)`);
console.log(`Shortened "https://typescript.org" -> code: ${code3} (1ms TTL)`);

console.log("\n--- Resolving codes ---");
console.log(`Resolve ${code1}: ${resolve(code1)}`);
console.log(`Resolve ${code2}: ${resolve(code2)}`);

// Wait for code3 to expire
setTimeout(() => {
  console.log("\n--- After 1ms TTL expiry ---");
  console.log(`Resolve ${code3} (should be null): ${resolve(code3)}`);
}, 10);

// Wait for code2 to expire
setTimeout(() => {
  console.log("\n--- After 5s TTL expiry ---");
  console.log(`Resolve ${code2} (should be null): ${resolve(code2)}`);
  console.log(`Resolve ${code1} (should still work): ${resolve(code1)}`);

  console.log("\n--- Purging expired entries ---");
  const purged = purgeExpired();
  console.log(`Purged ${purged} expired entries`);
  console.log(`Store size after purge: ${store.size}`);

  console.log("\n--- Deleting a code manually ---");
  const deleted = deleteCode(code1);
  console.log(`Deleted ${code1}: ${deleted}`);
  console.log(`Resolve ${code1} after deletion (should be null): ${resolve(code1)}`);
  console.log(`Store size after deletion: ${store.size}`);
}, 6000);
