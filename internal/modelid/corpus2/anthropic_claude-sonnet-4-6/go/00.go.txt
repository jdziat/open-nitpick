package main

import (
	"container/list"
	"fmt"
)

type entry struct {
	key   int
	value int
}

type LRUCache struct {
	capacity int
	list     *list.List
	items    map[int]*list.Element
}

func NewLRUCache(capacity int) *LRUCache {
	return &LRUCache{
		capacity: capacity,
		list:     list.New(),
		items:    make(map[int]*list.Element),
	}
}

func (c *LRUCache) Get(key int) (int, bool) {
	if elem, ok := c.items[key]; ok {
		c.list.MoveToFront(elem)
		return elem.Value.(*entry).value, true
	}
	return -1, false
}

func (c *LRUCache) Put(key, value int) {
	if elem, ok := c.items[key]; ok {
		c.list.MoveToFront(elem)
		elem.Value.(*entry).value = value
		return
	}

	if c.list.Len() == c.capacity {
		back := c.list.Back()
		if back != nil {
			c.list.Remove(back)
			delete(c.items, back.Value.(*entry).key)
		}
	}

	e := &entry{key: key, value: value}
	elem := c.list.PushFront(e)
	c.items[key] = elem
}

func (c *LRUCache) Len() int {
	return c.list.Len()
}

func main() {
	cache := NewLRUCache(3)

	cache.Put(1, 10)
	cache.Put(2, 20)
	cache.Put(3, 30)

	if v, ok := cache.Get(1); ok {
		fmt.Printf("Get(1) = %d\n", v)
	}

	// This should evict key 2 (least recently used)
	cache.Put(4, 40)

	if _, ok := cache.Get(2); !ok {
		fmt.Println("Get(2) = not found (evicted)")
	}

	if v, ok := cache.Get(3); ok {
		fmt.Printf("Get(3) = %d\n", v)
	}

	if v, ok := cache.Get(4); ok {
		fmt.Printf("Get(4) = %d\n", v)
	}

	// Update existing key
	cache.Put(1, 100)
	if v, ok := cache.Get(1); ok {
		fmt.Printf("Get(1) after update = %d\n", v)
	}

	fmt.Printf("Cache length: %d\n", cache.Len())
}
