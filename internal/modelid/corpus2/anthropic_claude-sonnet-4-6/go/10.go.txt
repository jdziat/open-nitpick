package main

import (
	"crypto/rand"
	"encoding/base64"
	"errors"
	"sync"
	"time"
)

// ErrNotFound is returned when a short code does not exist.
var ErrNotFound = errors.New("short code not found")

// ErrExpired is returned when a short code has expired.
var ErrExpired = errors.New("short code has expired")

// entry holds the original URL and its expiry time.
type entry struct {
	originalURL string
	expiresAt   time.Time
}

// Shortener is a minimal in-memory URL shortener.
type Shortener struct {
	mu    sync.RWMutex
	store map[string]entry
}

// NewShortener creates and returns a new Shortener.
func NewShortener() *Shortener {
	s := &Shortener{
		store: make(map[string]entry),
	}
	go s.cleanup()
	return s
}

// generateCode produces a random URL-safe base64 short code of the given byte length.
func generateCode(byteLen int) (string, error) {
	b := make([]byte, byteLen)
	if _, err := rand.Read(b); err != nil {
		return "", err
	}
	return base64.RawURLEncoding.EncodeToString(b), nil
}

// Shorten stores the given URL with the specified TTL and returns a short code.
// If ttl is zero or negative, the entry never expires (well, expires in 100 years).
func (s *Shortener) Shorten(originalURL string, ttl time.Duration) (string, error) {
	if originalURL == "" {
		return "", errors.New("original URL must not be empty")
	}
	if ttl <= 0 {
		ttl = 100 * 365 * 24 * time.Hour
	}

	var code string
	var err error

	s.mu.Lock()
	defer s.mu.Unlock()

	// Retry until we get a unique code (collision is extremely unlikely).
	for {
		code, err = generateCode(6)
		if err != nil {
			return "", err
		}
		if _, exists := s.store[code]; !exists {
			break
		}
	}

	s.store[code] = entry{
		originalURL: originalURL,
		expiresAt:   time.Now().Add(ttl),
	}
	return code, nil
}

// Resolve returns the original URL for the given short code.
// Returns ErrNotFound if the code is unknown and ErrExpired if it has expired.
func (s *Shortener) Resolve(code string) (string, error) {
	s.mu.RLock()
	e, ok := s.store[code]
	s.mu.RUnlock()

	if !ok {
		return "", ErrNotFound
	}
	if time.Now().After(e.expiresAt) {
		// Lazily delete the expired entry.
		s.mu.Lock()
		delete(s.store, code)
		s.mu.Unlock()
		return "", ErrExpired
	}
	return e.originalURL, nil
}

// Delete removes a short code from the store.
func (s *Shortener) Delete(code string) {
	s.mu.Lock()
	delete(s.store, code)
	s.mu.Unlock()
}

// cleanup periodically removes expired entries from the store.
func (s *Shortener) cleanup() {
	ticker := time.NewTicker(5 * time.Minute)
	defer ticker.Stop()
	for range ticker.C {
		now := time.Now()
		s.mu.Lock()
		for code, e := range s.store {
			if now.After(e.expiresAt) {
				delete(s.store, code)
			}
		}
		s.mu.Unlock()
	}
}

// ---- simple demonstration in main ----

import (
	"fmt"
	"log"
)

func main() {
	s := NewShortener()

	// Shorten a URL with a 2-second TTL.
	code, err := s.Shorten("https://www.example.com/some/long/path?query=1", 2*time.Second)
	if err != nil {
		log.Fatal(err)
	}
	fmt.Printf("Short code: %s\n", code)

	// Resolve immediately – should succeed.
	url, err := s.Resolve(code)
	if err != nil {
		log.Fatal(err)
	}
	fmt.Printf("Resolved URL: %s\n", url)

	// Wait for the entry to expire.
	fmt.Println("Waiting 3 seconds for expiry...")
	time.Sleep(3 * time.Second)

	// Resolve after expiry – should return ErrExpired.
	_, err = s.Resolve(code)
	if errors.Is(err, ErrExpired) {
		fmt.Println("Entry has expired as expected.")
	} else {
		fmt.Printf("Unexpected result: %v\n", err)
	}

	// Shorten another URL with a long TTL and then delete it.
	code2, err := s.Shorten("https://www.example.com/another/page", 10*time.Minute)
	if err != nil {
		log.Fatal(err)
	}
	fmt.Printf("Short code 2: %s\n", code2)
	s.Delete(code2)
	_, err = s.Resolve(code2)
	if errors.Is(err, ErrNotFound) {
		fmt.Println("Deleted entry not found as expected.")
	}
}
