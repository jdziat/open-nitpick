package main

import (
	"fmt"
	"strings"
	"unicode"
)

// Slugify converts a string into a URL-friendly slug.
// It lowercases the text, replaces runs of non-alphanumeric characters
// (including non-letter Unicode) with single hyphens, and trims leading/trailing hyphens.
func Slugify(s string) string {
	var builder strings.Builder
	inNonAlnum := false

	for _, r := range s {
		if unicode.IsLetter(r) || unicode.IsDigit(r) {
			inNonAlnum = false
			builder.WriteRune(unicode.ToLower(r))
		} else {
			if !inNonAlnum {
				inNonAlnum = true
				builder.WriteRune('-')
			}
		}
	}

	result := builder.String()
	result = strings.Trim(result, "-")
	return result
}

func main() {
	examples := []string{
		"Hello, World!",
		"  leading and trailing spaces  ",
		"This is a Test String",
		"Héllo Wörld",
		"日本語テスト",
		"multiple---hyphens",
		"Special @#$% Characters!!!",
		"Ünïcödé ïs grëât",
		"",
		"---",
		"foo_bar_baz",
		"CamelCaseString",
		"numbers 1 2 3",
	}

	for _, e := range examples {
		fmt.Printf("Input:  %q\nSlug:   %q\n\n", e, Slugify(e))
	}
}
