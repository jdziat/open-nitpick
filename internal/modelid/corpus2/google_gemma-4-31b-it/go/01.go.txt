package main

import (
	"bufio"
	"io"
	"strings"
)

type CSVParser struct {
	Delimiter rune
}

func NewCSVParser(delimiter rune) *CSVParser {
	return &CSVParser{Delimiter: delimiter}
}

func (p *CSVParser) Parse(r io.Reader) ([][]string, error) {
	var rows [][]string
	reader := bufio.NewReader(r)
	
	var currentRow []string
	var currentField strings.Builder
	inQuotes := false
	
	for {
		rn, _, err := reader.ReadRune()
		if err != nil {
			if err == io.EOF {
				if inQuotes {
					// Unclosed quote at EOF, treat as end of field/row
				}
				if currentField.Len() > 0 || len(currentRow) > 0 {
					currentRow = append(currentRow, currentField.String())
					rows = append(rows, currentRow)
				}
				break
			}
			return nil, err
		}

		if inQuotes {
			if rn == '"' {
				nextRune, _, err := reader.ReadRune()
				if err == nil && nextRune == '"' {
					currentField.WriteRune('"')
				} else {
					inQuotes = false
					if err == nil {
						reader.UnreadRune()
					}
				}
			} else {
				currentField.WriteRune(rn)
			}
		} else {
			switch rn {
			case '"':
				inQuotes = true
			case p.Delimiter:
				currentRow = append(currentRow, currentField.String())
				currentField.Reset()
			case '\n', '\r':
				currentRow = append(currentRow, currentField.String())
				rows = append(rows, currentRow)
				currentRow = nil
				currentField.Reset()
				if rn == '\r' {
					nextRune, _, _ := reader.ReadRune()
					if nextRune != '\n' {
						reader.UnreadRune()
					}
				}
			default:
				currentField.WriteRune(rn)
			}
		}
	}

	return rows, nil
}
