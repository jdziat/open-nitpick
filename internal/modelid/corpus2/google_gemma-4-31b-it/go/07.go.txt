package main

import (
	"encoding/json"
	"fmt"
	"os"
	"reflect"
	"strings"
)

type Config struct {
	Port    int    `json:"port" env:"APP_PORT" default:"8080"`
	APIKey  string `json:"api_key" env:"APP_API_KEY" required:"true"`
	Timeout int    `json:"timeout" env:"APP_TIMEOUT" default:"30"`
}

func LoadConfig(path string) (*Config, error) {
	cfg := &Config{}

	// 1. Set Defaults via reflection
	t := reflect.TypeOf(*cfg)
	v := reflect.ValueOf(cfg).Elem()
	for i := 0; i < t.NumField(); i++ {
		field := t.Field(i)
		def := field.Tag.Get("default")
		if def != "" {
			f := v.Field(i)
			switch f.Kind() {
			case reflect.String:
				f.SetString(def)
			case reflect.Int:
				var val int
				fmt.Sscanf(def, "%d", &val)
				f.SetInt(int64(val))
			}
		}
	}

	// 2. Load JSON file
	file, err := os.ReadFile(path)
	if err == nil {
		if err := json.Unmarshal(file, cfg); err != nil {
			return nil, fmt.Errorf("json error: %w", err)
		}
	}

	// 3. Environment Variable Overrides
	for i := 0; i < t.NumField(); i++ {
		field := t.Field(i)
		envKey := field.Tag.Get("env")
		if envKey == "" {
			continue
		}
		if val, exists := os.LookupEnv(envKey); exists {
			f := v.Field(i)
			switch f.Kind() {
			case reflect.String:
				f.SetString(val)
			case reflect.Int:
				var intVal int
				fmt.Sscanf(val, "%d", &intVal)
				f.SetInt(int64(intVal))
			}
		}
	}

	// 4. Validation
	for i := 0; i < t.NumField(); i++ {
		field := t.Field(i)
		if field.Tag.Get("required") == "true" {
			f := v.Field(i)
			if f.IsZero() {
				return nil, fmt.Errorf("missing required config: %s", field.Name)
			}
		}
	}

	return cfg, nil
}

func main() {
	// Setup dummy file for demonstration
	os.WriteFile("config.json", []byte(`{"port": 9000}`), 0644)
	defer os.Remove("config.json")

	// Set required env var
	os.Setenv("APP_API_KEY", "secret-123")

	cfg, err := LoadConfig("config.json")
	if err != nil {
		fmt.Printf("Error: %v\n", err)
		os.Exit(1)
	}

	fmt.Printf("Loaded Config: %+v\n", cfg)
}
