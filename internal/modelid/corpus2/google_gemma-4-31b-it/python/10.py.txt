import time
import string
import random

class URLShortener:
    def __init__(self):
        self.db = {}

    def shorten(self, url, expiry_seconds=3600):
        code = ''.join(random.choices(string.ascii_letters + string.digits, k=6))
        while code in self.db:
            code = ''.join(random.choices(string.ascii_letters + string.digits, k=6))
        
        expiry_time = time.time() + expiry_seconds
        self.db[code] = (url, expiry_time)
        return code

    def resolve(self, code):
        if code not in self.db:
            return None
        
        url, expiry_time = self.db[code]
        if time.time() > expiry_time:
            del self.db[code]
            return None
            
        return url

if __name__ == "__main__":
    shortener = URLShortener()
    
    # Test normal operation
    code = shortener.shorten("https://www.google.com")
    print(f"Short code: {code}")
    print(f"Resolved: {shortener.resolve(code)}")
    
    # Test expiry (1 second)
    exp_code = shortener.shorten("https://www.python.org", expiry_seconds=1)
    print(f"Expiring code: {exp_code}")
    time.sleep(2)
    print(f"Resolved after expiry: {shortener.resolve(exp_code)}")
