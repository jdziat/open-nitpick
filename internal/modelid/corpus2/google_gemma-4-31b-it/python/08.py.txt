import heapq

class PriorityQueue:
    class Wrapper:
        def __init__(self, item, cmp):
            self.item = item
            self.cmp = cmp

        def __lt__(self, other):
            # Returns True if self is "higher priority" than other
            # based on the caller-supplied comparison function.
            return self.cmp(self.item, other.item)

    def __init__(self, compare_func):
        """
        compare_func(a, b) should return True if 'a' has higher priority than 'b'.
        """
        self.compare_func = compare_func
        self.heap = []

    def push(self, item):
        heapq.heappush(self.heap, self.Wrapper(item, self.compare_func))

    def pop(self):
        if not self.heap:
            raise IndexError("pop from empty priority queue")
        return heapq.heappop(self.heap).item

    def peek(self):
        if not self.heap:
            raise IndexError("peek from empty priority queue")
        return self.heap[0].item

    def __len__(self):
        return len(self.heap)
