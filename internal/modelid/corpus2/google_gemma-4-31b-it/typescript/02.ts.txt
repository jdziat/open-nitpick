import { Lock } from 'async-lock';

export class TokenBucket {
    private capacity: number;
    private refillRatePerSecond: number;
    private tokens: number;
    private lastRefillTime: number;
    private lock: Lock;

    constructor(capacity: number, refillRatePerSecond: number) {
        this.capacity = capacity;
        this.refillRatePerSecond = refillRatePerSecond;
        this.tokens = capacity;
        this.lastRefillTime = Date.now();
        this.lock = new Lock();
    }

    private refill(): void {
        const now = Date.now();
        const elapsedMs = now - this.lastRefillTime;
        const refillAmount = (elapsedMs / 1000) * this.refillRatePerSecond;
        
        this.tokens = Math.min(this.capacity, this.tokens + refillAmount);
        this.lastRefillTime = now;
    }

    async tryConsume(tokensRequested: number = 1): Promise<boolean> {
        return this.lock.acquire('bucket', async () => {
            this.refill();

            if (this.tokens >= tokensRequested) {
                this.tokens -= tokensRequested;
                return true;
            }
            
            return false;
        });
    }

    async consume(tokensRequested: number = 1): Promise<void> {
        while (true) {
            const success = await this.tryConsume(tokensRequested);
            if (success) return;

            const waitTime = (tokensRequested - this.tokens) / this.refillRatePerSecond * 1000;
            await new Promise(resolve => setTimeout(resolve, waitTime));
        }
    }

    async getAvailableTokens(): Promise<number> {
        return this.lock.acquire('bucket', () => {
            this.refill();
            return this.tokens;
        });
    }
}

// Minimal async-lock implementation to ensure single-file portability
class Lock {
    private locked = false;
    private queue: (() => void)[] = [];

    async acquire<T>(key: string, callback: () => Promise<T> | T): Promise<T> {
        while (this.locked) {
            await new Promise<void>(resolve => this.queue.push(resolve));
        }
        this.locked = true;
        try {
            return await callback();
        } finally {
            this.locked = false;
            const next = this.queue.shift();
            if (next) next();
        }
    }
}
