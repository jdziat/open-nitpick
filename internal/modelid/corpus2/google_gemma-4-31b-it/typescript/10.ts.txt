interface ShortUrl {
    originalUrl: string;
    expiry: number;
}

export class UrlShortener {
    private storage: Map<string, ShortUrl> = new Map();

    public shorten(url: string, ttlMs: number): string {
        const code = Math.random().toString(36).substring(2, 8);
        const expiry = Date.now() + ttlMs;
        this.storage.set(code, { originalUrl: url, expiry });
        return code;
    }

    public resolve(code: string): string | null {
        const entry = this.storage.get(code);
        if (!entry) return null;

        if (Date.now() > entry.expiry) {
            this.storage.delete(code);
            return null;
        }

        return entry.originalUrl;
    }
}
