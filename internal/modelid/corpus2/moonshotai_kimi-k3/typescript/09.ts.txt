/**
 * date-range-iterator.ts
 *
 * A date-range iterator that yields each day between two dates inclusive,
 * with an option to skip weekends (Saturdays and Sundays).
 */

export interface DateRangeOptions {
  /** When true, Saturdays and Sundays are skipped. Defaults to false. */
  skipWeekends?: boolean;
}

const MILLIS_PER_DAY = 24 * 60 * 60 * 1000;

/**
 * Returns true if the given date falls on a Saturday or Sunday
 * (in the date's local time zone).
 */
export function isWeekend(date: Date): boolean {
  const day = date.getDay();
  return day === 0 || day === 6;
}

/**
 * Normalizes a date to midnight (start of day) in local time,
 * returning a new Date instance.
 */
function startOfDay(date: Date): Date {
  const d = new Date(date.getTime());
  d.setHours(0, 0, 0, 0);
  return d;
}

/**
 * Creates an iterator that yields each day between `start` and `end`,
 * inclusive of both endpoints.
 *
 * - Each yielded value is a new `Date` set to local midnight of that day.
 * - If `start` is after `end`, the range is empty and nothing is yielded.
 * - When `options.skipWeekends` is true, Saturdays and Sundays are omitted.
 *
 * @example
 * for (const day of dateRange(new Date('2024-01-01'), new Date('2024-01-07'), { skipWeekends: true })) {
 *   console.log(day.toDateString());
 * }
 */
export function* dateRange(
  start: Date,
  end: Date,
  options: DateRangeOptions = {}
): IterableIterator<Date> {
  const { skipWeekends = false } = options;

  const current = startOfDay(start);
  const last = startOfDay(end);

  if (current.getTime() > last.getTime()) {
    return;
  }

  while (current.getTime() <= last.getTime()) {
    if (!skipWeekends || !isWeekend(current)) {
      yield new Date(current.getTime());
    }
    current.setTime(current.getTime() + MILLIS_PER_DAY);
    // Guard against DST transitions shifting the time off midnight.
    current.setHours(0, 0, 0, 0);
  }
}

/**
 * A reusable iterable over a date range. Implements `Iterable<Date>` so it
 * can be consumed multiple times (each iteration starts fresh).
 *
 * @example
 * const range = new DateRange(new Date('2024-03-01'), new Date('2024-03-31'), { skipWeekends: true });
 * const days = [...range];
 */
export class DateRange implements Iterable<Date> {
  private readonly start: Date;
  private readonly end: Date;
  private readonly options: DateRangeOptions;

  constructor(start: Date, end: Date, options: DateRangeOptions = {}) {
    this.start = new Date(start.getTime());
    this.end = new Date(end.getTime());
    this.options = { ...options };
  }

  *[Symbol.iterator](): IterableIterator<Date> {
    yield* dateRange(this.start, this.end, this.options);
  }

  /** Returns all days in the range as an array. */
  toArray(): Date[] {
    return [...this];
  }
}
