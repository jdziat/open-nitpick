export interface CsvParseOptions {
  /** Field delimiter. Must be a single character. Defaults to ",". */
  delimiter?: string;
}

/**
 * Parses a CSV string into an array of rows, where each row is an array of
 * string fields.
 *
 * Supported features:
 *   - Fields quoted with double quotes ("like,this")
 *   - Escaped quotes inside quoted fields ("say ""hi""")
 *   - Line breaks (LF, CR, or CRLF) inside quoted fields
 *   - A configurable single-character delimiter
 *
 * Throws a SyntaxError on unterminated quoted fields and a RangeError for
 * invalid delimiters.
 */
export function parseCsv(input: string, options: CsvParseOptions = {}): string[][] {
  const delimiter = options.delimiter ?? ",";

  if (delimiter.length !== 1) {
    throw new RangeError(
      `Delimiter must be a single character, got ${JSON.stringify(delimiter)}`
    );
  }
  if (delimiter === '"' || delimiter === "\r" || delimiter === "\n") {
    throw new RangeError("Delimiter cannot be a quote or newline character");
  }

  const rows: string[][] = [];
  let currentRow: string[] = [];
  let currentField = "";
  let inQuotes = false;
  let rowStarted = false;

  const endField = (): void => {
    currentRow.push(currentField);
    currentField = "";
  };

  const endRow = (): void => {
    endField();
    rows.push(currentRow);
    currentRow = [];
    rowStarted = false;
  };

  let i = 0;
  while (i < input.length) {
    const ch = input[i];
    rowStarted = true;

    if (inQuotes) {
      if (ch === '"') {
        if (input[i + 1] === '"') {
          // Escaped quote: "" -> "
          currentField += '"';
          i += 2;
        } else {
          inQuotes = false;
          i += 1;
        }
      } else {
        currentField += ch;
        i += 1;
      }
      continue;
    }

    if (ch === '"' && currentField === "") {
      inQuotes = true;
      i += 1;
    } else if (ch === delimiter) {
      endField();
      i += 1;
    } else if (ch === "\n") {
      endRow();
      i += 1;
    } else if (ch === "\r") {
      endRow();
      i += input[i + 1] === "\n" ? 2 : 1;
    } else {
      currentField += ch;
      i += 1;
    }
  }

  if (inQuotes) {
    throw new SyntaxError("Unexpected end of input: unterminated quoted field");
  }

  // Flush the final row unless the input was empty or ended with a newline.
  if (rowStarted) {
    endRow();
  }

  return rows;
}
