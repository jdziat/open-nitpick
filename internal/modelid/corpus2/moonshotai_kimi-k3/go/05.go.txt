package main

import (
	"bufio"
	"flag"
	"fmt"
	"io"
	"os"
	"sort"
	"strings"
	"unicode"
)

// wordCount pairs a word with its frequency.
type wordCount struct {
	word  string
	count int
}

// tokenize splits text into normalized words: lowercased, with
// punctuation stripped. Apostrophes inside words are kept.
func tokenize(text string) []string {
	var words []string
	var b strings.Builder

	flush := func() {
		if b.Len() > 0 {
			words = append(words, b.String())
			b.Reset()
		}
	}

	for _, r := range text {
		switch {
		case unicode.IsLetter(r) || unicode.IsDigit(r):
			b.WriteRune(unicode.ToLower(r))
		case r == '\'' || r == '’':
			// Keep apostrophes only when they are inside a word.
			if b.Len() > 0 {
				b.WriteRune('\'')
			}
		default:
			flush()
		}
	}
	flush()

	// Remove trailing apostrophes that ended up at word edges.
	for i, w := range words {
		words[i] = strings.Trim(w, "'")
	}

	out := words[:0]
	for _, w := range words {
		if w != "" {
			out = append(out, w)
		}
	}
	return out
}

// countWords builds a frequency map from a token stream.
func countWords(words []string) map[string]int {
	freq := make(map[string]int, len(words))
	for _, w := range words {
		freq[w]++
	}
	return freq
}

// topK returns the k most frequent words, sorted by count descending
// and alphabetically for ties. If k <= 0 or exceeds the number of
// distinct words, all words are returned.
func topK(freq map[string]int, k int) []wordCount {
	pairs := make([]wordCount, 0, len(freq))
	for w, c := range freq {
		pairs = append(pairs, wordCount{w, c})
	}

	sort.Slice(pairs, func(i, j int) bool {
		if pairs[i].count != pairs[j].count {
			return pairs[i].count > pairs[j].count
		}
		return pairs[i].word < pairs[j].word
	})

	if k > 0 && k < len(pairs) {
		pairs = pairs[:k]
	}
	return pairs
}

func main() {
	k := flag.Int("k", 10, "number of top words to print (0 = all)")
	flag.Parse()

	var r io.Reader = os.Stdin
	if flag.NArg() > 0 {
		f, err := os.Open(flag.Arg(0))
		if err != nil {
			fmt.Fprintf(os.Stderr, "error: %v\n", err)
			os.Exit(1)
		}
		defer f.Close()
		r = f
	}

	data, err := io.ReadAll(bufio.NewReader(r))
	if err != nil {
		fmt.Fprintf(os.Stderr, "error reading input: %v\n", err)
		os.Exit(1)
	}

	freq := countWords(tokenize(string(data)))
	for _, wc := range topK(freq, *k) {
		fmt.Printf("%6d  %s\n", wc.count, wc.word)
	}
}
