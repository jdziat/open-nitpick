package matrix

import (
	"errors"
	"fmt"
	"math"
	"strings"
)

// ErrNotSquare is returned when an operation requires a square matrix.
var ErrNotSquare = errors.New("matrix: operation requires a square matrix")

// Matrix is a dense matrix of float64 values stored in row-major order.
type Matrix struct {
	rows, cols int
	data       []float64
}

// New creates a rows x cols matrix from values given in row-major order.
func New(rows, cols int, values ...float64) (Matrix, error) {
	if rows <= 0 || cols <= 0 {
		return Matrix{}, fmt.Errorf("matrix: invalid dimensions %dx%d", rows, cols)
	}
	if len(values) != rows*cols {
		return Matrix{}, fmt.Errorf("matrix: %d values cannot fill a %dx%d matrix", len(values), rows, cols)
	}
	data := make([]float64, len(values))
	copy(data, values)
	return Matrix{rows: rows, cols: cols, data: data}, nil
}

// Zeros creates a rows x cols matrix filled with zeros.
func Zeros(rows, cols int) (Matrix, error) {
	if rows <= 0 || cols <= 0 {
		return Matrix{}, fmt.Errorf("matrix: invalid dimensions %dx%d", rows, cols)
	}
	return Matrix{rows: rows, cols: cols, data: make([]float64, rows*cols)}, nil
}

// Identity creates the n x n identity matrix.
func Identity(n int) (Matrix, error) {
	m, err := Zeros(n, n)
	if err != nil {
		return Matrix{}, err
	}
	for i := 0; i < n; i++ {
		m.data[i*n+i] = 1
	}
	return m, nil
}

// Dims returns the number of rows and columns of m.
func (m Matrix) Dims() (rows, cols int) { return m.rows, m.cols }

// At returns the element at row r, column c.
func (m Matrix) At(r, c int) (float64, error) {
	if r < 0 || r >= m.rows || c < 0 || c >= m.cols {
		return 0, fmt.Errorf("matrix: index (%d,%d) out of bounds for %dx%d matrix", r, c, m.rows, m.cols)
	}
	return m.data[r*m.cols+c], nil
}

// Set assigns v to the element at row r, column c.
func (m *Matrix) Set(r, c int, v float64) error {
	if r < 0 || r >= m.rows || c < 0 || c >= m.cols {
		return fmt.Errorf("matrix: index (%d,%d) out of bounds for %dx%d matrix", r, c, m.rows, m.cols)
	}
	m.data[r*m.cols+c] = v
	return nil
}

// Mul returns the matrix product a*b. It returns an error if the number
// of columns of a does not equal the number of rows of b.
func Mul(a, b Matrix) (Matrix, error) {
	if a.cols != b.rows {
		return Matrix{}, fmt.Errorf("matrix: shape mismatch: cannot multiply %dx%d by %dx%d",
			a.rows, a.cols, b.rows, b.cols)
	}
	out, _ := Zeros(a.rows, b.cols)
	for i := 0; i < a.rows; i++ {
		for k := 0; k < a.cols; k++ {
			av := a.data[i*a.cols+k]
			if av == 0 {
				continue
			}
			for j := 0; j < b.cols; j++ {
				out.data[i*out.cols+j] += av * b.data[k*b.cols+j]
			}
		}
	}
	return out, nil
}

// Transpose returns the transpose of m.
func (m Matrix) Transpose() Matrix {
	out, _ := Zeros(m.cols, m.rows)
	for i := 0; i < m.rows; i++ {
		for j := 0; j < m.cols; j++ {
			out.data[j*out.cols+i] = m.data[i*m.cols+j]
		}
	}
	return out
}

// Det returns the determinant of m using LU decomposition with partial
// pivoting. It returns an error if m is not square.
func (m Matrix) Det() (float64, error) {
	if m.rows != m.cols {
		return 0, fmt.Errorf("%w: got %dx%d", ErrNotSquare, m.rows, m.cols)
	}
	n := m.rows
	a := make([]float64, len(m.data))
	copy(a, m.data)

	det := 1.0
	for col := 0; col < n; col++ {
		pivot := col
		max := math.Abs(a[col*n+col])
		for r := col + 1; r < n; r++ {
			if v := math.Abs(a[r*n+col]); v > max {
				max, pivot = v, r
			}
		}
		if max == 0 {
			return 0, nil // singular matrix
		}
		if pivot != col {
			for j := 0; j < n; j++ {
				a[col*n+j], a[pivot*n+j] = a[pivot*n+j], a[col*n+j]
			}
			det = -det
		}
		pv := a[col*n+col]
		det *= pv
		for r := col + 1; r < n; r++ {
			f := a[r*n+col] / pv
			for j := col; j < n; j++ {
				a[r*n+j] -= f * a[col*n+j]
			}
		}
	}
	return det, nil
}

// String returns a human-readable representation of m.
func (m Matrix) String() string {
	var sb strings.Builder
	for i := 0; i < m.rows; i++ {
		sb.WriteString("[")
		for j := 0; j < m.cols; j++ {
			if j > 0 {
				sb.WriteString(" ")
			}
			fmt.Fprintf(&sb, "%g", m.data[i*m.cols+j])
		}
		sb.WriteString("]")
		if i < m.rows-1 {
			sb.WriteString("\n")
		}
	}
	return sb.String()
}
