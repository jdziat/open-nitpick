package main

import (
	"fmt"
	"strings"
	"unicode"
)

// slugify lowercases text, replaces runs of non-alphanumeric characters
// with single hyphens, trims leading/trailing hyphens, and handles
// Unicode letters and digits.
func slugify(s string) string {
	s = strings.ToLower(s)

	var b strings.Builder
	b.Grow(len(s))

	pendingHyphen := false
	for _, r := range s {
		if unicode.IsLetter(r) || unicode.IsDigit(r) {
			if pendingHyphen && b.Len() > 0 {
				b.WriteByte('-')
			}
			pendingHyphen = false
			b.WriteRune(r)
		} else {
			pendingHyphen = true
		}
	}

	return b.String()
}

func main() {
	fmt.Println(slugify("Hello, World!"))
	fmt.Println(slugify("  Déjà Vu -- All Over Again  "))
	fmt.Println(slugify("日本語のタイトル Test 123"))
	fmt.Println(slugify("---Already---Slugged---"))
}
