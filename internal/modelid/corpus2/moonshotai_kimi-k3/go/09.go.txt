package main

import (
	"fmt"
	"iter"
	"time"
)

// DateRange returns an iterator that yields each day from start to end,
// inclusive. If skipWeekends is true, Saturdays and Sundays are omitted.
// Dates are normalized to midnight in their own location before iterating.
func DateRange(start, end time.Time, skipWeekends bool) iter.Seq[time.Time] {
	start = time.Date(start.Year(), start.Month(), start.Day(), 0, 0, 0, 0, start.Location())
	end = time.Date(end.Year(), end.Month(), end.Day(), 0, 0, 0, 0, end.Location())

	return func(yield func(time.Time) bool) {
		for d := start; !d.After(end); d = d.AddDate(0, 0, 1) {
			if skipWeekends {
				switch d.Weekday() {
				case time.Saturday, time.Sunday:
					continue
				}
			}
			if !yield(d) {
				return
			}
		}
	}
}

func main() {
	start := time.Date(2024, 6, 1, 0, 0, 0, 0, time.Local)
	end := time.Date(2024, 6, 10, 0, 0, 0, 0, time.Local)

	fmt.Println("All days:")
	for d := range DateRange(start, end, false) {
		fmt.Println(d.Format("Mon 2006-01-02"))
	}

	fmt.Println("\nWeekdays only:")
	for d := range DateRange(start, end, true) {
		fmt.Println(d.Format("Mon 2006-01-02"))
	}
}
