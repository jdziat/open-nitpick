// Package retry provides a helper for calling a function repeatedly with
// exponential backoff and jitter until it succeeds or a maximum number of
// attempts is reached.
package retry

import (
	"context"
	"errors"
	"math/rand"
	"time"
)

// Config controls how Do retries a function.
type Config struct {
	// MaxAttempts is the maximum number of times the function is called.
	// Values less than 1 are treated as 1.
	MaxAttempts int

	// InitialInterval is how long to wait before the first retry.
	// Non-positive values are treated as 100ms.
	InitialInterval time.Duration

	// MaxInterval caps the wait between attempts. Non-positive values
	// are treated as 30s.
	MaxInterval time.Duration

	// Multiplier is the factor by which the wait grows after each failed
	// attempt. Values less than 1 are treated as 2.
	Multiplier float64

	// Jitter is the maximum fraction of the computed wait that is added
	// or subtracted at random. For example, 0.5 lets each wait vary by up
	// to ±50%. Negative values are treated as 0 (no jitter).
	Jitter float64
}

// DefaultConfig is the configuration used by the package-level Do.
var DefaultConfig = Config{
	MaxAttempts:     3,
	InitialInterval: 100 * time.Millisecond,
	MaxInterval:     30 * time.Second,
	Multiplier:      2,
	Jitter:          0.5,
}

// Do calls fn using DefaultConfig. It is shorthand for
// DefaultConfig.Do(ctx, fn).
func Do(ctx context.Context, fn func() error) error {
	return DefaultConfig.Do(ctx, fn)
}

// Do calls fn until it returns nil or has been called MaxAttempts times.
// Between attempts it waits for an exponentially increasing interval,
// randomized by the configured jitter. If every attempt fails, the error
// from the final attempt is returned. If ctx is canceled while waiting
// between attempts, Do stops early and returns the last error joined with
// the context's error.
func (c Config) Do(ctx context.Context, fn func() error) error {
	c = c.withDefaults()

	delay := c.InitialInterval
	if delay > c.MaxInterval {
		delay = c.MaxInterval
	}

	var err error
	for attempt := 1; ; attempt++ {
		if err = fn(); err == nil {
			return nil
		}
		if attempt >= c.MaxAttempts {
			return err
		}

		timer := time.NewTimer(c.jittered(delay))
		select {
		case <-ctx.Done():
			timer.Stop()
			return errors.Join(err, ctx.Err())
		case <-timer.C:
		}

		delay = time.Duration(float64(delay) * c.Multiplier)
		if delay > c.MaxInterval {
			delay = c.MaxInterval
		}
	}
}

// withDefaults returns a copy of c with invalid fields replaced by
// sensible defaults.
func (c Config) withDefaults() Config {
	if c.MaxAttempts < 1 {
		c.MaxAttempts = 1
	}
	if c.InitialInterval <= 0 {
		c.InitialInterval = 100 * time.Millisecond
	}
	if c.MaxInterval <= 0 {
		c.MaxInterval = 30 * time.Second
	}
	if c.Multiplier < 1 {
		c.Multiplier = 2
	}
	if c.Jitter < 0 {
		c.Jitter = 0
	}
	return c
}

// jittered returns d perturbed by a random amount in the range
// ±c.Jitter*d, clamped to be non-negative.
func (c Config) jittered(d time.Duration) time.Duration {
	if c.Jitter == 0 {
		return d
	}
	jitter := (rand.Float64()*2 - 1) * c.Jitter * float64(d)
	out := time.Duration(float64(d) + jitter)
	if out < 0 {
		return 0
	}
	return out
}
