package main

import (
	"crypto/rand"
	"encoding/base64"
	"errors"
	"fmt"
	"sync"
	"time"
)

var (
	ErrNotFound = errors.New("short code not found")
	ErrExpired  = errors.New("short code expired")
)

type entry struct {
	url       string
	expiresAt time.Time
}

type Shortener struct {
	mu      sync.RWMutex
	entries map[string]entry
}

func NewShortener() *Shortener {
	return &Shortener{entries: make(map[string]entry)}
}

// Shorten stores url and returns a short code that expires after ttl.
func (s *Shortener) Shorten(url string, ttl time.Duration) (string, error) {
	code, err := generateCode(6)
	if err != nil {
		return "", err
	}
	s.mu.Lock()
	s.entries[code] = entry{url: url, expiresAt: time.Now().Add(ttl)}
	s.mu.Unlock()
	return code, nil
}

// Resolve returns the original URL for a code, or an error if it is
// unknown or expired.
func (s *Shortener) Resolve(code string) (string, error) {
	s.mu.RLock()
	e, ok := s.entries[code]
	s.mu.RUnlock()
	if !ok {
		return "", ErrNotFound
	}
	if time.Now().After(e.expiresAt) {
		s.mu.Lock()
		delete(s.entries, code)
		s.mu.Unlock()
		return "", ErrExpired
	}
	return e.url, nil
}

func generateCode(n int) (string, error) {
	b := make([]byte, n)
	if _, err := rand.Read(b); err != nil {
		return "", err
	}
	return base64.RawURLEncoding.EncodeToString(b)[:n], nil
}

func main() {
	s := NewShortener()

	code, err := s.Shorten("https://example.com/some/very/long/url", time.Minute)
	if err != nil {
		fmt.Println("shorten error:", err)
		return
	}
	fmt.Println("short code:", code)

	url, err := s.Resolve(code)
	if err != nil {
		fmt.Println("resolve error:", err)
		return
	}
	fmt.Println("resolved:", url)

	stale, _ := s.Shorten("https://example.com/gone", -time.Second)
	if _, err := s.Resolve(stale); err != nil {
		fmt.Println("expired resolve:", err)
	}
}
