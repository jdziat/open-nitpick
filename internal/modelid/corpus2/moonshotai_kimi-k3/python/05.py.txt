"""Word-frequency counter.

Reads text, normalises case and punctuation, and reports the top K
most frequent words with their counts.
"""

import argparse
import re
import sys
from collections import Counter


def tokenize(text):
    """Split text into normalised word tokens.

    Case is folded to lowercase and punctuation is removed, so that
    e.g. "Hello," and "hello" count as the same word. Apostrophes
    inside words ("don't") are preserved.
    """
    return re.findall(r"[a-z0-9]+(?:'[a-z0-9]+)*", text.lower())


def count_words(text):
    """Return a Counter mapping each normalised word to its frequency."""
    return Counter(tokenize(text))


def top_k(text, k):
    """Return the k most frequent words in text as (word, count) pairs.

    Results are ordered by descending count; ties are broken
    alphabetically so output is deterministic.
    """
    counts = count_words(text)
    return sorted(counts.items(), key=lambda item: (-item[1], item[0]))[:k]


def main(argv=None):
    parser = argparse.ArgumentParser(
        description="Count word frequencies and print the top K words."
    )
    parser.add_argument(
        "file",
        nargs="?",
        help="Text file to read. Reads from standard input if omitted.",
    )
    parser.add_argument(
        "-k",
        "--top",
        type=int,
        default=10,
        help="Number of top words to show (default: 10).",
    )
    parser.add_argument(
        "-e",
        "--encoding",
        default="utf-8",
        help="File encoding (default: utf-8).",
    )
    args = parser.parse_args(argv)

    if args.top < 1:
        parser.error("K must be a positive integer")

    if args.file:
        with open(args.file, "r", encoding=args.encoding) as fh:
            text = fh.read()
    else:
        text = sys.stdin.read()

    for word, count in top_k(text, args.top):
        print(f"{word}\t{count}")


if __name__ == "__main__":
    main()
