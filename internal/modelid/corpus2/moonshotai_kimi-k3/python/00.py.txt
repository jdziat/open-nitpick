"""LRU cache with fixed capacity and O(1) get/put operations.

Implemented with a hash map plus a doubly linked list ordered from
most recently used (front) to least recently used (back). When the
cache is full, the entry at the back of the list is evicted.
"""


class _Node:
    """A node in the doubly linked list."""

    __slots__ = ("key", "value", "prev", "next")

    def __init__(self, key=None, value=None):
        self.key = key
        self.value = value
        self.prev = None
        self.next = None


class LRUCache:
    """A fixed-capacity least-recently-used cache."""

    def __init__(self, capacity):
        if capacity <= 0:
            raise ValueError("capacity must be a positive integer")
        self.capacity = capacity
        self._map = {}
        # Sentinel head/tail nodes remove edge-case checks.
        self._head = _Node()  # Front: most recently used.
        self._tail = _Node()  # Back: least recently used.
        self._head.next = self._tail
        self._tail.prev = self._head

    def get(self, key):
        """Return the value stored under key, or -1 if it is absent.

        A successful lookup marks the entry as most recently used.
        """
        node = self._map.get(key)
        if node is None:
            return -1
        self._move_to_front(node)
        return node.value

    def put(self, key, value):
        """Insert or update key with value.

        If the cache is at capacity, the least recently used entry is
        evicted first.
        """
        node = self._map.get(key)
        if node is not None:
            node.value = value
            self._move_to_front(node)
            return
        if len(self._map) >= self.capacity:
            self._evict_lru()
        node = _Node(key, value)
        self._map[key] = node
        self._insert_at_front(node)

    def __len__(self):
        return len(self._map)

    def __contains__(self, key):
        return key in self._map

    # -- internal helpers -------------------------------------------------

    def _insert_at_front(self, node):
        node.prev = self._head
        node.next = self._head.next
        self._head.next.prev = node
        self._head.next = node

    def _remove(self, node):
        node.prev.next = node.next
        node.next.prev = node.prev

    def _move_to_front(self, node):
        self._remove(node)
        self._insert_at_front(node)

    def _evict_lru(self):
        lru = self._tail.prev
        self._remove(lru)
        del self._map[lru.key]


if __name__ == "__main__":
    cache = LRUCache(2)

    cache.put(1, "one")
    cache.put(2, "two")
    assert cache.get(1) == "one"   # Key 1 becomes most recently used.

    cache.put(3, "three")          # Evicts key 2.
    assert cache.get(2) == -1

    cache.put(4, "four")           # Evicts key 1.
    assert cache.get(1) == -1
    assert cache.get(3) == "three"
    assert cache.get(4) == "four"

    cache.put(3, "THREE")          # Update in place; no eviction.
    assert cache.get(3) == "THREE"
    assert len(cache) == 2

    print("All LRU cache checks passed.")
