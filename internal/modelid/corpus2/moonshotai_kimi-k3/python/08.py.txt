"""A priority queue backed by a binary heap.

The ordering is defined by a caller-supplied comparison function
``compare(a, b)`` which must return a negative number if ``a`` should be
dequeued before ``b``, zero if they are equivalent, and a positive number
otherwise. By default the queue behaves as a min-heap over natural ordering.
"""


class PriorityQueue:
    """Priority queue implemented with an array-based binary heap."""

    def __init__(self, compare=None, iterable=None):
        """Create a priority queue.

        Args:
            compare: Optional callable ``compare(a, b)`` returning a negative
                number, zero, or a positive number depending on whether ``a``
                has higher, equal, or lower priority than ``b``. Defaults to
                natural ordering (min-heap).
            iterable: Optional initial items, heapified in linear time.
        """
        if compare is None:
            compare = lambda a, b: (a > b) - (a < b)
        self._compare = compare
        self._heap = list(iterable) if iterable is not None else []
        for i in range(len(self._heap) // 2 - 1, -1, -1):
            self._sift_down(i)

    def push(self, item):
        """Insert ``item`` into the queue. O(log n)."""
        self._heap.append(item)
        self._sift_up(len(self._heap) - 1)

    def pop(self):
        """Remove and return the highest-priority item. O(log n).

        Raises:
            IndexError: If the queue is empty.
        """
        if not self._heap:
            raise IndexError("pop from empty priority queue")
        top = self._heap[0]
        last = self._heap.pop()
        if self._heap:
            self._heap[0] = last
            self._sift_down(0)
        return top

    def peek(self):
        """Return the highest-priority item without removing it. O(1).

        Raises:
            IndexError: If the queue is empty.
        """
        if not self._heap:
            raise IndexError("peek from empty priority queue")
        return self._heap[0]

    def _sift_up(self, index):
        heap = self._heap
        compare = self._compare
        item = heap[index]
        while index > 0:
            parent = (index - 1) // 2
            if compare(item, heap[parent]) >= 0:
                break
            heap[index] = heap[parent]
            index = parent
        heap[index] = item

    def _sift_down(self, index):
        heap = self._heap
        compare = self._compare
        size = len(heap)
        item = heap[index]
        while True:
            child = 2 * index + 1
            if child >= size:
                break
            if child + 1 < size and compare(heap[child + 1], heap[child]) < 0:
                child += 1
            if compare(heap[child], item) >= 0:
                break
            heap[index] = heap[child]
            index = child
        heap[index] = item

    def __len__(self):
        return len(self._heap)

    def __bool__(self):
        return bool(self._heap)

    def __repr__(self):
        return f"{type(self).__name__}({self._heap!r})"


if __name__ == "__main__":
    # Min-heap with default ordering.
    pq = PriorityQueue()
    for value in [5, 1, 9, 3, 7]:
        pq.push(value)
    assert pq.peek() == 1
    assert [pq.pop() for _ in range(5)] == [1, 3, 5, 7, 9]

    # Max-heap via a reversed comparison.
    pq = PriorityQueue(compare=lambda a, b: (b > a) - (b < a))
    for value in [5, 1, 9, 3, 7]:
        pq.push(value)
    assert [pq.pop() for _ in range(5)] == [9, 7, 5, 3, 1]

    # Custom ordering: tasks ordered by numeric priority field.
    pq = PriorityQueue(compare=lambda a, b: a[0] - b[0],
                       iterable=[(3, "low"), (1, "urgent"), (2, "mid")])
    assert pq.pop() == (1, "urgent")
    assert pq.peek() == (2, "mid")
    assert len(pq) == 2

    print("All checks passed.")
