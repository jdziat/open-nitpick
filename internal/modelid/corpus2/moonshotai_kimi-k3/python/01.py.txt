"""A parser for a subset of CSV.

Supports quoted fields, escaped quotes, and a configurable delimiter.
Parses text into rows of strings.
"""

import sys


def parse_csv(text, delimiter=","):
    """Parse CSV text into a list of rows, each a list of strings.

    Parsing rules:
      * Fields are separated by ``delimiter`` (a single character).
      * A field may be enclosed in double quotes; the quotes are removed.
      * Inside a quoted field, two consecutive double quotes stand for
        one literal double quote.
      * Quoted fields may contain delimiters and line breaks.
      * A double quote that does not open a field is taken literally.
      * Rows are separated by LF, CRLF, or CR line endings.
      * An unterminated quoted field runs to the end of the input.
    """
    if len(delimiter) != 1:
        raise ValueError("delimiter must be a single character")
    if delimiter in ('"', "\r", "\n"):
        raise ValueError("delimiter cannot be a quote or a newline")

    rows = []
    row = []
    field = []
    in_quotes = False
    quoted = False  # the current field began with a quote
    i = 0
    end = len(text)

    while i < end:
        ch = text[i]

        if in_quotes:
            if ch == '"':
                if i + 1 < end and text[i + 1] == '"':
                    field.append('"')
                    i += 1
                else:
                    in_quotes = False
            else:
                field.append(ch)
        elif ch == '"' and not field and not quoted:
            in_quotes = True
            quoted = True
        elif ch == delimiter:
            row.append("".join(field))
            field = []
            quoted = False
        elif ch == "\n" or ch == "\r":
            row.append("".join(field))
            rows.append(row)
            row = []
            field = []
            quoted = False
            if ch == "\r" and i + 1 < end and text[i + 1] == "\n":
                i += 1
        else:
            field.append(ch)

        i += 1

    if field or quoted or row:
        row.append("".join(field))
        rows.append(row)

    return rows


def load_csv(stream, delimiter=","):
    """Read all text from a file-like object and parse it."""
    return parse_csv(stream.read(), delimiter=delimiter)


if __name__ == "__main__":
    delim = sys.argv[1] if len(sys.argv) > 1 else ","
    for record in load_csv(sys.stdin, delimiter=delim):
        print(record)
