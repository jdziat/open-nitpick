"""JSON configuration loader with defaults, environment overrides, and validation.

Merge order (lowest to highest precedence):
    1. Defaults supplied by the caller.
    2. Values from the JSON configuration file.
    3. Environment variable overrides.

After merging, all required keys are validated; a MissingRequiredKeyError
is raised if any are absent or set to None.

Environment overrides can be supplied two ways:
    * env_prefix: any variable named "<PREFIX>_<KEY>" overrides <key>.
      Use a double underscore for nesting, e.g. "APP_DATABASE__HOST"
      overrides config["database"]["host"].
    * env_mapping: an explicit {environment_variable: "dotted.key"} mapping.

Environment values are coerced with json.loads, so "8080" becomes int,
"true" becomes bool, "[1, 2]" becomes list, and anything unparsable
remains a string.
"""

from __future__ import annotations

import json
import os
from typing import Any, Mapping, MutableMapping, Optional, Sequence, Tuple


class ConfigError(Exception):
    """Base class for all configuration errors."""


class ConfigFileError(ConfigError):
    """The configuration file could not be read or parsed."""


class MissingRequiredKeyError(ConfigError):
    """One or more required configuration keys are missing or None."""

    def __init__(self, missing_keys: Sequence[str]) -> None:
        self.missing_keys = sorted(missing_keys)
        super().__init__(
            "Missing required configuration key(s): "
            + ", ".join(self.missing_keys)
        )


def load_config(
    path: Optional[str | os.PathLike] = None,
    *,
    defaults: Optional[Mapping[str, Any]] = None,
    required_keys: Sequence[str] | str = (),
    env_prefix: str = "",
    env_mapping: Optional[Mapping[str, str]] = None,
    environ: Optional[Mapping[str, str]] = None,
) -> dict:
    """Load configuration from a JSON file with defaults and env overrides.

    Args:
        path: Path to the JSON configuration file. If None, no file is
            read. If given but missing, unreadable, or invalid JSON, a
            ConfigFileError is raised. The file must contain a JSON object.
        defaults: Mapping of default values used as the base layer.
        required_keys: Keys that must be present (and not None) in the
            final configuration. Use dotted names for nested keys,
            e.g. "database.host". A single string is also accepted.
        env_prefix: Prefix for convention-based environment overrides.
        env_mapping: Explicit {env var name: "dotted.key"} overrides.
        environ: Environment mapping to read; defaults to os.environ.

    Returns:
        The merged, validated configuration as a dict.

    Raises:
        ConfigFileError: If the file cannot be read or parsed.
        MissingRequiredKeyError: If any required key is absent or None.
    """
    if environ is None:
        environ = os.environ
    if isinstance(required_keys, str):
        required_keys = [required_keys]

    config: dict = dict(defaults) if defaults else {}

    if path is not None:
        config = _deep_merge(config, _read_json_file(path))

    config = _deep_merge(
        config, _env_overrides(env_prefix, env_mapping, environ)
    )

    _validate_required(config, required_keys)
    return config


def _read_json_file(path: str | os.PathLike) -> dict:
    """Read and parse a JSON file, raising ConfigFileError on failure."""
    try:
        with open(path, "r", encoding="utf-8") as handle:
            data = json.load(handle)
    except FileNotFoundError as exc:
        raise ConfigFileError(f"Configuration file not found: {path}") from exc
    except json.JSONDecodeError as exc:
        raise ConfigFileError(
            f"Invalid JSON in configuration file {path}: {exc}"
        ) from exc
    except OSError as exc:
        raise ConfigFileError(
            f"Could not read configuration file {path}: {exc}"
        ) from exc

    if not isinstance(data, dict):
        raise ConfigFileError(
            f"Configuration file {path} must contain a JSON object "
            f"at the top level, got {type(data).__name__}."
        )
    return data


def _deep_merge(
    base: Mapping[str, Any], override: Mapping[str, Any]
) -> dict:
    """Recursively merge override into base, returning a new dict."""
    merged = dict(base)
    for key, value in override.items():
        existing = merged.get(key)
        if isinstance(existing, Mapping) and isinstance(value, Mapping):
            merged[key] = _deep_merge(existing, value)
        else:
            merged[key] = value
    return merged


def _coerce_env_value(raw: str) -> Any:
    """Convert an environment variable string to a native Python value."""
    try:
        return json.loads(raw)
    except (json.JSONDecodeError, TypeError, ValueError):
        return raw


def _set_nested(
    config: MutableMapping[str, Any], dotted_key: str, value: Any
) -> None:
    """Set config["a"]["b"] = value for the dotted key "a.b"."""
    parts = [part for part in dotted_key.split(".") if part]
    if not parts:
        return
    node = config
    for part in parts[:-1]:
        child = node.get(part)
        if not isinstance(child, MutableMapping):
            child = {}
            node[part] = child
        node = child
    node[parts[-1]] = value


def _lookup(
    config: Mapping[str, Any], dotted_key: str
) -> Tuple[bool, Any]:
    """Resolve a dotted key, returning (found, value)."""
    node: Any = config
    for part in dotted_key.split("."):
        if not isinstance(node, Mapping) or part not in node:
            return False, None
        node = node[part]
    return True, node


def _env_overrides(
    env_prefix: str,
    env_mapping: Optional[Mapping[str, str]],
    environ: Mapping[str, str],
) -> dict:
    """Build a nested dict of overrides from environment variables."""
    overrides: dict = {}

    if env_prefix:
        prefix = env_prefix.rstrip("_") + "_"
        for name, raw in environ.items():
            if not name.startswith(prefix):
                continue
            suffix = name[len(prefix):]
            dotted = ".".join(
                part.lower() for part in suffix.split("__") if part
            )
            if dotted:
                _set_nested(overrides, dotted, _coerce_env_value(raw))

    for name, dotted in (env_mapping or {}).items():
        if name in environ:
            _set_nested(overrides, dotted, _coerce_env_value(environ[name]))

    return overrides


def _validate_required(
    config: Mapping[str, Any], required_keys: Sequence[str]
) -> None:
    """Raise MissingRequiredKeyError for keys that are absent or None."""
    missing = []
    for key in required_keys:
        found, value = _lookup(config, key)
        if not found or value is None:
            missing.append(key)
    if missing:
        raise MissingRequiredKeyError(missing)


if __name__ == "__main__":
    import tempfile

    demo_json = {"host": "example.com", "database": {"name": "mydb"}}
    with tempfile.NamedTemporaryFile(
        "w", suffix=".json", delete=False
    ) as tmp:
        json.dump(demo_json, tmp)
        demo_path = tmp.name

    os.environ["DEMO_PORT"] = "9090"
    os.environ["DEMO_DATABASE__HOST"] = "db.internal"

    try:
        loaded = load_config(
            demo_path,
            defaults={
                "host": "localhost",
                "port": 8080,
                "debug": False,
                "database": {"host": "localhost", "name": "dev"},
            },
            required_keys=["host", "port", "database.host"],
            env_prefix="DEMO",
        )
        print(json.dumps(loaded, indent=2))
    finally:
        os.unlink(demo_path)
