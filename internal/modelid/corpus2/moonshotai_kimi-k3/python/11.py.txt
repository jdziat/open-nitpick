"""A matrix type with multiplication, transpose, and determinant."""

from fractions import Fraction
from numbers import Number


class MatrixError(Exception):
    """Base class for matrix errors."""


class ShapeMismatchError(MatrixError):
    """Raised when matrix shapes are incompatible for an operation."""


class NotSquareError(MatrixError):
    """Raised when an operation requires a square matrix."""


class Matrix:
    """A dense 2-D matrix of numbers."""

    def __init__(self, rows):
        rows = [list(row) for row in rows]
        if not rows or not rows[0]:
            raise ShapeMismatchError("matrix needs at least one row and one column")
        width = len(rows[0])
        if any(len(row) != width for row in rows):
            raise ShapeMismatchError("all rows must have the same length")
        self._rows = rows

    # -- basics ----------------------------------------------------------

    @property
    def shape(self):
        """Return (number of rows, number of columns)."""
        return (len(self._rows), len(self._rows[0]))

    def to_list(self):
        """Return a deep copy of the matrix as a list of lists."""
        return [row[:] for row in self._rows]

    def __getitem__(self, index):
        return self._rows[index]

    def __eq__(self, other):
        return isinstance(other, Matrix) and self._rows == other._rows

    def __repr__(self):
        return f"Matrix({self._rows!r})"

    # -- transpose --------------------------------------------------------

    def transpose(self):
        """Return a new matrix with rows and columns swapped."""
        return Matrix([list(column) for column in zip(*self._rows)])

    @property
    def T(self):
        return self.transpose()

    # -- multiplication ---------------------------------------------------

    def matmul(self, other):
        """Matrix-multiply self by other, checking inner dimensions."""
        if not isinstance(other, Matrix):
            raise TypeError(f"expected a Matrix, got {type(other).__name__}")
        (m, n), (p, q) = self.shape, other.shape
        if n != p:
            raise ShapeMismatchError(
                f"cannot multiply a {m}x{n} matrix by a {p}x{q} matrix"
            )
        return Matrix([
            [sum(self._rows[i][k] * other._rows[k][j] for k in range(n))
             for j in range(q)]
            for i in range(m)
        ])

    def __matmul__(self, other):
        return self.matmul(other)

    def __mul__(self, other):
        if isinstance(other, Matrix):
            return self.matmul(other)
        if isinstance(other, Number):
            return Matrix([[value * other for value in row] for row in self._rows])
        return NotImplemented

    def __rmul__(self, other):
        if isinstance(other, Number):
            return Matrix([[other * value for value in row] for row in self._rows])
        return NotImplemented

    # -- determinant ------------------------------------------------------

    def determinant(self):
        """Return the determinant of a square matrix.

        Uses the Bareiss algorithm (fraction-free Gaussian elimination),
        which is exact for integer and floating-point entries.
        """
        rows, cols = self.shape
        if rows != cols:
            raise NotSquareError(
                f"determinant needs a square matrix, got {rows}x{cols}"
            )
        if rows == 1:
            return self._rows[0][0]
        a = [[Fraction(v) for v in row] for row in self._rows]
        sign = 1
        previous = Fraction(1)
        for k in range(rows - 1):
            if a[k][k] == 0:
                pivot = next((r for r in range(k + 1, rows) if a[r][k]), None)
                if pivot is None:
                    return 0
                a[k], a[pivot] = a[pivot], a[k]
                sign = -sign
            for i in range(k + 1, rows):
                for j in range(k + 1, rows):
                    a[i][j] = (a[i][j] * a[k][k] - a[i][k] * a[k][j]) / previous
            previous = a[k][k]
        result = sign * a[-1][-1]
        return int(result) if result.denominator == 1 else float(result)

    det = determinant


if __name__ == "__main__":
    a = Matrix([[1, 2], [3, 4]])
    b = Matrix([[5, 6], [7, 8]])

    print("a       =", a.to_list())
    print("a @ b   =", (a @ b).to_list())
    print("a * b   =", (a * b).to_list())
    print("a.T     =", a.T.to_list())
    print("det(a)  =", a.determinant())
    print("3 * a   =", (3 * a).to_list())

    for attempt in (
        lambda: a @ Matrix([[1, 2, 3]]),
        lambda: Matrix([[1, 2, 3]]).determinant(),
        lambda: Matrix([[1, 2], [3]]),
    ):
        try:
            attempt()
        except MatrixError as exc:
            print(f"{type(exc).__name__}: {exc}")
