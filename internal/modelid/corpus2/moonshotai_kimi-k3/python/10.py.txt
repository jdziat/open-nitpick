"""Minimal in-memory URL shortener with expiry."""

import secrets
import string
import time


class URLShortener:
    _ALPHABET = string.ascii_letters + string.digits

    def __init__(self, code_length=6, default_ttl=3600.0):
        self.code_length = code_length
        self.default_ttl = default_ttl  # seconds; None means never expire
        self._store = {}  # code -> (url, expires_at or None)

    def shorten(self, url, ttl=None):
        """Store url and return a short code.

        ttl is the lifetime in seconds; None falls back to default_ttl.
        If both are None, the code never expires.
        """
        if ttl is None:
            ttl = self.default_ttl
        expires_at = None if ttl is None else time.monotonic() + ttl
        while True:
            code = "".join(
                secrets.choice(self._ALPHABET) for _ in range(self.code_length)
            )
            if code not in self._store:
                break
        self._store[code] = (url, expires_at)
        return code

    def resolve(self, code):
        """Return the original URL, or None if the code is unknown or expired."""
        entry = self._store.get(code)
        if entry is None:
            return None
        url, expires_at = entry
        if expires_at is not None and time.monotonic() >= expires_at:
            del self._store[code]
            return None
        return url


if __name__ == "__main__":
    shortener = URLShortener(default_ttl=60)
    code = shortener.shorten("https://example.com/a/very/long/url")
    print(f"{code} -> {shortener.resolve(code)}")
    print("missing ->", shortener.resolve("nope"))
