export interface DateRangeObjectOptions {
  readonly skipWeekends?: boolean;
}

export type DateRangeOptions = boolean | DateRangeObjectOptions;

function normalizeToDay(date: Date): Date {
  if (Number.isNaN(date.getTime())) {
    throw new RangeError("Invalid date");
  }
  return new Date(date.getFullYear(), date.getMonth(), date.getDate());
}

export function* dateRange(
  start: Date,
  end: Date,
  options: DateRangeOptions = false
): Generator<Date, void, unknown> {
  const normalizedStart = normalizeToDay(start);
  const normalizedEnd = normalizeToDay(end);

  if (normalizedStart > normalizedEnd) {
    return;
  }

  const skipWeekends =
    typeof options === "boolean" ? options : Boolean(options.skipWeekends);

  const cursor = new Date(normalizedStart);

  while (cursor <= normalizedEnd) {
    const day = cursor.getDay();
    if (!skipWeekends || (day !== 0 && day !== 6)) {
      yield new Date(cursor);
    }
    cursor.setDate(cursor.getDate() + 1);
  }
}

export default dateRange;
