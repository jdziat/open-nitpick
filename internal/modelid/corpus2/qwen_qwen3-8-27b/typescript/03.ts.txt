export type RetryResult<T> = T | Error;

const sleep = (ms: number): Promise<void> =>
  new Promise<void>((resolve) => setTimeout(resolve, ms));

export async function retry<T>(
  fn: () => T | Promise<T>,
  maxAttempts: number,
  baseDelayMs = 100,
  maxDelayMs = 10_000,
): Promise<RetryResult<T>> {
  if (!Number.isFinite(maxAttempts) || maxAttempts < 1) {
    return new Error('retry: maxAttempts must be at least 1');
  }

  const attempts = Math.floor(maxAttempts);
  const base = Math.max(0, baseDelayMs);
  const cap = Math.max(base, maxDelayMs);
  let lastError: Error = new Error('retry: no attempts made');

  for (let attempt = 1; attempt <= attempts; attempt += 1) {
    try {
      return await fn();
    } catch (err: unknown) {
      lastError = err instanceof Error ? err : new Error(String(err));

      if (attempt === attempts) {
        break;
      }

      const exponential = Math.min(cap, base * 2 ** (attempt - 1));
      const jitter = Math.random() * exponential;
      await sleep(Math.min(cap, exponential + jitter));
    }
  }

  return lastError;
}
