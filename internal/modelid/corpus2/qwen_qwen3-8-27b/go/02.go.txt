package ratelimit

import (
	"context"
	"errors"
	"sync"
	"time"
)

var (
	ErrInvalid    = errors.New("ratelimit: invalid token count")
	ErrZeroRate   = errors.New("ratelimit: zero refill rate")
	ErrExceedsCap = errors.New("ratelimit: request exceeds capacity")
)

// Limiter is a token-bucket rate limiter.
type Limiter struct {
	mu       sync.Mutex
	capacity float64
	rate     float64
	tokens   float64
	last     time.Time
}

// New returns a limiter with the given bucket capacity and refill rate
// in tokens per second.
func New(capacity, rate float64) *Limiter {
	if capacity != capacity || capacity < 0 {
		capacity = 0
	}
	if rate != rate || rate < 0 {
		rate = 0
	}
	return &Limiter{
		capacity: capacity,
		rate:     rate,
		tokens:   capacity,
		last:     time.Now(),
	}
}

func (l *Limiter) refill(now time.Time) {
	if l.last.IsZero() {
		l.last = now
		return
	}
	if l.rate <= 0 {
		return
	}
	elapsed := now.Sub(l.last)
	if elapsed <= 0 {
		return
	}
	l.tokens += elapsed.Seconds() * l.rate
	if l.tokens > l.capacity {
		l.tokens = l.capacity
	}
	l.last = now
}

// Allow reports whether n tokens can be taken immediately.
func (l *Limiter) Allow(n float64) bool {
	if n != n {
		return false
	}
	if n <= 0 {
		return true
	}

	l.mu.Lock()
	defer l.mu.Unlock()

	l.refill(time.Now())
	if l.tokens >= n {
		l.tokens -= n
		return true
	}
	return false
}

// TryAcquire is an alias for Allow.
func (l *Limiter) TryAcquire(n float64) bool {
	return l.Allow(n)
}

// Wait blocks until n tokens are available or ctx is canceled.
func (l *Limiter) Wait(ctx context.Context, n float64) error {
	if n != n {
		return ErrInvalid
	}
	if n <= 0 {
		return nil
	}

	for {
		select {
		case <-ctx.Done():
			return ctx.Err()
		default:
		}

		l.mu.Lock()
		now := time.Now()
		l.refill(now)

		if l.tokens >= n {
			l.tokens -= n
			l.mu.Unlock()
			return nil
		}
		if n > l.capacity {
			l.mu.Unlock()
			return ErrExceedsCap
		}
		if l.rate <= 0 {
			l.mu.Unlock()
			return ErrZeroRate
		}

		seconds := (n - l.tokens) / l.rate
		const maxSeconds = float64(time.Hour)
		if seconds > maxSeconds {
			seconds = maxSeconds
		}
		if seconds <= 0 {
			seconds = 1
		}
		wait := time.Duration(seconds * float64(time.Second))
		l.mu.Unlock()

		select {
		case <-ctx.Done():
			return ctx.Err()
		case <-time.After(wait):
		}
	}
}

// Acquire is an alias for Wait.
func (l *Limiter) Acquire(ctx context.Context, n float64) error {
	return l.Wait(ctx, n)
}
