package main

import (
	"fmt"
	"io"
	"os"
	"sort"
	"strconv"
	"strings"
	"unicode"
)

type WordCount struct {
	Word  string
	Count int
}

func TopKWords(text string, k int) []WordCount {
	if k <= 0 {
		return nil
	}

	counts := make(map[string]int)
	var b strings.Builder

	flush := func() {
		if b.Len() > 0 {
			counts[b.String()]++
			b.Reset()
		}
	}

	for _, r := range text {
		if unicode.IsLetter(r) || unicode.IsNumber(r) {
			b.WriteRune(unicode.ToLower(r))
		} else {
			flush()
		}
	}
	flush()

	result := make([]WordCount, 0, len(counts))
	for word, count := range counts {
		result = append(result, WordCount{Word: word, Count: count})
	}

	sort.Slice(result, func(i, j int) bool {
		if result[i].Count != result[j].Count {
			return result[i].Count > result[j].Count
		}
		return result[i].Word < result[j].Word
	})

	if len(result) > k {
		result = result[:k]
	}
	return result
}

func main() {
	k := 10
	if len(os.Args) > 1 {
		var err error
		k, err = strconv.Atoi(os.Args[1])
		if err != nil {
			fmt.Fprintln(os.Stderr, "usage: wordfreq [k]")
			os.Exit(1)
		}
	}

	data, err := io.ReadAll(os.Stdin)
	if err != nil {
		fmt.Fprintln(os.Stderr, err)
		os.Exit(1)
	}

	for _, wc := range TopKWords(string(data), k) {
		fmt.Printf("%s\t%d\n", wc.Word, wc.Count)
	}
}
