package main

import (
	"errors"
	"fmt"
	"math"
)

// Matrix represents a row-major matrix.
type Matrix struct {
	Rows int
	Cols int
	Data []float64
}

func NewMatrix(rows, cols int, data []float64) (*Matrix, error) {
	if rows < 0 || cols < 0 {
		return nil, fmt.Errorf("invalid matrix dimensions %dx%d", rows, cols)
	}
	expected := rows * cols
	if len(data) != expected {
		return nil, fmt.Errorf("matrix data length %d does not match dimensions %dx%d", len(data), rows, cols)
	}
	m := &Matrix{Rows: rows, Cols: cols, Data: make([]float64, expected)}
	copy(m.Data, data)
	return m, nil
}
