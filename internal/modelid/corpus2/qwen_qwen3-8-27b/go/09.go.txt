package main

import (
	"fmt"
	"time"
)

type DateRangeIterator struct {
	current      time.Time
	end          time.Time
	skipWeekends bool
}

func NewDateRangeIterator(start, end time.Time, skipWeekends bool) *DateRangeIterator {
	return &DateRangeIterator{
		current:      time.Date(start.Year(), start.Month(), start.Day(), 0, 0, 0, 0, time.UTC),
		end:          time.Date(end.Year(), end.Month(), end.Day(), 0, 0, 0, 0, time.UTC),
		skipWeekends: skipWeekends,
	}
}

func (it *DateRangeIterator) Next() (time.Time, bool) {
	for !it.current.After(it.end) {
		weekday := it.current.Weekday()
		if !it.skipWeekends || (weekday != time.Saturday && weekday != time.Sunday) {
			day := it.current
			it.current = it.current.AddDate(0, 0, 1)
			return day, true
		}
		it.current = it.current.AddDate(0, 0, 1)
	}
	return time.Time{}, false
}

func main() {
	start := time.Date(2026, time.February, 17, 0, 0, 0, 0, time.UTC)
	end := time.Date(2026, time.February, 21, 0, 0, 0, 0, time.UTC)

	it := NewDateRangeIterator(start, end, true)
	for day, ok := it.Next(); ok; day, ok = it.Next() {
		fmt.Println(day.Format("2006-01-02"))
	}
}
