#!/usr/bin/env python3
"""Count the most frequent words in text."""

import argparse
import re
import sys
from collections import Counter

_WORD_RE = re.compile(r"[^\W_]+")


def normalize_words(text):
    return _WORD_RE.findall(text.lower())


def word_frequencies(text, k=10):
    counts = Counter(normalize_words(text))
    return sorted(counts.items(), key=lambda item: (-item[1], item[0]))[:k]


def main(argv=None):
    parser = argparse.ArgumentParser(description="Print the top K word frequencies.")
    parser.add_argument("files", nargs="*", help="Input files (reads stdin if omitted).")
    parser.add_argument("-k", "--top", type=int, default=10, help="Number of words to display (default: 10).")
    args = parser.parse_args(argv)

    if args.k < 0:
        parser.error("k must be non-negative")

    chunks = []
    for path in args.files:
        with open(path, "r", encoding="utf-8", errors="replace") as f:
            chunks.append(f.read())

    if not args.files:
        chunks.append(sys.stdin.read())

    text = "\n".join(chunks)

    for word, count in word_frequencies(text, args.k):
        print(f"{word}\t{count}")


if __name__ == "__main__":
    main()
