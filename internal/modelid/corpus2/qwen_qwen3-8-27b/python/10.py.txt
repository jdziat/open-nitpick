import time
import secrets


class URLShortener:
    def __init__(self):
        self._store = {}

    def shorten(self, url, ttl=None, expires_in=None):
        if expires_in is not None:
            ttl = expires_in

        code = secrets.token_urlsafe(6)
        while code in self._store:
            code = secrets.token_urlsafe(6)

        expires_at = None if ttl is None else time.monotonic() + ttl
        self._store[code] = (url, expires_at)
        return code

    def resolve(self, code):
        entry = self._store.get(code)
        if entry is None:
            return None

        url, expires_at = entry
        if expires_at is not None and time.monotonic() >= expires_at:
            del self._store[code]
            return None

        return url


_default_shortener = URLShortener()


def shorten(url, ttl=None, expires_in=None):
    return _default_shortener.shorten(url, ttl, expires_in)


def resolve(code):
    return _default_shortener.resolve(code)
