from datetime import date, timedelta
from typing import Iterator


def date_range(start: date, end: date, skip_weekends: bool = False) -> Iterator[date]:
    """Yield each day from start to end inclusive, optionally skipping weekends."""
    if start > end:
        return

    current = start
    one_day = timedelta(days=1)

    while current <= end:
        if not skip_weekends or current.weekday() < 5:
            yield current
        current += one_day


date_range_iterator = date_range
day_range = date_range
