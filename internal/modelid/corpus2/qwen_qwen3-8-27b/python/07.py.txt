"""Load JSON configuration with defaults, environment-variable overrides, and validation."""

from __future__ import annotations

import json
import os
import sys
from typing import Any, Dict, Iterable, List, Mapping, Optional, Tuple, Union

DEFAULT_CONFIG: Dict[str, Any] = {
    "app": {
        "name": "example",
        "debug": False,
        "port": 8000,
    },
    "database": {
        "host": "localhost",
        "port": 5432,
        "name": "app",
        "user": "app",
        "password": "",
    },
}

REQUIRED_KEYS: Tuple[str, ...] = (
    "app.name",
    "database.host",
    "database.port",
    "database.name",
)

ENV_PREFIX = "CONFIG"


class ConfigError(ValueError):
    """Raised when configuration is invalid."""


def _is_mapping(value: Any) -> bool:
    return isinstance(value, Mapping)


def _deep_merge(base: Mapping[str, Any], override: Mapping[str, Any]) -> Dict[str, Any]:
    merged: Dict[str, Any] = {}
    for key, value in base.items():
        merged[key] = dict(value) if _is_mapping(value) else value
    for key, value in override.items():
        if key in merged and _is_mapping(merged[key]) and _is_mapping(value):
            merged[key] = _deep_merge(merged[key], value)
        else:
            merged[key] = dict(value) if _is_mapping(value) else value
    return merged


def _parse_env_value(raw: str) -> Any:
    text = raw.strip()
    lowered = text.lower()
    if lowered in {"true", "yes", "on", "1"}:
        return True
    if lowered in {"false", "no", "off", "0"}:
        return False
    if lowered in {"null", "none"}:
        return None
    try:
        return int(text)
    except ValueError:
        pass
    try:
        return float(text)
    except ValueError:
        pass
    if text and text[0] in "{[\"":
        try:
            return json.loads(text)
        except json.JSONDecodeError:
            pass
    return raw


def _set_path(config: Dict[str, Any], parts: List[str], value: Any) -> None:
    current = config
    for part in parts[:-1]:
        if not isinstance(current.get(part), dict):
            current[part] = {}
        current = current[part]
    current[parts[-1]] = value


def _env_overrides(prefix: str) -> Dict[str, Any]:
    prefix = prefix.strip()
    if prefix.endswith("_"):
        prefix = prefix[:-1]
    if not prefix:
        return {}

    overrides: Dict[str, Any] = {}
    for name, value in os.environ.items():
        if name.startswith(prefix + "__"):
            path = name[len(prefix) + 2 :]
        elif name.startswith(prefix + "_"):
            path = name[len(prefix) + 1 :]
        else:
            continue

        if not path:
            continue

        if "__" in path:
            parts = [part for part in path.split("__") if part]
        else:
            parts = [path]

        if not parts:
            continue

        _set_path(overrides, parts, _parse_env_value(value))

    return overrides


def _get_path(config: Mapping[str, Any], path: str) -> Tuple[bool, Any]:
    current: Any = config
    for part in path.split("."):
        if isinstance(current, Mapping) and part in current:
            current = current[part]
        else:
            return False, None
    return True, current


def validate_config(config: Mapping[str, Any], required_keys: Iterable[str]) -> List[str]:
    errors: List[str] = []
    for key in required_keys:
        found, value = _get_path(config, key)
        if not found or value is None:
            errors.append(f"Missing required key: {key}")
    return errors


def load_config(
    path: Optional[Union[str, os.PathLike]] = "config.json",
    defaults: Optional[Mapping[str, Any]] = None,
