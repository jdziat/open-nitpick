from typing import Callable, Generic, List, Optional, TypeVar

T = TypeVar("T")


class PriorityQueue(Generic[T]):
    def __init__(self, compare: Callable[[T, T], object]):
        if not callable(compare):
            raise TypeError("compare must be callable")
        self._compare = compare
        self._heap: List[T] = []

    def _comes_before(self, left: T, right: T) -> bool:
        result = self._compare(left, right)
        if isinstance(result, bool):
            return result
        return result < 0

    def push(self, item: T) -> None:
        self._heap.append(item)
        index = len(self._heap) - 1

        while index > 0:
            parent = (index - 1) // 2
            if not self._comes_before(self._heap[index], self._heap[parent]):
                break
            self._heap[index], self._heap[parent] = (
                self._heap[parent],
                self._heap[index],
            )
            index = parent

    def peek(self) -> T:
        if not self._heap:
            raise IndexError("peek from empty priority queue")
        return self._heap[0]

    def pop(self) -> T:
        if not self._heap:
            raise IndexError("pop from empty priority queue")

        if len(self._heap) == 1:
            return self._heap.pop()

        result = self._heap[0]
        self._heap[0] = self._heap.pop()
        index = 0

        while True:
            left = 2 * index + 1
            right = left + 1
            best = index

            if (
                left < len(self._heap)
                and self._comes_before(self._heap[left], self._heap[best])
            ):
                best = left

            if (
                right < len(self._heap)
                and self._comes_before(self._heap[right], self._heap[best])
            ):
                best = right

            if best == index:
                break

            self._heap[index], self._heap[best] = (
                self._heap[best],
                self._heap[index],
            )
            index = best

        return result

    def __len__(self) -> int:
        return len(self._heap)

    def __bool__(self) -> bool:
        return bool(self._heap)

    def is_empty(self) -> bool:
        return not self._heap
