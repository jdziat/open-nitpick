def slugify(text: str) -> str:
    text = text.lower()
    parts = []
    pending_hyphen = False

    for char in text:
        if char.isalnum():
            if pending_hyphen and parts:
                parts.append("-")
            parts.append(char)
            pending_hyphen = False
        else:
            if parts:
                pending_hyphen = True

    return "".join(parts)
