import random
import time
from typing import Any, Callable, Optional, Union, TypeVar

T = TypeVar("T")
Result = Union[T, Exception]


def retry(
    func: Callable[..., T],
    *args: Any,
    attempts: int = 3,
    initial_delay: float = 1.0,
    backoff_factor: float = 2.0,
    max_delay: Optional[float] = 60.0,
    jitter: float = 0.1,
    **kwargs: Any,
) -> Result:
    """
    Call a function up to `attempts` times using exponential backoff and jitter.

    Returns the function's result on success, or the last exception if every
    attempt fails.
    """
    if not callable(func):
        raise TypeError("func must be callable")
    if attempts < 1:
        raise ValueError("attempts must be at least 1")
    if initial_delay < 0:
        raise ValueError("initial_delay must not be negative")
    if backoff_factor <= 0:
        raise ValueError("backoff_factor must be greater than zero")
    if max_delay is not None and max_delay < 0:
        raise ValueError("max_delay must not be negative")
    if jitter < 0:
        raise ValueError("jitter must not be negative")

    last_error: Optional[Exception] = None

    for attempt in range(attempts):
        try:
            return func(*args, **kwargs)
        except Exception as error:
            last_error = error

        if attempt == attempts - 1:
            break

        delay = initial_delay * (backoff_factor ** attempt)
        if max_delay is not None:
            delay = min(delay, max_delay)

        if jitter:
            delay += random.uniform(0, delay * jitter)
            if max_delay is not None:
                delay = min(delay, max_delay)

        if delay > 0:
            time.sleep(delay)

    return last_error  # type: ignore[return-value]
