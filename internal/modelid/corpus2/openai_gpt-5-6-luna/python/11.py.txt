class MatrixShapeError(ValueError):
    """Raised when matrix dimensions are incompatible."""


class Matrix:
    def __init__(self, rows):
        if isinstance(rows, Matrix):
            self._data = rows._data
            return

        try:
            data = [tuple(row) for row in rows]
        except TypeError as exc:
            raise TypeError("Matrix data must be an iterable of rows") from exc

        if data:
            columns = len(data[0])
            if any(len(row) != columns for row in data):
                raise MatrixShapeError("All matrix rows must have the same length")

        self._data = tuple(data)

    @property
    def rows(self):
        return len(self._data)

    @property
    def columns(self):
        return len(self._data[0]) if self._data else 0

    @property
    def shape(self):
        return (self.rows, self.columns)

    @property
    def data(self):
        return self._data

    def __getitem__(self, index):
        return self._data[index]

    def __len__(self):
        return self.rows

    def __repr__(self):
        return f"Matrix({[list(row) for row in self._data]!r})"

    def __eq__(self, other):
        if not isinstance(other, Matrix):
            return NotImplemented
        return self._data == other._data

    def __mul__(self, other):
        if not isinstance(other, Matrix):
            return NotImplemented

        if self.columns != other.rows:
            raise MatrixShapeError(
                f"Cannot multiply matrices with shapes "
                f"{self.shape} and {other.shape}"
            )

        result = []
        for i in range(self.rows):
            row = []
            for j in range(other.columns):
                value = 0
                for k in range(self.columns):
                    value += self[i][k] * other[k][j]
                row.append(value)
            result.append(row)

        return Matrix(result)

    def __matmul__(self, other):
        return self * other

    def multiply(self, other):
        return self * other

    def transpose(self):
        if not self._data:
            return Matrix([])

        return Matrix(
            [
                [self._data[i][j] for i in range(self.rows)]
                for j in range(self.columns)
            ]
        )

    @property
    def T(self):
        return self.transpose()

    def determinant(self):
        if self.rows != self.columns:
            raise MatrixShapeError(
                f"Determinant is only defined for square matrices, got {self.shape}"
            )

        n = self.rows
        if n == 0:
            return 1
        if n == 1:
            return self[0][0]

        matrix = [list(row) for row in self._data]
        sign = 1
        previous_pivot = 1

        for pivot_index in range(n - 1):
            pivot_row = pivot_index
            while (
                pivot_row < n
                and matrix[pivot_row][pivot_index] == 0
            ):
                pivot_row += 1

            if pivot_row == n:
                return 0

            if pivot_row != pivot_index:
                matrix[pivot_index], matrix[pivot_row] = (
                    matrix[pivot_row],
                    matrix[pivot_index],
                )
                sign = -sign

            pivot = matrix[pivot_index][pivot_index]

            for i in range(pivot_index + 1, n):
                for j in range(pivot_index + 1, n):
                    numerator = (
                        matrix[i][j] * pivot
                        - matrix[i][pivot_index] * matrix[pivot_index][j]
                    )

                    if (
                        isinstance(numerator, int)
                        and isinstance(previous_pivot, int)
                        and previous_pivot != 0
                        and numerator % previous_pivot == 0
                    ):
                        matrix[i][j] = numerator // previous_pivot
                    else:
                        matrix[i][j] = numerator / previous_pivot

            previous_pivot = pivot

        return sign * matrix[n - 1][n - 1]

    def det(self):
        return self.determinant()
