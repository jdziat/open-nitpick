package ratelimiter

import (
	"context"
	"sync"
	"time"
)

type Limiter struct {
	mu               sync.Mutex
	capacity         float64
	refillPerSecond  float64
	tokens           float64
	lastRefill       time.Time
}

func New(capacity int, refillPerSecond float64) *Limiter {
	if capacity <= 0 {
		panic("ratelimiter: capacity must be greater than zero")
	}
	if refillPerSecond <= 0 {
		panic("ratelimiter: refill rate must be greater than zero")
	}

	now := time.Now()
	return &Limiter{
		capacity:        float64(capacity),
		refillPerSecond: refillPerSecond,
		tokens:          float64(capacity),
		lastRefill:      now,
	}
}

func (l *Limiter) Allow() bool {
	return l.AllowN(1)
}

func (l *Limiter) AllowN(n int) bool {
	if n <= 0 {
		return true
	}

	required := float64(n)

	l.mu.Lock()
	defer l.mu.Unlock()

	if required > l.capacity {
		return false
	}

	l.refill(time.Now())

	if l.tokens < required {
		return false
	}

	l.tokens -= required
	return true
}

func (l *Limiter) Wait(ctx context.Context) error {
	return l.WaitN(ctx, 1)
}

func (l *Limiter) WaitN(ctx context.Context, n int) error {
	if n <= 0 {
		return nil
	}
	if ctx == nil {
		panic("ratelimiter: nil context")
	}

	required := float64(n)
	if required > l.capacity {
		return ErrRequestTooLarge
	}

	for {
		l.mu.Lock()
		l.refill(time.Now())

		if l.tokens >= required {
			l.tokens -= required
			l.mu.Unlock()
			return nil
		}

		wait := time.Duration((required-l.tokens)/l.refillPerSecond*float64(time.Second))
		if wait <= 0 {
			wait = time.Nanosecond
		}
		l.mu.Unlock()

		timer := time.NewTimer(wait)
		select {
		case <-ctx.Done():
			if !timer.Stop() {
				<-timer.C
			}
			return ctx.Err()
		case <-timer.C:
		}
	}
}

func (l *Limiter) refill(now time.Time) {
	elapsed := now.Sub(l.lastRefill).Seconds()
	if elapsed <= 0 {
		return
	}

	l.tokens += elapsed * l.refillPerSecond
	if l.tokens > l.capacity {
		l.tokens = l.capacity
	}
	l.lastRefill = now
}
