package retry

import (
	"context"
	"errors"
	"math/rand"
	"sync"
	"time"
)

const maxDuration = time.Duration(1<<63 - 1)

var (
	randomMu sync.Mutex
	random   = rand.New(rand.NewSource(time.Now().UnixNano()))
)

func Retry(attempts int, initialBackoff, maxBackoff time.Duration, fn func() error) error {
	return RetryContext(context.Background(), attempts, initialBackoff, maxBackoff, fn)
}

func RetryContext(ctx context.Context, attempts int, initialBackoff, maxBackoff time.Duration, fn func() error) error {
	if attempts <= 0 {
		return errors.New("retry: attempts must be greater than zero")
	}
	if fn == nil {
		return errors.New("retry: nil function")
	}
	if ctx == nil {
		ctx = context.Background()
	}
	if initialBackoff < 0 {
		initialBackoff = 0
	}
	if maxBackoff < 0 {
		maxBackoff = 0
	}
	if maxBackoff > 0 && initialBackoff > maxBackoff {
		initialBackoff = maxBackoff
	}

	backoff := initialBackoff
	var lastErr error

	for attempt := 0; attempt < attempts; attempt++ {
		if err := ctx.Err(); err != nil {
			return err
		}

		lastErr = fn()
		if lastErr == nil {
			return nil
		}
		if attempt == attempts-1 {
			break
		}

		delay := jitter(backoff, maxBackoff)
		if delay > 0 {
			timer := time.NewTimer(delay)
			select {
			case <-ctx.Done():
				if !timer.Stop() {
					<-timer.C
				}
				return ctx.Err()
			case <-timer.C:
			}
		}

		if backoff == 0 {
			continue
		}
		if backoff > maxDuration/2 {
			backoff = maxDuration
		} else {
			backoff *= 2
		}
		if maxBackoff > 0 && backoff > maxBackoff {
			backoff = maxBackoff
		}
	}

	return lastErr
}

func jitter(delay, maxBackoff time.Duration) time.Duration {
	if delay <= 0 {
		return 0
	}
	if maxBackoff > 0 && delay > maxBackoff {
		delay = maxBackoff
	}
	if delay > maxDuration/2 {
		delay = maxDuration / 2
	}

	randomMu.Lock()
	offset := random.Int63n(int64(delay))
	randomMu.Unlock()

	result := delay/2 + time.Duration(offset)
	if maxBackoff > 0 && result > maxBackoff {
		return maxBackoff
	}
	return result
}
