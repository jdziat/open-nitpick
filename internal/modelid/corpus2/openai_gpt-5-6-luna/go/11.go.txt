package matrix

import (
	"errors"
	"fmt"
	"math"
)

var ErrShapeMismatch = errors.New("matrix shape mismatch")

type Matrix [][]float64

func (m Matrix) shape() (int, int, error) {
	rows := len(m)
	if rows == 0 {
		return 0, 0, nil
	}

	cols := len(m[0])
	for _, row := range m[1:] {
		if len(row) != cols {
			return 0, 0, ErrShapeMismatch
		}
	}
	return rows, cols, nil
}

func (m Matrix) Multiply(other Matrix) (Matrix, error) {
	rows, cols, err := m.shape()
	if err != nil {
		return nil, err
	}

	otherRows, otherCols, err := other.shape()
	if err != nil {
		return nil, err
	}

	if cols != otherRows {
		return nil, ErrShapeMismatch
	}

	result := make(Matrix, rows)
	for i := range result {
		result[i] = make([]float64, otherCols)
		for j := 0; j < otherCols; j++ {
			for k := 0; k < cols; k++ {
				result[i][j] += m[i][k] * other[k][j]
			}
		}
	}

	return result, nil
}

func Multiply(a, b Matrix) (Matrix, error) {
	return a.Multiply(b)
}

func (m Matrix) Mul(other Matrix) (Matrix, error) {
	return m.Multiply(other)
}

func (m Matrix) Transpose() Matrix {
	rows, cols, err := m.shape()
	if err != nil {
		return nil
	}

	result := make(Matrix, cols)
	for i := range result {
		result[i] = make([]float64, rows)
		for j := 0; j < rows; j++ {
			result[i][j] = m[j][i]
		}
	}

	return result
}

func (m Matrix) Determinant() (float64, error) {
	rows, cols, err := m.shape()
	if err != nil {
		return 0, err
	}
	if rows != cols {
		return 0, fmt.Errorf("%w: determinant requires a square matrix", ErrShapeMismatch)
	}
	if rows == 0 {
		return 1, nil
	}

	a := make([][]float64, rows)
	for i := range m {
		a[i] = append([]float64(nil), m[i]...)
	}

	determinant := 1.0
	sign := 1.0

	for i := 0; i < rows; i++ {
		pivot := i
		maxValue := math.Abs(a[i][i])
		for r := i + 1; r < rows; r++ {
			if value := math.Abs(a[r][i]); value > maxValue {
				maxValue = value
				pivot = r
			}
		}

		if maxValue == 0 {
			return 0, nil
		}

		if pivot != i {
			a[i], a[pivot] = a[pivot], a[i]
			sign = -sign
		}

		pivotValue := a[i][i]
		determinant *= pivotValue

		for r := i + 1; r < rows; r++ {
			factor := a[r][i] / pivotValue
			for c := i + 1; c < rows; c++ {
				a[r][c] -= factor * a[i][c]
			}
		}
	}

	return determinant * sign, nil
}

func Determinant(m Matrix) (float64, error) {
	return m.Determinant()
}
