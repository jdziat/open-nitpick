import { readFileSync } from "node:fs";
import { resolve } from "node:path";

export interface ConfigLoaderOptions<T extends Record<string, unknown>> {
  configPath?: string;
  defaults?: Partial<T>;
  envPrefix?: string;
  env?: NodeJS.ProcessEnv;
  requiredKeys?: string[];
}

type AnyRecord = Record<string, unknown>;

const hasOwn = (value: object, key: PropertyKey): boolean =>
  Object.prototype.hasOwnProperty.call(value, key);

const isPlainObject = (value: unknown): value is AnyRecord =>
  typeof value === "object" &&
  value !== null &&
  !Array.isArray(value) &&
  Object.getPrototypeOf(value) === Object.prototype;

function clone<T>(value: T): T {
  if (Array.isArray(value)) {
    return value.map(clone) as T;
  }

  if (isPlainObject(value)) {
    const result: AnyRecord = {};
    for (const [key, child] of Object.entries(value)) {
      result[key] = clone(child);
    }
    return result as T;
  }

  return value;
}

function merge<T extends AnyRecord>(base: T, ...sources: AnyRecord[]): T {
  const result = clone(base);

  for (const source of sources) {
    for (const [key, value] of Object.entries(source)) {
      if (isPlainObject(value) && isPlainObject(result[key])) {
        result[key] = merge(result[key] as AnyRecord, value);
      } else {
        result[key] = clone(value);
      }
    }
  }

  return result;
}

function normalizeKey(key: string): string {
  return key.replace(/[-_]/g, "").toLowerCase();
}

function camelCase(key: string): string {
  return key
    .toLowerCase()
    .replace(/[-_]+([a-z0-9])/g, (_, character: string) =>
      character.toUpperCase()
    );
}

function findExistingKey(object: AnyRecord, segment: string): string {
  if (hasOwn(object, segment)) {
    return segment;
  }

  const normalized = normalizeKey(segment);
  const existing = Object.keys(object).find(
    (key) => normalizeKey(key) === normalized
  );

  return existing ?? camelCase(segment);
}

function getPath(object: AnyRecord, path: string[]): unknown {
  let current: unknown = object;

  for (const segment of path) {
    if (!isPlainObject(current)) {
      return undefined;
    }

    const key = findExistingKey(current, segment);
    if (!hasOwn(current, key)) {
      return undefined;
    }

    current = current[key];
  }

  return current;
}

function setPath(object: AnyRecord, path: string[], value: unknown): void {
  let current = object;

  for (let index = 0; index < path.length - 1; index++) {
    const segment = path[index];
    if (segment === "__proto__" || segment === "constructor" || segment === "prototype") {
      throw new Error(`Invalid configuration key: ${path.join(".")}`);
    }

    const key = findExistingKey(current, segment);

    if (!isPlainObject(current[key])) {
      current[key] = {};
    }

    current = current[key] as AnyRecord;
  }

  const finalSegment = path[path.length - 1];
  if (
    finalSegment === "__proto__" ||
    finalSegment === "constructor" ||
    finalSegment === "prototype"
  ) {
    throw new Error(`Invalid configuration key: ${path.join(".")}`);
  }

  current[findExistingKey(current, finalSegment)] = value;
}

function parseEnvironmentValue(raw: string, current: unknown): unknown {
  if (typeof current === "boolean") {
    const normalized = raw.trim().toLowerCase();
    if (["true", "1", "yes", "on"].includes(normalized)) return true;
    if (["false", "0", "no", "off"].includes(normalized)) return false;
    throw new Error(`Invalid boolean environment value: "${raw}"`);
  }

  if (typeof current === "number") {
    const value = Number(raw);
    if (!Number.isFinite(value)) {
      throw new Error(`Invalid numeric environment value: "${raw}"`);
    }
    return value;
  }

  if (typeof current === "bigint") {
    try {
      return BigInt(raw);
    } catch {
      throw new Error(`Invalid bigint environment value: "${raw}"`);
    }
  }

  if (isPlainObject(current) || Array.isArray(current)) {
    try {
      return JSON.parse(raw);
    } catch {
      throw new Error(`Invalid JSON environment value: "${raw}"`);
    }
  }

  if (current === null || current === undefined) {
    try {
      return JSON.parse(raw);
    } catch {
      return raw;
    }
  }

  return raw;
}

function isMissing(value: unknown): boolean {
  return (
    value === undefined ||
    value === null ||
    (typeof value === "string" && value.trim() === "")
  );
}

function readConfigurationFile(path: string): AnyRecord {
  let contents: string;

  try {
    contents = readFileSync(resolve(path), "utf8");
  } catch (error) {
    const message = error instanceof Error ? error.message : String(error);
    throw new Error(`Unable to read configuration file "${path}": ${message}`);
  }

  let parsed: unknown;

  try {
    parsed = JSON.parse(contents);
  } catch (error) {
    const message = error instanceof Error ? error.message : String(error);
    throw new Error(`Invalid JSON in configuration file "${path}": ${message}`);
  }

  if (!isPlainObject(parsed)) {
    throw new Error("Configuration file must contain a JSON object");
  }

  return parsed;
}

export function loadConfig<T extends Record<string, unknown>>(
  options: ConfigLoaderOptions<T> = {}
): T {
  const configPath = options.configPath ?? "config.json";
  const envPrefix = options.envPrefix ?? "APP_";
  const environment = options.env ?? process.env;

  const fileConfig = readConfigurationFile(configPath);
  let config = merge(
    (options.defaults ?? {}) as AnyRecord,
    fileConfig
  );

  for (const [name, rawValue] of Object.entries(environment)) {
    if (rawValue === undefined || !name.startsWith(envPrefix)) {
      continue;
    }

    const key = name.slice(envPrefix.length);
    if (!key) {
      continue;
    }

    const path = key.split("__").map((segment) => segment.trim()).filter(Boolean);
    if (path.length === 0) {
      continue;
    }

    const currentValue = getPath(config, path);
    let value: unknown;

    try {
      value = parseEnvironmentValue(rawValue, currentValue);
    } catch (error) {
      const message = error instanceof Error ? error.message : String(error);
      throw new Error(`Invalid environment variable "${name}": ${message}`);
    }

    setPath(config, path, value);
  }

  const missing = (options.requiredKeys ?? []).filter((key) => {
    const path = key.split(".").filter(Boolean);
    return path.length === 0 || isMissing(getPath(config, path));
  });

  if (missing.length > 0) {
    throw new Error(`Missing required configuration keys: ${missing.join(", ")}`);
  }

  return config as T;
}

export default loadConfig;
