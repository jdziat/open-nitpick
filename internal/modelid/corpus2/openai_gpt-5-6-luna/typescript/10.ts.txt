export class URLShortener {
  private readonly entries = new Map<
    string,
    { url: string; expiresAt: number }
  >();
  private nextId = 0;

  constructor(private readonly defaultTtlMs = 24 * 60 * 60 * 1000) {
    if (!Number.isFinite(defaultTtlMs) || defaultTtlMs <= 0) {
      throw new Error("Default TTL must be a positive finite number.");
    }
  }

  shorten(url: string, ttlMs = this.defaultTtlMs): string {
    if (!url.trim()) {
      throw new Error("URL must not be empty.");
    }

    if (!Number.isFinite(ttlMs) || ttlMs <= 0) {
      throw new Error("TTL must be a positive finite number.");
    }

    const code = this.encode(this.nextId++);
    this.entries.set(code, {
      url,
      expiresAt: Date.now() + ttlMs,
    });

    return code;
  }

  resolve(code: string): string | null {
    const entry = this.entries.get(code);

    if (!entry) {
      return null;
    }

    if (Date.now() >= entry.expiresAt) {
      this.entries.delete(code);
      return null;
    }

    return entry.url;
  }

  private encode(value: number): string {
    const alphabet =
      "0123456789abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ";

    if (value === 0) {
      return alphabet[0];
    }

    let code = "";

    while (value > 0) {
      code = alphabet[value % alphabet.length] + code;
      value = Math.floor(value / alphabet.length);
    }

    return code;
  }
}
