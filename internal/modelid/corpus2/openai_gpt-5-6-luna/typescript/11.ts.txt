export type MatrixData = ReadonlyArray<ReadonlyArray<number>>;

export class MatrixShapeError extends Error {
  constructor(message: string) {
    super(message);
    this.name = "MatrixShapeError";
    Object.setPrototypeOf(this, new.target.prototype);
  }
}

export class Matrix {
  private readonly _data: number[][];

  constructor(data: MatrixData) {
    if (!Array.isArray(data)) {
      throw new TypeError("Matrix data must be an array of rows.");
    }

    const columnCount = data.length === 0 ? 0 : data[0].length;

    for (let rowIndex = 0; rowIndex < data.length; rowIndex++) {
      const row = data[rowIndex];

      if (!Array.isArray(row)) {
        throw new MatrixShapeError(`Row ${rowIndex} is not an array.`);
      }

      if (row.length !== columnCount) {
        throw new MatrixShapeError(
          `Matrix rows must have the same length; row 0 has length ${columnCount}, ` +
          `but row ${rowIndex} has length ${row.length}.`
        );
      }

      for (const value of row) {
        if (typeof value !== "number" || !Number.isFinite(value)) {
          throw new TypeError("Matrix entries must be finite numbers.");
        }
      }
    }

    this._data = data.map((row) => Array.from(row));
  }

  get rowCount(): number {
    return this._data.length;
  }

  get columnCount(): number {
    return this._data.length === 0 ? 0 : this._data[0].length;
  }

  get rows(): number {
    return this.rowCount;
  }

  get columns(): number {
    return this.columnCount;
  }

  get cols(): number {
    return this.columnCount;
  }

  get data(): MatrixData {
    return this._data.map((row) => row.slice());
  }

  get(row: number, column: number): number {
    if (
      !Number.isInteger(row) ||
      !Number.isInteger(column) ||
      row < 0 ||
      row >= this.rowCount ||
      column < 0 ||
      column >= this.columnCount
    ) {
      throw new RangeError("Matrix index is out of bounds.");
    }

    return this._data[row][column];
  }

  multiply(other: Matrix): Matrix {
    if (this.columnCount !== other.rowCount) {
      throw new MatrixShapeError(
        `Cannot multiply a ${this.rowCount}×${this.columnCount} matrix by ` +
        `a ${other.rowCount}×${other.columnCount} matrix: ` +
        `the left column count must equal the right row count.`
      );
    }

    const result: number[][] = Array.from(
      { length: this.rowCount },
      () => Array(other.columnCount).fill(0)
    );

    for (let i = 0; i < this.rowCount; i++) {
      for (let k = 0; k < this.columnCount; k++) {
        const value = this._data[i][k];
        for (let j = 0; j < other.columnCount; j++) {
          result[i][j] += value * other._data[k][j];
        }
      }
    }

    return new Matrix(result);
  }

  transpose(): Matrix {
    const result: number[][] = Array.from(
      { length: this.columnCount },
      () => Array(this.rowCount)
    );

    for (let i = 0; i < this.rowCount; i++) {
      for (let j = 0; j < this.columnCount; j++) {
        result[j][i] = this._data[i][j];
      }
    }

    return new Matrix(result);
  }

  determinant(): number {
    if (this.rowCount !== this.columnCount) {
      throw new MatrixShapeError(
        `Determinant is defined only for square matrices; received ` +
        `${this.rowCount}×${this.columnCount}.`
      );
    }

    const size = this.rowCount;
    if (size === 0) {
      return 1;
    }

    const values = this._data.map((row) => row.slice());
    let determinant = 1;

    for (let column = 0; column < size; column++) {
      let pivotRow = column;

      for (let row = column + 1; row < size; row++) {
        if (Math.abs(values[row][column]) > Math.abs(values[pivotRow][column])) {
          pivotRow = row;
        }
      }

      if (values[pivotRow][column] === 0) {
        return 0;
      }

      if (pivotRow !== column) {
        [values[column], values[pivotRow]] = [values[pivotRow], values[column]];
        determinant = -determinant;
      }

      const pivot = values[column][column];
      determinant *= pivot;

      for (let row = column + 1; row < size; row++) {
        const factor = values[row][column] / pivot;

        for (let currentColumn = column + 1; currentColumn < size; currentColumn++) {
          values[row][currentColumn] -= factor * values[column][currentColumn];
        }
      }
    }

    return determinant;
  }

  toArray(): number[][] {
    return this._data.map((row) => row.slice());
  }

  clone(): Matrix {
    return new Matrix(this._data);
  }

  static identity(size: number): Matrix {
    if (!Number.isInteger(size) || size < 0) {
      throw new RangeError("Identity matrix size must be a non-negative integer.");
    }

    const data = Array.from({ length: size }, (_, row) =>
      Array.from({ length: size }, (_, column) => row === column ? 1 : 0)
    );

    return new Matrix(data);
  }
}
