export interface DateRangeOptions {
  skipWeekends?: boolean;
}

function startOfDay(date: Date): Date {
  const result = new Date(date.getTime());
  result.setHours(0, 0, 0, 0);
  return result;
}

function isWeekend(date: Date): boolean {
  const day = date.getDay();
  return day === 0 || day === 6;
}

export function* dateRange(
  start: Date,
  end: Date,
  options: DateRangeOptions = {}
): IterableIterator<Date> {
  if (Number.isNaN(start.getTime()) || Number.isNaN(end.getTime())) {
    throw new RangeError("Start and end dates must be valid dates.");
  }

  const current = startOfDay(start);
  const final = startOfDay(end);
  const step = current.getTime() <= final.getTime() ? 1 : -1;

  while (
    step > 0
      ? current.getTime() <= final.getTime()
      : current.getTime() >= final.getTime()
  ) {
    if (!options.skipWeekends || !isWeekend(current)) {
      yield new Date(current.getTime());
    }

    current.setDate(current.getDate() + step);
  }
}

export default dateRange;
