    const form = new FormData()
    const key = 'prop'

    const data = new Blob()

    form.set(key, data)

    const retrievedBlob = form.get(key)

    expect(retrievedBlob).toBeInstanceOf(Blob)
  })

  test('supports File', () => {
    const form = new FormData()
    const key = 'prop'

    const data = new File([], 'name')

    form.set(key, data)

    const retrievedBlob = form.get(key)

    expect(retrievedBlob).toBeInstanceOf(File)
  })
