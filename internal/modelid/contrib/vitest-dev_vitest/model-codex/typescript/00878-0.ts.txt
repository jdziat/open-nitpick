function raceWith<A, B>(
  promise: Promise<A>,
  other?: Promise<B>,
): Promise<{ ok: true; value: A } | { ok: false; value: B }> {
  const left = promise.then(value => ({ ok: true as const, value }))
  if (!other) {
    return left
  }
  return Promise.race([
    left,
    other.then(value => ({ ok: false as const, value })),
  ])
}
