test('signal', async () => {

    'basic.test.ts': `

test('signal', async () => {
  let aborted = false
  await expect(

        aborted = true
      })
      return new Promise(() => {})
    }, { timeout: 100, interval: 10 }).toMatchKvSnapshot()
