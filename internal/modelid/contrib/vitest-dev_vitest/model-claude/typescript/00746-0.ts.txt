function groupByConsecutiveAction(mocks: PendingSuiteMock[]): PendingSuiteMock[][] {
  const groups: PendingSuiteMock[][] = []
  for (const mock of mocks) {
    const last = groups.at(-1)
    if (last?.[0].action === mock.action) {
      last.push(mock)
    }
    else {
      groups.push([mock])
    }
  }
  return groups
}
