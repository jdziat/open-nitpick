func allowsSharedCache(cc string) bool {
	shareable := false

	for part := range strings.SplitSeq(cc, ",") {
		part = strings.TrimSpace(utils.ToLower(part))
		switch {
		case part == "":
			continue
		case part == "private":
			return false
		case part == "public":
			shareable = true
		case strings.HasPrefix(part, "s-maxage="):
			shareable = true
		default:
			continue
		}
	}

	return shareable
}
