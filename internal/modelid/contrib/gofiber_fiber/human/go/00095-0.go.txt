	"sort"
type pair struct {
	k []string
	v []string
}

func (p *pair) Len() int {
	return len(p.k)
}

func (p *pair) Swap(i, j int) {
	p.k[i], p.k[j] = p.k[j], p.k[i]
	p.v[i], p.v[j] = p.v[j], p.v[i]
}

func (p *pair) Less(i, j int) bool {
	return p.k[i] < p.k[j]
