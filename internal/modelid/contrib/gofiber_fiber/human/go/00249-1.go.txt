func isToken68Char(c byte) bool {
	return (c >= 'A' && c <= 'Z') ||
		(c >= 'a' && c <= 'z') ||
		(c >= '0' && c <= '9') ||
		c == '-' || c == '.' || c == '_' || c == '~' || c == '+' || c == '/' || c == '='
}

				i := 1

					return "", ErrMissingOrMalformedAPIKey
				}
				token := rest[i:]

					return "", ErrMissingOrMalformedAPIKey
				}
				seenEq := false
