	 * MCP server. Issued tokens are audience-bound to it, and it is published as

const appendResourceIdentifier = (
	resources: readonly string[] | undefined,
	resource: string,
): readonly string[] =>
	resources?.includes(resource) ? resources : [...(resources ?? []), resource];
