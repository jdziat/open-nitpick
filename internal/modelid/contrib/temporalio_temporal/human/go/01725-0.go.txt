	if util == nil || util.tsi == nil {
		return nil
	}
	ff := util.tsi.GetFastForwardInfo()
	if ff == nil || ff.GetTargetTime() == nil {
		return nil
	}
	targetTime := ff.GetTargetTime().AsTime()
	if targetTime.IsZero() {
		return nil
	}
