def _prune_summary(revision_count: int, incomplete_count: int) -> str:

    parts: list[str] = []
    if revision_count:
        parts.append(f"{revision_count} unreferenced revision(s)")
    if incomplete_count:
        parts.append(f"{incomplete_count} incomplete download(s)")
    return " and ".join(parts)
