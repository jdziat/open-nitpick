        report = scan_cache_dir(tmp_path)
        assert len(report.repos) == 1  # Scan still worked !
        assert len(report.warnings) == 1
        assert (
            str(report.warnings[0])
            == f"Repo type must be `dataset`, `model` or `space`, found `not-model` ({repo_path})."
