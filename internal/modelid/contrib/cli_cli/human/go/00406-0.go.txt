func fromAPISession(s session) *Session {
	return &Session{
		ID:            s.ID,
		Name:          s.Name,
		UserID:        s.UserID,
		AgentID:       s.AgentID,
		Logs:          s.Logs,
		State:         s.State,
		OwnerID:       s.OwnerID,
		RepoID:        s.RepoID,
		ResourceType:  s.ResourceType,

		CreatedAt:     s.CreatedAt,
		CompletedAt:   s.CompletedAt,

		EventType:     s.EventType,
	}
}
