func runGitCommands(client *git.Client, commands [][]string) error {
	for _, args := range commands {
		cmd, err := client.Command(ctx.Background(), args...)
		if err != nil {
			return err
		}
		if _, err := cmd.Output(); err != nil {
			return err
		}
	}
	return nil
}
