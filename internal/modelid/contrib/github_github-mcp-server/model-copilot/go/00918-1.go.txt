func pullRequestUpdateHasNonFormParams(args map[string]any) bool {
	for key, value := range args {
		if value == nil {
			continue
		}
		if _, ok := pullRequestUpdateFormParams[key]; !ok {
			return true
		}
	}
	return false
}
