def _eventbus_name(agent_id: str, prefix: str = 'RustAgent') -> str:
	suffix = re.sub(r'\W', '_', str(agent_id)[-8:])
	if suffix and suffix[0].isdigit():
		suffix = 'a' + suffix
	return f'{prefix}_{suffix or "agent"}'
