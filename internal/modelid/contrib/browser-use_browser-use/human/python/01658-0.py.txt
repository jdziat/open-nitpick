								const maxScroll = currentElement.scrollHeight - currentElement.clientHeight;
								// Calculate scroll amount (1/3 of provided dy for gentler scrolling)
								let scrollAmount = dy / 3;
								// Ensure we don't scroll beyond bounds
								if (scrollAmount > 0) {
